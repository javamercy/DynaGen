import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead rollout heuristic.
    For each candidate, simulate a complete nearest-neighbour tour starting from
    the current node, visiting all unvisited nodes, and finally returning to the
    destination. Return the candidate that yields the smallest total tour length.
    """
    best_node = None
    best_total = float('inf')
    
    for candidate in unvisited_nodes:
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        last = candidate
        total = distance_matrix[current_node][candidate]
        
        # Greedy nearest neighbor simulation over all remaining nodes
        while remaining:
            nearest = min(remaining, key=lambda x: distance_matrix[last][x])
            total += distance_matrix[last][nearest]
            remaining.remove(nearest)
            last = nearest
        
        # Finally, return to the destination
        total += distance_matrix[last][destination_node]
        
        if total < best_total:
            best_total = total
            best_node = candidate
    
    return best_node
