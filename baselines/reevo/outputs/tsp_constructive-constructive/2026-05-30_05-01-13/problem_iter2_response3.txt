import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a full nearest-neighbor lookahead that incorporates
    the destination in the multi-step estimation.
    """
    unvisited = list(unvisited_nodes)
    best_node = None
    best_total = float('inf')

    # Number of unvisited nodes, used to decide simulation depth
    n_unvisited = len(unvisited)

    # If only one node remains, return it immediately (no lookahead needed)
    if n_unvisited == 1:
        return unvisited[0]

    # Candidate evaluation loop
    for candidate in unvisited:
        # Immediate cost from current node to candidate
        cost1 = distance_matrix[current_node][candidate]

        # Simulate a greedy nearest-neighbor path from candidate, visiting
        # all remaining nodes, and finally going to the destination.
        remaining = [n for n in unvisited if n != candidate]
        current = candidate
        simulated_cost = 0.0

        # Perform greedy nearest-neighbor on the remaining nodes
        while remaining:
            # Find the nearest unvisited node
            nearest_idx = np.argmin([distance_matrix[current][r] for r in remaining])
            next_node = remaining[nearest_idx]
            simulated_cost += distance_matrix[current][next_node]
            current = next_node
            remaining.pop(nearest_idx)

        # After visiting all nodes, go to the destination
        simulated_cost += distance_matrix[current][destination_node]

        total_cost = cost1 + simulated_cost

        if total_cost < best_total:
            best_total = total_cost
            best_node = candidate

    return best_node
