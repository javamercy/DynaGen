import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit with look-ahead, regret, savings, and statistical properties."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_score = float('inf')
    best_node = None

    for n in unvisited_nodes:
        remaining = unvisited_nodes - {n}

        # Statistical properties of distances from n to remaining nodes
        if remaining:
            dists = [distance_matrix[n][x] for x in remaining]
            mean_d = np.mean(dists)
            std_d = np.std(dists)

            # Look-ahead: average of the two smallest distances (2-nearest neighbor proxy)
            sorted_dists = sorted(dists)
            lookahead = (sorted_dists[0] + (sorted_dists[1] if len(sorted_dists) > 1 else sorted_dists[0])) / 2.0

            # Regret: difference between second best and best edge from n
            regret = sorted_dists[1] - sorted_dists[0] if len(sorted_dists) > 1 else 0.0
        else:
            mean_d = 0.0
            std_d = 0.0
            lookahead = 0.0
            regret = 0.0

        # Direct cost and destination proximity
        direct_cost = distance_matrix[current_node][n]
        dest_prox = distance_matrix[n][destination_node]

        # Clarke-Wright savings: larger is better
        savings = direct_cost + dest_prox - distance_matrix[current_node][destination_node]

        # Combined score (lower is better)
        score = (0.2 * direct_cost +
                 0.1 * dest_prox -
                 0.2 * savings -
                 0.15 * mean_d +
                 0.1 * std_d +
                 0.15 * lookahead -
                 0.2 * regret)

        if score < best_score:
            best_score = score
            best_node = n

    return best_node
