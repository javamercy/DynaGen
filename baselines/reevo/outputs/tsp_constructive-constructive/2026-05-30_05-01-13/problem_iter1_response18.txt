```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a look-ahead heuristic.
    Combines immediate cost, nearest neighbor look-ahead, distance to destination,
    regret (range of distances to remaining nodes) and standard deviation.
    """
    # Weights for the components
    w1 = 0.30  # immediate distance from current node
    w2 = 0.25  # look-ahead: minimum distance from candidate to remaining nodes
    w3 = 0.20  # distance from candidate to destination node
    w4 = 0.15  # regret: normalized range of distances (max-min)/max
    w5 = 0.10  # diversity: standard deviation of distances from candidate

    scores = {}
    for node in unvisited_nodes:
        # Immediate travel cost
        immediate = distance_matrix[current_node][node]

        # Compute look-ahead statistics from the candidate node to the rest
        remaining = unvisited_nodes - {node}
        if remaining:
            dists = [distance_matrix[node][v] for v in remaining]
            min_next = min(dists)
            max_next = max(dists)
            std_next = np.std(dists)
            # Regret: how much worse it could be if we postpone this node
            if max_next > 0:
                regret = (max_next - min_next) / max_next
            else:
                regret = 0.0
        else:
            min_next = 0.0
            std_next = 0.0
            regret = 0.0

        # Distance to the final destination (starting city)
        dest_dist = distance_matrix[node][destination_node]

        # Combined score (lower is better)
        score = (w1 * immediate +
                 w2 * min_next +
                 w3 * dest_dist -
                 w4 * regret -
                 w5 * std_next)

        scores[node] = score

    return min(scores, key=scores.get)
```
