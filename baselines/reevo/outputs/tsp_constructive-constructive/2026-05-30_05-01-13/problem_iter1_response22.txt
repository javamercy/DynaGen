```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead heuristic.
    For each candidate, evaluate the total cost of a two-step path:
    current -> candidate -> best_next (nearest unvisited after candidate)
    plus the direct distance from best_next to the destination.
    This simulates the immediate future and prioritizes nodes that lead to compact routes.
    """
    best_node = None
    best_score = float('inf')
    
    for candidate in unvisited_nodes:
        # cost to go from current to candidate
        cost1 = distance_matrix[current_node][candidate]
        
        remaining = unvisited_nodes - {candidate}
        if not remaining:
            # candidate is the last node: must go to destination
            cost2 = distance_matrix[candidate][destination_node]
            total = cost1 + cost2
        else:
            # find the nearest unvisited node from candidate (excluding candidate itself)
            best_next = min(remaining, key=lambda n: distance_matrix[candidate][n])
            cost2 = distance_matrix[candidate][best_next]
            # estimate future cost from best_next to destination (direct)
            cost3 = distance_matrix[best_next][destination_node]
            total = cost1 + cost2 + cost3
        
        if total < best_score:
            best_score = total
            best_node = candidate
    
    return best_node
```
