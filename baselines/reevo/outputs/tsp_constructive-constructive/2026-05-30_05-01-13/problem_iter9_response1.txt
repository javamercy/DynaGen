import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a constructive heuristic that adapts from exploration to exploitation.
    
    Combines:
    - distance from current node (greedy)
    - distance to destination (end-game)
    - savings (Clarke-Wright style)
    - regret (penalty if node is more expensive now than its alternative)
    - connectivity (average distance to other unvisited nodes)
    - 2-step lookahead (current->candidate->best_follower->destination)
    Weights shift from exploration (emphasising savings, regret, connectivity) to exploitation (emphasising direct distances) as unvisited set shrinks.
    """
    if not unvisited_nodes:
        return None  # should not happen in normal usage

    unvisited = list(unvisited_nodes)
    n_unvisited = len(unvisited)
    total_estimate = n_unvisited + 2  # current and destination are considered visited
    # exploration factor: high when many nodes remain
    alpha = n_unvisited / total_estimate

    # Precompute raw terms for all candidates to enable minimal normalization
    raw_data = {}
    for node in unvisited:
        d_cur = distance_matrix[current_node, node]
        d_dest = distance_matrix[node, destination_node]

        # Other unvisited nodes (excluding node itself)
        others = [j for j in unvisited if j != node]
        # Connectivity: average distance to others + destination (include destination for relevance)
        if others:
            avg_conn = np.mean([distance_matrix[node, j] for j in others] + [d_dest])
        else:
            avg_conn = d_dest  # only destination left

        # Regret-like: how much cheaper it is to connect node to its nearest other unvisited than to current
        if others:
            min_other = min([distance_matrix[node, j] for j in others])
        else:
            min_other = d_dest  # only destination
        regret = d_cur - min_other  # positive -> postpone is better

        # Savings (Clarke-Wright): saving by merging edges (current->dest) and (node->dest) into one edge (current->node)
        savings = distance_matrix[current_node, destination_node] + d_dest - d_cur

        # 2-step lookahead: go to node, then to best follower, then to destination
        if others:
            best_follower = min(others, key=lambda j: distance_matrix[node, j] + distance_matrix[j, destination_node])
            lookahead = distance_matrix[node, best_follower] + distance_matrix[best_follower, destination_node]
        else:
            lookahead = d_dest  # direct to destination

        raw_data[node] = (d_cur, d_dest, avg_conn, regret, savings, lookahead)

    # Extract arrays for normalization
    d_cur_vals = np.array([raw_data[n][0] for n in unvisited])
    d_dest_vals = np.array([raw_data[n][1] for n in unvisited])
    avg_conn_vals = np.array([raw_data[n][2] for n in unvisited])
    regret_vals = np.array([raw_data[n][3] for n in unvisited])
    savings_vals = np.array([raw_data[n][4] for n in unvisited])
    lookahead_vals = np.array([raw_data[n][5] for n in unvisited])

    # Compute maxima (avoid division by zero)
    max_dcur = max(d_cur_vals.max(), 1e-9)
    max_ddest = max(d_dest_vals.max(), 1e-9)
    max_avg = max(avg_conn_vals.max(), 1e-9)
    max_reg = max(abs(regret_vals).max(), 1e-9)  # absolute for sign?
    max_sav = max(savings_vals.max(), 1e-9)
    max_look = max(lookahead_vals.max(), 1e-9)

    # Weights: exploitation terms dominate when alpha is small
    w_cur = 0.3 + 0.4 * (1 - alpha)
    w_dest = 0.2 + 0.3 * (1 - alpha)
    w_look = 0.5                 # always important
    # Exploration terms: fade out with alpha
    w_sav = 0.25 * alpha         # savings (high savings good -> subtract)
    w_conn = 0.15 * alpha        # connectivity (high avg -> isolated -> take now)
    w_reg = 0.10 * alpha         # regret (positive regret -> bad -> add)

    best_score = None
    best_node = None
    for node in unvisited:
        d_cur, d_dest, avg_conn, regret, savings, lookahead = raw_data[node]

        # Normalise each term to [0,1] using max values
        nd_cur = d_cur / max_dcur
        nd_dest = d_dest / max_ddest
        nd_avg = avg_conn / max_avg
        nd_reg = max(0, regret) / max_reg   # only positive regret matters
        nd_sav = savings / max_sav
        nd_look = lookahead / max_look

        # Final score: lower is better
        score = (w_cur * nd_cur +
                 w_dest * nd_dest +
                 w_look * nd_look +
                 w_sav * (-nd_sav) +            # high savings reduces score
                 w_conn * (-nd_avg) +           # high avg (bad connectivity) reduces score
                 w_reg * nd_reg)                # positive regret increases score

        if best_score is None or score < best_score:
            best_score = score
            best_node = node

    return best_node
