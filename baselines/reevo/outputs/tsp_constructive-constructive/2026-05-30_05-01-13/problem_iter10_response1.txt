```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved constructive heuristic for TSP.

    Combines Clarke-Wright savings, regret (distance to nearest other node or destination),
    and future cost (distance to nearest unvisited or destination).
    All components are normalized to [0,1]; weights adapt based on the remaining fraction
    of nodes to visit.
    """
    # Base case: only one candidate
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    nodes = np.array(list(unvisited_nodes), dtype=int)
    n = len(nodes)
    total_nodes = distance_matrix.shape[0]

    # Remaining fraction (scaled from ~1.0 when many nodes left to 0.0 when one left)
    remaining_frac = n / (total_nodes - 1) if total_nodes > 1 else 1.0

    # Adaptive weights: savings more important early, regret and future cost later
    w_sav = 0.6 * (1 - remaining_frac) + 0.3          # range [0.3, 0.9]
    w_reg = 0.4 * remaining_frac + 0.1                # range [0.1, 0.5]
    w_fut = 0.2 + 0.2 * remaining_frac                # range [0.2, 0.4]

    # ---- Compute components ----
    d_cur = distance_matrix[current_node, nodes]
    d_dest = distance_matrix[destination_node, nodes]
    base_saving = distance_matrix[current_node, destination_node]
    savings = base_saving - d_cur - d_dest

    # Build distance matrix from each candidate to all other unvisited + destination
    inter = distance_matrix[nodes[:, None], nodes]          # (n, n)
    np.fill_diagonal(inter, np.inf)                         # ignore self
    dest_col = distance_matrix[nodes, destination_node][:, None]  # (n, 1)
    full = np.hstack((inter, dest_col))                     # (n, n+1)

    # Two smallest distances per candidate -> best and second-best
    part = np.partition(full, 1, axis=1)
    best = part[:, 0]
    second_best = part[:, 1]
    regret = second_best - best
    future_cost = best

    # ---- Normalize each component to [0,1] ----
    def normalize(x):
        lo, hi = np.min(x), np.max(x)
        if hi - lo < 1e-12:                # all equal → zero vector
            return np.zeros_like(x)
        return (x - lo) / (hi - lo)

    savings_norm = normalize(savings)
    regret_norm  = normalize(regret)
    future_norm  = normalize(future_cost)

    # ---- Composite score (maximize) ----
    scores = w_sav * savings_norm + w_reg * regret_norm - w_fut * future_norm
    best_idx = int(np.argmax(scores))
    return nodes[best_idx]
```
