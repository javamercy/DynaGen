import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    # If only one node remains, it must be the next
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    n_total = distance_matrix.shape[0]
    frac_remaining = len(unvisited_nodes) / n_total
    
    # Adaptive weights: emphasize savings and regret when many nodes remain,
    # emphasize destination proximity when few remain.
    w_direct = 0.4
    w_dest = 0.1 + 0.3 * (1.0 - frac_remaining)           # grows as remaining shrinks
    w_connect = 0.1
    w_savings = 0.2 * frac_remaining                       # larger when many left
    w_regret = 0.2 * (1.0 - frac_remaining)                # larger when few left, but adjust

    # Precompute distances from current to destination (constant)
    dist_curr_dest = distance_matrix[current_node][destination_node]

    scores = {}
    cost2_values = {}
    remain_set = unvisited_nodes

    # First compute the 2-step look-ahead costs for regret
    for node in remain_set:
        imm = distance_matrix[current_node][node]
        remaining = remain_set - {node}
        if remaining:
            # best continuation from node to a remaining city and then to destination
            best_next = min(
                distance_matrix[node][j] + distance_matrix[j][destination_node]
                for j in remaining
            )
            cost2 = imm + best_next
        else:
            cost2 = imm + distance_matrix[node][destination_node]
        cost2_values[node] = cost2

    # Find best and second best cost2
    cost2_list = list(cost2_values.values())
    cost2_list.sort()
    best_cost2 = cost2_list[0]
    second_best_cost2 = cost2_list[1] if len(cost2_list) > 1 else best_cost2

    # Now compute scores
    for node in remain_set:
        imm = distance_matrix[current_node][node]
        dest = distance_matrix[node][destination_node]

        # Connectivity: average distance to other unvisited nodes
        rem = remain_set - {node}
        if rem:
            avg_dist_others = np.mean([distance_matrix[node][j] for j in rem])
        else:
            avg_dist_others = 0.0

        # Savings (Clarke-Wright style)
        savings = dist_curr_dest + dest - imm

        # Regret based on 2-step look-ahead gap
        cost2_node = cost2_values[node]
        if cost2_node == best_cost2:
            regret = second_best_cost2 - best_cost2
        else:
            regret = 0.0

        # Composite score (maximize)
        score = -w_direct * imm - w_dest * dest + w_connect * (-avg_dist_others) + w_savings * savings + w_regret * regret
        scores[node] = score

    # Return node with maximum score
    return max(scores, key=scores.get)
