```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node using a simple adaptive scoring:
    - immediate travel cost from current node
    - look‑ahead: average of the two shortest distances from the candidate to
      other unvisited nodes
    - proximity to the final destination

    Weights shift smoothly from look‑ahead‑heavy when many nodes remain to
    destination‑heavy when few nodes remain.
    """
    if not unvisited_nodes:
        raise ValueError("No unvisited nodes")

    n_remaining = len(unvisited_nodes)
    best_score = float('inf')
    best_node = None

    # Adaptive weights: w_look + w_dest = 1.0
    # They transition smoothly as n_remaining changes.
    w_look = 0.8 * n_remaining / (n_remaining + 5)
    w_dest = 0.2 + 0.8 * 5 / (n_remaining + 5)

    for node in unvisited_nodes:
        # 1. Immediate travel cost
        immediate = distance_matrix[current_node, node]

        # 2. Look‑ahead: average of the two shortest edges from candidate
        #    to other unvisited nodes
        others = list(unvisited_nodes - {node})
        m = len(others)
        if m >= 2:
            dists = distance_matrix[node, others]
            # np.partition is more efficient if m is large, but for small m sort is fine
            sorted_dists = np.sort(dists)
            look_ahead = (sorted_dists[0] + sorted_dists[1]) / 2.0
        elif m == 1:
            look_ahead = distance_matrix[node, others[0]]
        else:  # no other nodes (should not happen because current_node not in unvisited)
            look_ahead = 0.0

        # 3. Proximity to final destination
        dest_cost = distance_matrix[destination_node, node]

        # Composite cost (lower is better)
        total = immediate + w_look * look_ahead + w_dest * dest_cost

        if total < best_score:
            best_score = total
            best_node = node

    return best_node
```
