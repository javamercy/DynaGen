import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node using a two-step lookahead with strong regret.
    The regret term (difference between second best and best next distance)
    is weighted fully to penalise choices that force a much worse second step.
    The weights are adjusted based on the number of remaining nodes to
    balance exploration (lookahead) and destination proximity.
    """
    n_remaining = len(unvisited_nodes)
    best_score = float('inf')
    best_node = None

    for node in unvisited_nodes:
        # 1. Immediate cost from current to candidate
        immediate = distance_matrix[current_node, node]

        # 2. Lookahead and regret using remaining nodes after candidate
        remaining = list(unvisited_nodes - {node})
        m = len(remaining)

        if m >= 2:
            # Distances from candidate to all other unvisited nodes
            dists = distance_matrix[node, remaining]
            sorted_dists = np.sort(dists)
            best_next = sorted_dists[0]
            second_best = sorted_dists[1]
            # Regret term: full difference to penalise wasteful choices
            regret = second_best - best_next
            lookahead = best_next + regret  # strong penalty
        elif m == 1:
            lookahead = distance_matrix[node, remaining[0]]
        else:
            lookahead = 0.0

        # 3. Destination proximity (distance from candidate to final destination)
        dest_cost = distance_matrix[destination_node, node]

        # 4. Adaptive weighting: when few nodes remain, destination matters more
        if n_remaining > 5:
            # Early stage: focus on lookahead, slight regard for destination
            total = immediate + 0.8 * lookahead + 0.2 * dest_cost
        elif n_remaining > 2:
            # Middle stage: balanced
            total = immediate + 0.6 * lookahead + 0.4 * dest_cost
        else:
            # Late stage: dest proximity critical
            total = immediate + 0.4 * lookahead + 0.6 * dest_cost

        if total < best_score:
            best_score = total
            best_node = node

    return best_node
