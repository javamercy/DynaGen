```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """Select next node using 2-step lookahead and savings."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = float('inf')

    for j in unvisited_nodes:
        # Immediate cost
        immediate = distance_matrix[current_node][j]

        # Lookahead: best distance from j to any other unvisited node or destination
        lookahead = float('inf')
        for k in unvisited_nodes:
            if k != j:
                lookahead = min(lookahead, distance_matrix[j][k])
        # Also consider destination as possible next step
        lookahead = min(lookahead, distance_matrix[j][destination_node])

        # Savings: how much we save by going j before destination
        direct = distance_matrix[current_node][destination_node]
        via_j = distance_matrix[current_node][j] + distance_matrix[j][destination_node]
        savings = direct - via_j   # positive if via_j is shorter

        # Score combines immediate, lookahead, and savings (negative because high savings is good)
        score = immediate + lookahead - 0.5 * savings

        if score < best_score:
            best_score = score
            best_node = j

    return best_node
```
