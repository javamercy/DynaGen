import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """Select the next node using adaptive multi-criteria weighting.

    Combines insertion savings, regret (cost of not taking the best edge),
    connectivity (isolation), direct cost, and destination proximity.
    Weights are adjusted based on the number of remaining nodes.
    """
    n_remaining = len(unvisited_nodes)
    if n_remaining == 1:
        return next(iter(unvisited_nodes))

    # Adaptive weights: early vs late stage
    if n_remaining > 5:
        w_sav = 0.30
        w_reg = 0.25
        w_con = 0.20
        w_direct = 0.15
        w_dest = 0.10
    else:
        w_sav = 0.20
        w_reg = 0.30
        w_con = 0.10
        w_direct = 0.20
        w_dest = 0.20

    # Set of future nodes (unvisited + destination) for regret/min_next
    future_nodes = unvisited_nodes | {destination_node}

    scores = {}
    for node in unvisited_nodes:
        # Direct cost from current to candidate (lower is better -> negative)
        direct = distance_matrix[current_node, node]
        # Destination proximity (lower is better -> negative)
        dest = distance_matrix[node, destination_node]
        # Insertion savings: dist(current,dest) - (direct + dest)
        savings = distance_matrix[current_node, destination_node] - direct - dest

        # Regret and minimum next move (using distances to all future nodes)
        dists = sorted([
            distance_matrix[node, f]
            for f in future_nodes
            if f != node
        ])
        if len(dists) >= 2:
            regret = dists[1] - dists[0]   # difference between second best and best
            min_next = dists[0]
        elif len(dists) == 1:
            regret = dists[0]
            min_next = dists[0]
        else:  # should not happen because at least destination exists
            regret = 0.0
            min_next = 0.0

        # Connectivity: average distance to other unvisited nodes (isolation)
        others = unvisited_nodes - {node}
        if others:
            connectivity = np.mean([distance_matrix[node, o] for o in others])
        else:
            connectivity = 0.0

        # Composite score (maximise)
        score = (w_sav * savings
                 + w_reg * regret
                 + w_con * connectivity
                 + w_direct * (-direct)
                 + w_dest * (-dest)
                 - 0.1 * min_next)
        scores[node] = score

    return max(scores, key=scores.get)
