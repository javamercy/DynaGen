```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Simplified next‑node selection with aggressive weight shifting.

    Uses only two components:
      - Clarke‑Wright savings (higher is better)
      - Distance to destination (lower is better, penalised)

    The weight on savings dominates when many nodes remain; the weight on
    destination proximity grows sharply as few nodes are left. No per‑iteration
    normalisation is performed – raw distances are scaled by the global maximum
    distance to keep values in a comparable range.
    """
    if not unvisited_nodes:
        raise ValueError("No unvisited nodes")
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n = len(unvisited_nodes)
    # Aggressive shift: weight on savings decreases quickly when n is small
    # t ∈ [0,1], close to 1 for large n, close to 0 for n=0
    t = n / (n + 2)               # e.g. n=1 → 0.33, n=2 → 0.5, n=5 → 0.71, n=10 → 0.83
    w_sav = 0.7 * t               # savings weight, high early
    w_dest = 1.0 - 0.7 * t        # destination penalty weight, high late

    # Precomputed global maximum distance for scaling
    max_dist = np.max(distance_matrix)
    if max_dist == 0:
        max_dist = 1.0

    cur_to_dest = distance_matrix[current_node, destination_node]

    best_score = -np.inf
    best_node = None

    for u in unvisited_nodes:
        direct = distance_matrix[current_node, u]
        dest   = distance_matrix[u, destination_node]
        saving = cur_to_dest - direct - dest

        # Scale distances by max_dist to avoid units dominating
        score = w_sav * saving - w_dest * (dest / max_dist)

        if score > best_score:
            best_score = score
            best_node = u

    return best_node
```
