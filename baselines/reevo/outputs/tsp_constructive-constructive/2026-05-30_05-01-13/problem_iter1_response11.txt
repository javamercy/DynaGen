import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a two-step lookahead with nearest neighbor remainder."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_node = None
    best_total_cost = float('inf')
    
    for candidate in unvisited_nodes:
        # Cost of moving to candidate
        total_cost = distance_matrix[current_node][candidate]
        
        # Remaining nodes after visiting candidate
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        
        # Simulate a greedy nearest neighbor path from candidate to destination
        curr = candidate
        path_cost = 0.0
        unvisited_rem = set(remaining)  # work on a copy
        while unvisited_rem:
            # Find nearest neighbor among unvisited
            nearest = min(unvisited_rem, key=lambda x: distance_matrix[curr][x])
            path_cost += distance_matrix[curr][nearest]
            curr = nearest
            unvisited_rem.remove(nearest)
        # Add final edge to destination
        path_cost += distance_matrix[curr][destination_node]
        total_cost += path_cost
        
        if total_cost < best_total_cost:
            best_total_cost = total_cost
            best_node = candidate
    
    return best_node
