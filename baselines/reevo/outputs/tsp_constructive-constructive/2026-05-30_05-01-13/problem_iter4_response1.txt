import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Select the next node using only insertion savings and regret,
    with adaptive weights and minimal components."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n = len(unvisited_nodes)
    # adaptive weights: savings dominates early, regret dominates late
    w_sav = min(0.7, 0.3 + 0.4 * (n / 10))
    w_reg = 1.0 - w_sav

    future = unvisited_nodes | {destination_node}
    scores = {}
    for node in unvisited_nodes:
        # insertion savings (larger is better)
        savings = (
            distance_matrix[current_node][destination_node]
            - distance_matrix[current_node][node]
            - distance_matrix[node][destination_node]
        )

        # regret: difference between two smallest distances from node to any future node
        dists = [distance_matrix[node][f] for f in future if f != node]
        dists.sort()
        if len(dists) >= 2:
            regret = dists[1] - dists[0]
        elif len(dists) == 1:
            regret = dists[0]  # only one option -> high regret to postpone
        else:
            regret = 0.0

        scores[node] = w_sav * savings + w_reg * regret

    return max(scores, key=scores.get)
