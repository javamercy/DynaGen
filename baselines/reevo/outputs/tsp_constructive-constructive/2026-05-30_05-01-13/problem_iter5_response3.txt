```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a regret-based heuristic with lookahead.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = np.inf

    for j in unvisited_nodes:
        # Immediate cost
        c_immediate = distance_matrix[current_node][j]

        # Lookahead: cost after visiting j
        others = unvisited_nodes - {j}
        if others:
            lookahead = min(distance_matrix[j][k] for k in others)
        else:
            lookahead = distance_matrix[j][destination_node]

        cost_now = c_immediate + lookahead

        # Savings: distance from j to destination (higher -> want to visit earlier)
        savings = distance_matrix[j][destination_node]

        # Regret: cost of best alternative (visit another node i first, then j)
        best_alternative = np.inf
        for i in unvisited_nodes:
            if i == j:
                continue
            c_i = distance_matrix[current_node][i]
            others_i = unvisited_nodes - {i, j}
            if others_i:
                lookahead_i = min(distance_matrix[i][k] for k in others_i)
            else:
                lookahead_i = distance_matrix[i][destination_node]
            # Cost of going i -> j after visiting i
            cost_i_j = c_i + distance_matrix[i][j]
            # Combined two-step cost
            alt_cost = cost_i_j + lookahead_i
            if alt_cost < best_alternative:
                best_alternative = alt_cost

        regret = best_alternative - cost_now

        # Combined score: lower is better
        alpha = 0.3  # weight for savings
        beta = 0.5   # weight for regret
        score = cost_now - alpha * savings - beta * regret

        if score < best_score:
            best_score = score
            best_node = j

    return best_node
```
