```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a 2-step lookahead heuristic."""
    unvisited = list(unvisited_nodes)
    best_node = None
    best_total = float('inf')
    
    for candidate in unvisited:
        # Immediate cost to candidate
        cost1 = distance_matrix[current_node][candidate]
        
        # Remaining nodes after visiting candidate
        remaining = [n for n in unvisited if n != candidate]
        
        if not remaining:
            # Only candidate left, then go directly to destination
            total_cost = cost1 + distance_matrix[candidate][destination_node]
        else:
            # Look ahead one more step: find best second choice
            best_next_cost = float('inf')
            for next_node in remaining:
                # cost from candidate to next_node, then from next_node to destination
                future_cost = distance_matrix[candidate][next_node] + distance_matrix[next_node][destination_node]
                if future_cost < best_next_cost:
                    best_next_cost = future_cost
            total_cost = cost1 + best_next_cost
        
        if total_cost < best_total:
            best_total = total_cost
            best_node = candidate
    
    return best_node
```
