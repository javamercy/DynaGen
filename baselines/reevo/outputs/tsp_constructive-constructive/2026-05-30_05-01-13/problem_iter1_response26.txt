import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a two-step lookahead with regret and diversification.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = float('inf')
    # Precompute average distance to all unvisited nodes for normalization
    all_unvisited = list(unvisited_nodes)
    avg_dist = np.mean([distance_matrix[current_node][n] for n in all_unvisited])

    for candidate in unvisited_nodes:
        direct = distance_matrix[current_node][candidate]
        # Lookahead: smallest next step from candidate to another unvisited node
        remaining = [n for n in all_unvisited if n != candidate]
        if remaining:
            next_min = min(distance_matrix[candidate][n] for n in remaining)
        else:
            next_min = 0
        # Distance to destination (normalized)
        dest_dist = distance_matrix[candidate][destination_node]
        # Regret: difference between the best and second best choice from current (excluding candidate)
        # Simulate by looking at second best direct distance
        all_directs = [distance_matrix[current_node][n] for n in all_unvisited]
        sorted_directs = sorted(all_directs)
        best_direct = sorted_directs[0]
        second_best = sorted_directs[1] if len(sorted_directs) > 1 else best_direct
        regret = second_best - best_direct
        # Combine with weights, using normalization
        score = (0.3 * direct / avg_dist +
                 0.4 * next_min / (np.mean([distance_matrix[current_node][n] for n in all_unvisited]) + 1e-8) +
                 0.2 * dest_dist / (np.mean([distance_matrix[current_node][n] for n in all_unvisited]) + 1e-8) -
                 0.1 * regret)
        if score < best_score:
            best_score = score
            best_node = candidate
    return best_node
