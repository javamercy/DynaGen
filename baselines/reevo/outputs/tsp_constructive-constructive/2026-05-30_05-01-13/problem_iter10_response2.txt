import numpy as np
from typing import Set

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: Set[int],
    distance_matrix: np.ndarray
) -> int:
    """
    Improved constructive heuristic for TSP.

    Combines:
    - direct distance from current node
    - distance to destination
    - 2-step lookahead (current -> candidate -> best follower -> destination)
    - Clarke-Wright savings
    - regret (cost difference between best and second-best future connection)
    - isolation (average distance to other unvisited nodes)

    All components are normalised to [0,1]. Weights shift smoothly from
    exploration (savings, isolation, regret) to exploitation (direct distances,
    lookahead) as the number of remaining nodes decreases.
    """
    n_remaining = len(unvisited_nodes)
    if n_remaining == 0:
        return current_node
    if n_remaining == 1:
        return next(iter(unvisited_nodes))

    # Precompute raw values for all candidates
    unvisited = list(unvisited_nodes)
    raw = []
    for node in unvisited:
        d_cur = distance_matrix[current_node, node]
        d_dest = distance_matrix[node, destination_node]

        # Unvisited nodes except current candidate
        others = [j for j in unvisited if j != node]

        # 2-step lookahead: best follower (minimises node->follower + follower->dest)
        if others:
            best_follower = min(
                others,
                key=lambda j: distance_matrix[node, j] + distance_matrix[j, destination_node]
            )
            lookahead = distance_matrix[node, best_follower] + distance_matrix[best_follower, destination_node]
        else:
            lookahead = d_dest

        # Clarke-Wright savings (positive is good)
        savings = distance_matrix[current_node, destination_node] + d_dest - d_cur

        # Regret: how much worse it is to connect candidate to its second-best
        # unvisited node vs its best unvisited node (excluding itself)
        if others:
            dists = [distance_matrix[node, j] for j in others]
            sorted_dists = sorted(dists)
            best = sorted_dists[0]
            second_best = sorted_dists[1] if len(sorted_dists) > 1 else best
            regret = second_best - best
        else:
            regret = 0.0

        # Isolation: average distance to other unvisited nodes (including destination)
        if others:
            isolation = np.mean([distance_matrix[node, j] for j in others] + [d_dest])
        else:
            isolation = d_dest

        raw.append((d_cur, d_dest, lookahead, savings, regret, isolation))

    # Extract arrays for normalisation
    d_cur_vals = np.array([r[0] for r in raw])
    d_dest_vals = np.array([r[1] for r in raw])
    look_vals = np.array([r[2] for r in raw])
    save_vals = np.array([r[3] for r in raw])
    reg_vals = np.array([r[4] for r in raw])
    isol_vals = np.array([r[5] for r in raw])

    # Normalisation divisors (avoid zero)
    eps = 1e-12
    max_dcur = max(d_cur_vals.max(), eps)
    max_ddest = max(d_dest_vals.max(), eps)
    max_look = max(look_vals.max(), eps)
    max_save = max(save_vals.max(), eps)
    max_reg = max(reg_vals.max(), eps)
    max_isol = max(isol_vals.max(), eps)

    # Adaptive weights: sigmoid-like transition based on remaining fraction
    fraction_left = n_remaining / (n_remaining + 5)  # 0..1, concave
    # Exploration terms (high when many nodes left)
    w_sav = 0.30 * fraction_left
    w_isol = 0.20 * fraction_left
    w_reg = 0.15 * fraction_left
    # Exploitation terms (high when few nodes left)
    w_cur = 0.30 + 0.15 * (1 - fraction_left)
    w_dest = 0.15 + 0.25 * (1 - fraction_left)
    w_look = 0.50 * (1 - fraction_left) + 0.10  # always some weight

    # Compute normalised scores (lower is better)
    best_score = float('inf')
    best_node = None
    for idx, node in enumerate(unvisited):
        d_cur_norm = d_cur_vals[idx] / max_dcur
        d_dest_norm = d_dest_vals[idx] / max_ddest
        look_norm = look_vals[idx] / max_look
        save_norm = save_vals[idx] / max_save
        reg_norm = reg_vals[idx] / max_reg
        isol_norm = isol_vals[idx] / max_isol

        # Composite score: minimise
        score = (
            w_cur * d_cur_norm +
            w_dest * d_dest_norm +
            w_look * look_norm +
            w_sav * (1.0 - save_norm) +      # high savings → low contribution
            w_reg * reg_norm +               # high regret → increase score (bad)
            w_isol * isol_norm               # high isolation → pick it now? Actually we want to pick isolated nodes now.
        )
        # Note: for isolation, high isolation means node is far from others → desirable to insert now.
        # So we treat it like a penalty if not chosen: lower score (i.e., better) for high isolation?
        # In the formula, if isolation is high, isol_norm is high, increasing score → that encourages picking it?
        # Actually we want to minimise score, so high isolation would make score higher (worse).
        # That is wrong: we want to prefer isolated nodes. So we should subtract a term proportional to isolation.
        # Let's correct: we want to add a bonus for high isolation. So use -w_isol * isol_norm.
        # Let's re-evaluate: In v0 they added isolation, v1 they subtracted (since they minimized).
        # In v1's formula: score = (w_cur * nd_cur + ... + w_conn * (-nd_avg)). They subtracted isolation.
        # So we should subtract a term for isolation.
        score_corrected = (
            w_cur * d_cur_norm +
            w_dest * d_dest_norm +
            w_look * look_norm +
            w_sav * (1.0 - save_norm) +
            w_reg * reg_norm -
            w_isol * isol_norm     # high isolation → reduce score → better
        )

        if score_corrected < best_score:
            best_score = score_corrected
            best_node = node

    return best_node if best_node is not None else unvisited[0]
