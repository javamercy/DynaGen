import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Look-ahead heuristic that scores each candidate node based on:
    - direct cost from current node
    - cost to destination node
    - regret: difference between best and average next step from candidate
    - isolation: distance from candidate to its nearest unvisited neighbor
    """
    if len(unvisited_nodes) == 1:
        return list(unvisited_nodes)[0]

    # Precompute distances for all candidates
    candidates = np.array(list(unvisited_nodes))
    n_candidates = len(candidates)
    dist_current = distance_matrix[current_node, candidates]
    dist_dest = distance_matrix[destination_node, candidates]

    # For each candidate, compute distances to other unvisited nodes (including destination)
    # We'll treat destination as always "available" after current node is visited.
    # Build a matrix of distances from candidate to all other relevant nodes.
    # Include destination because it's the final target.
    all_nodes = np.concatenate([candidates, [destination_node]])

    # Compute distances from each candidate to all other candidates + destination
    # Shape: (n_candidates, n_candidates+1)
    dist_from_cand = distance_matrix[np.ix_(candidates, all_nodes)]

    # For each candidate, remove its own index (distance to itself)
    # Create mask to exclude self-distance
    self_mask = np.zeros((n_candidates, n_candidates + 1), dtype=bool)
    np.fill_diagonal(self_mask[:, :n_candidates], True)  # exclude candidate to itself
    # (destination doesn't have self issue because it's not a candidate)

    # Compute for each candidate: min and average distance to other unvisited nodes (including destination)
    # Use masked array to ignore self
    masked_dist = np.ma.array(dist_from_cand, mask=self_mask)
    min_dist = np.min(masked_dist, axis=1).data
    avg_dist = np.mean(masked_dist, axis=1).data

    # Regret: best - average (positive if average is larger than best)
    regret = min_dist - avg_dist

    # Isolation: distance from candidate to its nearest neighbor (excluding itself, include destination)
    # Already have min_dist above.

    # Normalize factors (robust scaling using median and IQR)
    def robust_norm(x):
        med = np.median(x)
        iqr = np.percentile(x, 75) - np.percentile(x, 25)
        if iqr == 0:
            return np.zeros_like(x)
        return (x - med) / iqr

    ndist_cur = robust_norm(dist_current)
    ndist_dest = robust_norm(dist_dest)
    nregret = robust_norm(regret)
    nisolation = robust_norm(min_dist)

    # Weights that adapt based on number of remaining nodes
    # When many nodes left, prioritize low regret and low isolation; later prioritize direct distance
    remaining = n_candidates
    w_cur = 0.3 + 0.3 * (remaining / len(unvisited_nodes))  # increase weight of current distance later
    w_dest = 0.2 * (1 - remaining / len(unvisited_nodes))  # decrease destination weight early
    w_regret = 0.3
    w_iso = 0.2

    # Score: minimize a weighted sum (we want to minimize cost, so lower score is better)
    score = (w_cur * ndist_cur +
             w_dest * ndist_dest +
             w_regret * nregret +
             w_iso * nisolation)

    # Pick the node with minimum score
    next_node = candidates[np.argmin(score)]
    return int(next_node)
