import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a two-step look-ahead heuristic."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # Cost from current to candidate
        direct_cost = distance_matrix[current_node][candidate]
        remaining = unvisited_nodes - {candidate}

        # Look ahead: find the best two-step continuation from candidate
        # (candidate -> next -> destination)
        best_future = float('inf')
        for next_node in remaining:
            future_cost = distance_matrix[candidate][next_node] + distance_matrix[next_node][destination_node]
            if future_cost < best_future:
                best_future = future_cost

        total_cost = direct_cost + best_future

        if total_cost < best_total:
            best_total = total_cost
            best_node = candidate

    return best_node
