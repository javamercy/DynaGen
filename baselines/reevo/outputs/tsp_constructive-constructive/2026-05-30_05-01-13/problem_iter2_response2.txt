```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved regret-based heuristic with dynamic alpha and two-step look-ahead.
    Score = (direct cost + best_two_step_future) - alpha * regret
    alpha = 0.5 * (len(unvisited_nodes) / total_nodes) to gradually shift from flexibility to greediness.
    """
    total_nodes = distance_matrix.shape[0]
    n_unvisited = len(unvisited_nodes)
    if n_unvisited == 1:
        return next(iter(unvisited_nodes))

    # dynamic alpha: more emphasis on regret when many nodes remain
    alpha = 0.5 * (n_unvisited / total_nodes)

    best_score = float('inf')
    best_node = None

    for candidate in unvisited_nodes:
        # immediate cost from current to candidate
        direct_cost = distance_matrix[current_node][candidate]

        remaining = unvisited_nodes - {candidate}
        n_remaining = len(remaining)

        # two-step look-ahead: best cost from candidate to destination via any remaining node
        best_future = float('inf')
        for next_node in remaining:
            cost = distance_matrix[candidate][next_node] + distance_matrix[next_node][destination_node]
            if cost < best_future:
                best_future = cost

        # regret: difference between second-best and best edge from candidate to remaining nodes
        if n_remaining >= 2:
            dists = sorted(distance_matrix[candidate][n] for n in remaining)
            regret = dists[1] - dists[0]
        else:
            regret = 0.0

        # total score: minimize immediate+future cost and maximize regret (subtract)
        score = (direct_cost + best_future) - alpha * regret

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
```
