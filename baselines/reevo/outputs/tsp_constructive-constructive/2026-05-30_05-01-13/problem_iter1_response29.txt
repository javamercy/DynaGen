import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a look-ahead heuristic.
    The heuristic combines immediate insertion cost (relative to the destination)
    with an estimate of future travel cost based on the node's connectivity to
    other unvisited nodes (average distance to all other unvisited nodes).
    """
    # Weight for the future cost component (average distance to other unvisited nodes)
    beta = 0.3

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        # Immediate insertion cost if we insert 'node' between current_node and destination_node
        insertion_cost = (distance_matrix[current_node][node] +
                          distance_matrix[node][destination_node] -
                          distance_matrix[current_node][destination_node])

        # Average distance from 'node' to all other unvisited nodes (excluding 'node' itself)
        other_unvisited = unvisited_nodes - {node}
        if other_unvisited:
            avg_dist = np.mean([distance_matrix[node][k] for k in other_unvisited])
        else:
            avg_dist = 0.0

        # Score: lower insertion cost and lower average distance to others are better
        score = insertion_cost + beta * avg_dist

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
