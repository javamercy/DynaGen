```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead heuristic that combines
    Clarke-Wright savings with the minimum distance from the candidate
    to the remaining unvisited nodes.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Weights for the scoring function
    w_savings = 1.0          # importance of Clarke-Wright savings
    w_lookahead = 0.5        # importance of the next-step minimum distance (penalty)

    d_cur_dest = distance_matrix[current_node, destination_node]
    best_node = None
    best_score = -np.inf

    for node in unvisited_nodes:
        d_cur_node = distance_matrix[current_node, node]
        d_node_dest = distance_matrix[node, destination_node]

        # Clarke-Wright saving: how much we "save" by going via node instead of directly to destination
        saving = d_cur_dest - d_cur_node - d_node_dest

        # Look-ahead: minimum distance from this node to any other unvisited node
        others = [n for n in unvisited_nodes if n != node]
        min_next = min(distance_matrix[node, n] for n in others) if others else 0.0

        # Score: high saving and low future distance are desirable
        score = w_savings * saving - w_lookahead * min_next

        if score > best_score:
            best_score = score
            best_node = node

    return best_node
```
