```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node to visit using a simple adaptive scoring that balances
    immediate cost, look-ahead (average of the two shortest onward edges), and
    proximity to the final destination. Weights shift smoothly from look-ahead-
    heavy to destination-heavy as the number of unvisited nodes decreases.
    """
    if not unvisited_nodes:
        raise ValueError("No unvisited nodes")

    n_remaining = len(unvisited_nodes)
    best_score = float('inf')
    best_node = None

    # Smooth weighting factor: when many nodes remain, look-ahead dominates;
    # when few remain, destination proximity dominates.
    w_look = 0.8 * n_remaining / (n_remaining + 5)
    w_dest = 0.2 + 0.8 * 5 / (n_remaining + 5)   # complements w_look

    for node in unvisited_nodes:
        # 1. Immediate travel cost from current node
        immediate = distance_matrix[current_node, node]

        # 2. Look-ahead: average of the two shortest distances from candidate
        #    to the other remaining unvisited nodes.
        remaining = list(unvisited_nodes - {node})
        m = len(remaining)

        if m >= 2:
            dists = distance_matrix[node, remaining]
            sorted_dists = np.sort(dists)
            look_ahead = (sorted_dists[0] + sorted_dists[1]) / 2.0
        elif m == 1:
            look_ahead = distance_matrix[node, remaining[0]]
        else:
            look_ahead = 0.0

        # 3. Distance from candidate to final destination
        dest_cost = distance_matrix[destination_node, node]

        # Composite cost (lower is better)
        total = immediate + w_look * look_ahead + w_dest * dest_cost

        if total < best_score:
            best_score = total
            best_node = node

    return best_node
```
