```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node using a fully vectorized heuristic that combines
    Clarke‑Wright savings, positive regret (cost of postponing), connectivity
    diversity, and destination proximity.
    """
    if not unvisited_nodes:
        raise ValueError("No unvisited nodes to select from.")

    candidates = np.array(list(unvisited_nodes))
    n = len(candidates)
    if n == 1:
        return candidates[0]

    # ----- Precomputed vectors -----
    direct = distance_matrix[current_node, candidates]            # dist(current, u)
    dest   = distance_matrix[candidates, destination_node]        # dist(u, destination)
    savings = distance_matrix[current_node, destination_node] - (direct + dest)

    # ----- Distance submatrix among candidates -----
    D = distance_matrix[np.ix_(candidates, candidates)]           # D[i,j] = dist(candidate_i, candidate_j)
    # For connectivity: mean and std of distances to other candidates (excluding self)
    sum_d = D.sum(axis=1) - np.diag(D)                            # sum over all j != i
    mean_conn = sum_d / (n - 1)
    sum_sq_d = (D ** 2).sum(axis=1) - np.diag(D) ** 2
    var_conn = sum_sq_d / (n - 1) - mean_conn ** 2
    var_conn = np.maximum(var_conn, 0.0)                          # numerical safety
    std_conn = np.sqrt(var_conn)

    # ----- Regret (cost of postponing) -----
    # For each candidate u (row), compute min over v != u of:
    #   dist(current, v) + dist(v, u) - direct[u]
    # D.T[i,j] = dist(candidates[j], candidates[i]) = dist(v, u) for v=j, u=i
    cost_via_v = direct[np.newaxis, :] + D.T - direct[:, np.newaxis]   # shape (n, n)
    # Set diagonal to infinity so that v==u is ignored
    np.fill_diagonal(cost_via_v, np.inf)
    min_cost = np.min(cost_via_v, axis=1)                         # min over v
    regret = np.maximum(min_cost, 0.0)                            # positive regret

    # ----- Composite score -----
    # Weights (same as in the better version)
    w_dir    = -0.30
    w_dest   = -0.20
    w_sav    =  0.35
    w_regret =  0.30
    w_conn_avg = -0.15
    w_conn_std =  0.10

    scores = (w_dir * direct +
              w_dest * dest +
              w_sav * savings +
              w_regret * regret +
              w_conn_avg * mean_conn +
              w_conn_std * std_conn)

    best_idx = np.argmax(scores)
    return candidates[best_idx]
```
