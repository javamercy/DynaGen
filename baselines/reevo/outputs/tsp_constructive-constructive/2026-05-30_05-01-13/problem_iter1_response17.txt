```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead regret heuristic.
    For each candidate u, compute:
        - distance from current node to u (d_cur_u)
        - best distance from u to any other unvisited node (or to destination if last)
        - second best distance from u to any other unvisited node (or same if last)
        - regret = second_best - best
    Score = d_cur_u - alpha * regret   (alpha balances between current cost and urgency)
    Choose candidate with minimal score.
    If only one node remains, regret is 0 and score is simply d_cur_u.
    """
    n_remaining = len(unvisited_nodes)
    alpha = 0.3  # weight on regret (higher means more urgent)

    best_score = float('inf')
    best_node = None

    for u in unvisited_nodes:
        # distance from current to u
        d_cur_u = distance_matrix[current_node][u]

        # distances from u to other unvisited nodes (excluding u itself)
        other_nodes = [v for v in unvisited_nodes if v != u]
        if not other_nodes:
            # u is the last unvisited node; best and second best are both the return
            best = distance_matrix[u][destination_node]
            second_best = best
        else:
            dist_list = [distance_matrix[u][v] for v in other_nodes]
            # add also the return to destination as a pseudo-node
            dist_list.append(distance_matrix[u][destination_node])
            dist_list.sort()
            best = dist_list[0]
            second_best = dist_list[1] if len(dist_list) > 1 else best

        regret = second_best - best
        # score: low distance is good, high regret (urgency) is good -> subtract regret
        score = d_cur_u - alpha * regret

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
```
