```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved heuristic combining Clarke-Wright savings, regret, proximity, and future cost
    with adaptive weights that vary continuously with the number of unvisited nodes.
    No normalization of components is applied.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n = len(unvisited_nodes)
    base_saving = distance_matrix[current_node][destination_node]

    best_score = -np.inf
    best_node = None

    # Adaptive weights: as nodes remain, savings dominate; when few, regret and future cost matter more.
    # Smooth transition based on n
    if n > 15:
        w_cur, w_sav, w_reg, w_fut = 0.10, 0.60, 0.15, 0.15
    elif n > 10:
        w_cur, w_sav, w_reg, w_fut = 0.15, 0.50, 0.20, 0.15
    elif n > 5:
        w_cur, w_sav, w_reg, w_fut = 0.20, 0.35, 0.25, 0.20
    else:
        w_cur, w_sav, w_reg, w_fut = 0.25, 0.25, 0.30, 0.20

    for u in unvisited_nodes:
        d_cur_u = distance_matrix[current_node][u]
        d_u_dest = distance_matrix[u][destination_node]
        saving = base_saving - d_cur_u - d_u_dest

        # Collect distances to all other unvisited nodes and to destination
        others = [distance_matrix[u][v] for v in unvisited_nodes if v != u]
        others.append(distance_matrix[u][destination_node])
        others.sort()

        min1 = others[0]
        min2 = others[1] if len(others) > 1 else min1  # if only one other, min2 = min1 -> regret 0
        regret = min2 - min1

        future = min1  # minimum distance from u to any future node (including destination)

        # Composite score to maximize
        score = -w_cur * d_cur_u + w_sav * saving + w_reg * regret - w_fut * future

        if score > best_score:
            best_score = score
            best_node = u

    return best_node
```
