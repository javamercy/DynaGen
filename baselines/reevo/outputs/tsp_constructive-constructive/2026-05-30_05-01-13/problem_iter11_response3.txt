```python
import numpy as np
import math

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using adaptive weights, z-score normalization, and multi-step lookahead."""
    unvisited = list(unvisited_nodes)
    n_remaining = len(unvisited)
    if n_remaining == 1:
        return unvisited[0]

    total_nodes = distance_matrix.shape[0]
    alpha = n_remaining / total_nodes  # exploration factor, high when many nodes left

    # Prepare per-candidate feature lists
    f_immediate = []
    f_neg_savings = []   # -savings (minimize)
    f_regret = []        # regret value (positive, will be negated later)
    f_connect = []       # average distance to other unvisited
    f_lookahead = []     # best two-step cost to destination

    # Precompute distances for regret sampling (optional)
    # For large n, sample a subset of 'other' nodes for regret calculation
    if n_remaining > 100:
        sample_size = 30
        # we will sample u's per candidate later
    else:
        sample_size = n_remaining

    for v in unvisited:
        # 1. Immediate cost
        d_curr_v = distance_matrix[current_node][v]
        f_immediate.append(d_curr_v)

        # 2. Savings (negative for minimization)
        d_v_dest = distance_matrix[v][destination_node]
        d_curr_dest = distance_matrix[current_node][destination_node]
        savings = d_curr_dest - (d_curr_v + d_v_dest)
        f_neg_savings.append(-savings)  # high savings => low negative => good

        # 3. Regret: urgency to insert now vs. later
        # Insertion cost now
        cost_now = d_curr_v + d_v_dest - d_curr_dest
        # Compute best insertion cost later (by inserting another node first)
        best_cost_later = float('inf')
        # Select subset of other nodes to evaluate
        others = [u for u in unvisited if u != v]
        if len(others) > sample_size:
            # random sample
            idx = np.random.choice(len(others), sample_size, replace=False)
            sample_others = [others[i] for i in idx]
        else:
            sample_others = others
        for u in sample_others:
            # After inserting u, tour becomes C-u-D
            cost1 = distance_matrix[current_node][v] + distance_matrix[v][u] - distance_matrix[current_node][u]
            cost2 = distance_matrix[u][v] + distance_matrix[v][destination_node] - distance_matrix[u][destination_node]
            cost_later = min(cost1, cost2)
            if cost_later < best_cost_later:
                best_cost_later = cost_later
        regret = max(0.0, best_cost_later - cost_now)  # if delaying is worse, regret positive
        f_regret.append(regret)

        # 4. Connectivity: average distance to other unvisited nodes (lower is better)
        if len(others) > 0:
            avg_dist = np.mean([distance_matrix[v][u] for u in others])
        else:
            avg_dist = 0.0
        f_connect.append(avg_dist)

        # 5. Lookahead: best two-step cost from v to destination via a next node
        best_two_step = float('inf')
        for w in others:
            cost = distance_matrix[v][w] + distance_matrix[w][destination_node]
            if cost < best_two_step:
                best_two_step = cost
        if best_two_step == float('inf'):  # no other node
            best_two_step = d_v_dest
        f_lookahead.append(best_two_step)

    # Z-score normalization for each feature
    def zscore(vals):
        arr = np.array(vals, dtype=float)
        mean = np.mean(arr)
        std = np.std(arr)
        if std < 1e-8:
            return np.zeros_like(arr)
        return (arr - mean) / std

    z_imm = zscore(f_immediate)
    z_neg_sav = zscore(f_neg_savings)
    z_reg = zscore(f_regret)
    z_conn = zscore(f_connect)
    z_look = zscore(f_lookahead)

    # Adaptive weights shift from exploration to exploitation
    w_imm = 0.2 + 0.3 * (1 - alpha)          # exploitation increases
    w_sav = 0.3 - 0.2 * alpha                # exploration decreases
    w_reg = 0.3 - 0.2 * alpha                # exploration
    w_conn = 0.2 - 0.1 * alpha               # exploration
    w_look = 0.2 + 0.1 * (1 - alpha)         # exploitation (lookahead helps)

    # Combined score (minimize)
    scores = (w_imm * z_imm +
              w_sav * z_neg_sav +
              w_reg * (-z_reg) +   # negative regret because we want to maximize regret (so high regret -> low score)
              w_conn * z_conn +
              w_look * z_look)

    best_idx = np.argmin(scores)
    return unvisited[best_idx]
```
