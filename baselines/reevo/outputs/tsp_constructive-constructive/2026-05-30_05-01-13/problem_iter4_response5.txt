```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a heuristic that combines:
    - Clarke-Wright savings: dist(current, dest) - dist(current, u) - dist(u, dest)
    - Regret: difference between the best and second-best distance from u to other unvisited nodes
    - Proximity: negative of the direct distance from the current node (prefer closer nodes)
    
    The score is a weighted sum (saving weight 0.5, regret weight 0.3, proximity weight 0.2).
    Higher scores indicate better candidates.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    base_saving = distance_matrix[current_node][destination_node]
    best_score = -np.inf
    best_node = None

    for u in unvisited_nodes:
        d_cur_u = distance_matrix[current_node][u]
        d_u_dest = distance_matrix[u][destination_node]
        saving = base_saving - d_cur_u - d_u_dest

        # Regret: difference between second-best and best distance to other unvisited nodes
        others = unvisited_nodes - {u}
        if len(others) >= 2:
            dists = [distance_matrix[u][v] for v in others]
            sorted_dists = sorted(dists)
            regret = sorted_dists[1] - sorted_dists[0]  # >= 0
        elif len(others) == 1:
            regret = 0.0
        else:  # no others (should not happen due to early return)
            regret = 0.0

        proximity = -d_cur_u  # negative, so smaller distances give higher scores

        # Fixed weights: saving dominates, regret helps avoid poor future choices,
        # proximity provides local improvement.
        score = 0.5 * saving + 0.3 * regret + 0.2 * proximity

        if score > best_score:
            best_score = score
            best_node = u

    return best_node
```
