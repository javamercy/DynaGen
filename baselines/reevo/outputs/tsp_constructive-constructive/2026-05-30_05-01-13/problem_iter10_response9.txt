import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Improved constructive heuristic for TSP.

    Uses:
    - one-step lookahead
    - Clarke-Wright savings
    - regret (cost of delaying a node)
    - adaptive weight based on number of remaining candidates
    - z-score normalization for robust combination

    All computations are vectorized for efficiency.
    """
    # Ensure current node is not considered as candidate
    candidates = list(unvisited_nodes - {current_node})
    
    if len(candidates) == 0:
        return current_node
    if len(candidates) == 1:
        return candidates[0]

    n = len(candidates)
    # Sort for deterministic tie-breaking
    candidates.sort()
    idx = np.arange(n)

    # Distances from current node to each candidate
    d_cur = distance_matrix[current_node, candidates]          # (n,)
    # Distances from each candidate to the destination node
    d_dest = distance_matrix[candidates, destination_node]     # (n,)
    # Direct distance current -> destination
    d_cur_dest = distance_matrix[current_node, destination_node]

    # ---- 1. One-step lookahead ----
    # For each candidate, the minimum distance to any *other* candidate
    dists_between = distance_matrix[np.ix_(candidates, candidates)]  # (n, n)
    np.fill_diagonal(dists_between, np.inf)
    lookahead = np.min(dists_between, axis=1)                         # (n,)

    # Path quality = cost of going to candidate + best onward step
    path_quality = d_cur + lookahead

    # ---- 2. Clarke-Wright savings ----
    # savings = d_cur + d_dest - d_cur_dest (higher is better)
    savings = d_cur + d_dest - d_cur_dest

    # ---- 3. Regret ----
    # Insertion cost = d_cur + d_dest
    insert_cost = d_cur + d_dest
    # Find the smallest and second smallest insertion cost among all candidates
    sorted_idx = np.argsort(insert_cost)
    smallest_cost = insert_cost[sorted_idx[0]]
    second_smallest_cost = insert_cost[sorted_idx[1]]
    # For each candidate, regret = best alternative insertion cost - own cost
    regret = np.where(
        insert_cost == smallest_cost,
        second_smallest_cost - insert_cost,
        smallest_cost - insert_cost,
    )  # positive values mean the candidate is clearly best

    # ---- 4. Z-score normalization ----
    def z_score(arr: np.ndarray) -> np.ndarray:
        mu = np.mean(arr)
        sigma = np.std(arr)
        if sigma == 0:
            return np.zeros_like(arr)
        return (arr - mu) / sigma

    z_path = z_score(path_quality)      # lower is better (original value)
    z_savings = z_score(savings)        # higher is better
    z_regret = z_score(regret)          # higher is better

    # ---- 5. Adaptive weight ----
    w = n / (n + 10.0)  # between ~0.17 (n=2) and ~0.83 (n=50)

    # ---- 6. Score (minimize) ----
    score = (1.0 - w) * z_path - w * (z_savings + z_regret)

    best = int(np.argmin(score))
    return candidates[best]
