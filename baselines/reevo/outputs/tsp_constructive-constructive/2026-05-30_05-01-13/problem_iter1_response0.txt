import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a two-step look-ahead heuristic.
    For each candidate v, we compute the cost of the path:
        current -> v -> best_next -> destination
    where best_next is the node in U\{v} that minimizes (dist(v, w) + dist(w, destination)).
    We then add a penalty proportional to the number of remaining nodes after v and w.
    The candidate with the minimum total score is chosen.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Precompute average distance for the penalty term
    avg_dist = np.mean(distance_matrix)
    
    best_score = float('inf')
    best_node = None
    
    for v in unvisited_nodes:
        # Immediate cost to go from current to v
        immediate = distance_matrix[current_node][v]
        
        # Look-ahead: find best next step w from v (w != v, w in unvisited)
        min_lookahead = float('inf')
        for w in unvisited_nodes:
            if w == v:
                continue
            # cost from v to w, then w to destination
            cost_look = distance_matrix[v][w] + distance_matrix[w][destination_node]
            if cost_look < min_lookahead:
                min_lookahead = cost_look
        
        # Penalty for nodes not covered in the two-step look-ahead
        remaining_count = len(unvisited_nodes) - 2  # v and best_next are visited
        penalty = remaining_count * avg_dist * 0.5  # weight factor
        
        total_score = immediate + min_lookahead + penalty
        
        # Also consider a direct path if look-ahead failed (should not happen)
        if min_lookahead == float('inf'):
            total_score = immediate + distance_matrix[v][destination_node]
        
        if total_score < best_score:
            best_score = total_score
            best_node = v
    
    return best_node
