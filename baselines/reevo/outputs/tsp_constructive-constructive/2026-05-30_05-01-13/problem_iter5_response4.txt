import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using multi-step lookahead and savings.
    For each candidate, compute a two-step cost (current->node + min next from node)
    and adjust by a savings term (benefit of going via node versus direct to destination).
    The savings weight is adaptive: higher when few nodes remain.
    """
    best_node = None
    best_score = float('inf')
    n_remaining = len(unvisited_nodes)
    # Adaptive savings weight: small when many nodes left, large when few left.
    lambda_savings = 1.0 / (1.0 + n_remaining)

    for node in unvisited_nodes:
        # Cost of immediate step
        cost1 = distance_matrix[current_node][node]
        # Two-step lookahead: best next step after node
        remaining = unvisited_nodes - {node}
        if remaining:
            cost2 = min(distance_matrix[node][j] for j in remaining)
        else:
            cost2 = distance_matrix[node][destination_node]
        two_step_cost = cost1 + cost2

        # Savings: benefit of going via node vs direct to destination
        savings = distance_matrix[current_node][destination_node] - (cost1 + distance_matrix[node][destination_node])
        # Score: lower is better
        score = two_step_cost - lambda_savings * savings

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
