```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 1‑step lookahead heuristic.
    For each candidate node, the remaining path to the destination node is
    approximated via nearest neighbor (greedy) over the still unvisited nodes.
    The candidate that minimises the sum of the immediate edge cost and the
    estimated future cost is chosen.
    """
    best_candidate = None
    best_total = float('inf')

    # Precompute distance from each node to destination, used in greedy
    # but we need it inside the loop anyway.
    for candidate in unvisited_nodes:
        # Cost to move current -> candidate
        first_cost = distance_matrix[current_node][candidate]

        # Simulate the rest of the tour from candidate to destination
        # using nearest neighbour.
        remaining = set(unvisited_nodes) - {candidate}
        cur = candidate
        sim_distance = 0.0
        while remaining:
            # Closest node in remaining to current
            next_node = min(remaining, key=lambda n: distance_matrix[cur][n])
            sim_distance += distance_matrix[cur][next_node]
            remaining.remove(next_node)
            cur = next_node
        # Finally go to destination
        sim_distance += distance_matrix[cur][destination_node]

        total_estimate = first_cost + sim_distance

        if total_estimate < best_total:
            best_total = total_estimate
            best_candidate = candidate

    return best_candidate
```
