import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node with a two-step look-ahead heuristic.
    
    For each candidate node, compute a score that combines:
      1. Direct distance from current to candidate.
      2. Estimated cost of the best continuation from candidate (nearest neighbor to the remaining unvisited nodes,
         including the destination as a final anchor).
    The candidate with the smallest total is chosen.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Precompute distances from destination to all nodes for fast lookup
    dest_dist = distance_matrix[destination_node]
    
    best_score = float('inf')
    best_node = None
    
    for candidate in unvisited_nodes:
        # Direct cost from current to candidate
        direct_cost = distance_matrix[current_node][candidate]
        
        # Look-ahead: from the candidate, consider the remaining nodes (excluding candidate)
        remaining = unvisited_nodes - {candidate}
        if remaining:
            # Best next step from candidate: find the node in remaining with minimal distance
            # to candidate plus that node's distance to destination (a sort of 2-step regret)
            # We'll use a simple nearest neighbor to destination as a proxy for good continuation
            look_cost = float('inf')
            for next_node in remaining:
                cost = distance_matrix[candidate][next_node] + dest_dist[next_node]
                if cost < look_cost:
                    look_cost = cost
        else:
            look_cost = dest_dist[candidate]  # If only candidate left, go directly to destination
        
        # Combined score: we can weight the look_ahead less (e.g., 0.6 direct, 0.4 look_ahead)
        score = 0.6 * direct_cost + 0.4 * look_cost
        
        if score < best_score:
            best_score = score
            best_node = candidate
    
    return best_node
