import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a regret-based lookahead mechanism.
    For each candidate u, compute the insertion cost if added now (between current and destination)
    and the best possible insertion cost if added later (after some other node w).
    The node with the largest regret (difference best_later - now) is chosen,
    as postponing it would be more costly.
    """
    best_node = None
    best_regret = -float('inf')
    
    # Precompute baseline distance to avoid repeated indexing
    baseline = distance_matrix[current_node][destination_node]
    
    for u in unvisited_nodes:
        # Cost of inserting u between current and destination now
        now_cost = distance_matrix[current_node][u] + distance_matrix[u][destination_node] - baseline
        
        # Best cost of inserting u after some other node w
        best_later = float('inf')
        for w in unvisited_nodes:
            if w == u:
                continue
            later_cost = distance_matrix[w][u] + distance_matrix[u][destination_node] - distance_matrix[w][destination_node]
            if later_cost < best_later:
                best_later = later_cost
        
        # If no alternative exists (only u left), later cost equals now cost
        if best_later == float('inf'):
            best_later = now_cost
        
        regret = best_later - now_cost
        if regret > best_regret:
            best_regret = regret
            best_node = u
    
    return best_node
