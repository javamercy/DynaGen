import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a composite heuristic:
    - path quality (distance + one-step lookahead)
    - savings (Clarke-Wright style)
    - regret (cost of delaying a node)
    - adaptive weighting to shift from exploration to exploitation
    """
    # Ensure current node is not in unvisited set
    unvisited = unvisited_nodes - {current_node}
    if len(unvisited) == 0:
        return current_node
    if len(unvisited) == 1:
        return next(iter(unvisited))

    candidates = list(unvisited)
    n = len(candidates)

    # Precompute distances from current and to destination
    dist_curr = np.array([distance_matrix[current_node][c] for c in candidates])
    dist_dest = np.array([distance_matrix[c][destination_node] for c in candidates])
    # Distance from current to destination (used in savings)
    curr_dest = distance_matrix[current_node][destination_node]

    # Lookahead: for each candidate, the best distance to another unvisited node
    lookahead = np.zeros(n)
    for i, c in enumerate(candidates):
        others = [distance_matrix[c][o] for o in candidates if o != c]
        if others:
            lookahead[i] = min(others)

    # Path quality = distance from current + best next step
    path_quality = dist_curr + lookahead

    # Savings = dist_curr + dist_dest - curr_dest (higher is better)
    savings = dist_curr + dist_dest - curr_dest

    # Insertion cost (for regret) = dist_curr + dist_dest
    insert_cost = dist_curr + dist_dest

    # Regret: difference between second best and best insertion cost for each candidate
    sorted_indices = np.argsort(insert_cost)
    # Minimum and second minimum insertion cost
    min_cost = insert_cost[sorted_indices[0]]
    # Second min (may be same if duplicates, but we take the next smallest)
    second_min_cost = insert_cost[sorted_indices[1]]

    regret = np.empty(n)
    for i in range(n):
        # Best alternative insertion cost if we skip this candidate
        if i == sorted_indices[0]:
            best_other = second_min_cost
        else:
            best_other = min_cost
        regret[i] = best_other - insert_cost[i]   # positive if candidate is clearly best

    # Normalize all three terms to z-scores for robust combination
    def z_score(arr):
        mu = np.mean(arr)
        sigma = np.std(arr)
        if sigma == 0:
            return np.zeros_like(arr)
        return (arr - mu) / sigma

    z_path = z_score(path_quality)
    z_savings = z_score(savings)        # higher savings -> higher z_savings (good)
    z_regret = z_score(regret)          # higher regret -> higher z_regret (good)

    # Adaptive weight: high when many nodes remain (exploration), low when few remain (exploitation)
    w = n / (n + 10)   # ranges from ~0.5 (n=10) to 1 (n large), but here n >=2 so w > 0.16
    # For n=2, w=2/12≈0.17; for n=50, w=50/60≈0.83
    # We want low weight on path_quality when exploring, high weight on regret & savings.

    # Score: lower is better
    #   - path quality: lower z_path is better (already has sign matching)
    #   - savings and regret: we subtract them because higher values are better
    score = (1 - w) * z_path - w * (z_savings + z_regret)

    best_idx = np.argmin(score)
    return candidates[best_idx]
