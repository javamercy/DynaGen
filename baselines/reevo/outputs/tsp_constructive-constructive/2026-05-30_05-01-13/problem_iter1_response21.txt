import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead mechanism that simulates a greedy
    nearest-neighbor tour for the remainder of the route.
    """
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]

    best_node = None
    best_total = float('inf')

    for candidate in unvisited:
        # Simulate a greedy NN tour starting with current -> candidate
        # Then repeatedly go to the nearest unvisited node, finally return to destination
        remaining = [v for v in unvisited if v != candidate]
        current = candidate
        tour_length = distance_matrix[current_node][candidate]

        while remaining:
            # Find nearest unvisited node
            nearest = min(remaining, key=lambda v: distance_matrix[current][v])
            tour_length += distance_matrix[current][nearest]
            current = nearest
            remaining.remove(nearest)

        # Return to destination
        tour_length += distance_matrix[current][destination_node]

        if tour_length < best_total:
            best_total = tour_length
            best_node = candidate

    return best_node
