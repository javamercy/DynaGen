import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using proximity, savings, and two-step lookahead with adaptive weights."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = list(unvisited_nodes)
    n = len(unvisited)

    # Basic distances
    d_curr = distance_matrix[current_node][unvisited]                                # (n,)
    d_dest = distance_matrix[destination_node][unvisited]                            # (n,)
    d_curr_dest = distance_matrix[current_node, destination_node]

    # ---- Component 1: Proximity (smaller is better) ----
    prox = d_curr.copy()

    # ---- Component 2: Savings (larger is better) ----
    # Savings = direct distance - detour via node
    saving = d_curr_dest - (d_curr + d_dest)  # could be negative
    # We'll use negative saving so that larger saving -> smaller score component
    savings_comp = -saving  # lower is better

    # ---- Component 3: Two-step lookahead (smaller is better) ----
    # For each candidate next node i, compute the best cost to continue with one more node
    lookahead = np.full(n, np.inf)
    for i, node in enumerate(unvisited):
        # Remaining after visiting node
        remaining = [u for u in unvisited if u != node]
        if not remaining:
            # If only one candidate, just go directly to destination from node
            lookahead[i] = d_curr[i] + d_dest[i]   # actually this is just one step, but keep consistent
        else:
            # Best j (second node) among remaining: minimize (dist(node, j) + dist(j, dest))
            dists_to_j = distance_matrix[node][remaining]
            dists_j_to_dest = distance_matrix[destination_node][remaining]
            best_cost = np.min(dists_to_j + dists_j_to_dest)
            lookahead[i] = d_curr[i] + best_cost

    # ---- Normalize each component to [0,1] over the candidates ----
    def normalize(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    prox_norm = normalize(prox)
    sav_norm  = normalize(savings_comp)
    look_norm = normalize(lookahead)

    # ---- Adaptive weights based on number of unvisited ----
    if n > 10:
        w_prox, w_sav, w_look = 0.15, 0.35, 0.50
    elif n > 5:
        w_prox, w_sav, w_look = 0.25, 0.35, 0.40
    else:
        w_prox, w_sav, w_look = 0.40, 0.30, 0.30

    # Composite score (lower is better)
    scores = w_prox * prox_norm + w_sav * sav_norm + w_look * look_norm

    # Return the node with minimum score
    best_idx = int(np.argmin(scores))
    return unvisited[best_idx]
