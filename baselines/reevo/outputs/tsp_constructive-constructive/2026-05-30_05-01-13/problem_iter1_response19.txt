```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a 2-step lookahead with regret.

    For each candidate node, compute:
      - immediate cost: distance from current node to candidate
      - lookahead cost: best distance from candidate to another unvisited node,
        penalised by the regret (difference between best and second best)
      - destination proximity: distance from candidate to the final destination node

    The candidate with the smallest weighted sum is chosen.
    """
    best_score = float('inf')
    best_node = None

    for node in unvisited_nodes:
        immediate = distance_matrix[current_node, node]

        # Remaining nodes after visiting this candidate
        remaining = list(unvisited_nodes - {node})
        n_rem = len(remaining)

        if n_rem >= 2:
            # Distances from candidate to all other unvisited nodes
            dists = distance_matrix[node, remaining]
            sorted_dists = np.sort(dists)
            best_next = sorted_dists[0]
            second_best = sorted_dists[1]
            # Regret term: difference between second best and best
            lookahead = best_next + 0.5 * (second_best - best_next)
        elif n_rem == 1:
            lookahead = distance_matrix[node, remaining[0]]
        else:
            lookahead = 0.0

        # Proximity to the final destination (helps avoid long final returns)
        dest_cost = distance_matrix[destination_node, node]

        # Weighted total score (smaller is better)
        total = immediate + 0.6 * lookahead + 0.4 * dest_cost

        if total < best_score:
            best_score = total
            best_node = node

    return best_node
```
