import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 1-step lookahead with greedy completion.
    For each candidate, simulates a full nearest-neighbour path from current_node
    through the candidate and all remaining nodes, then back to destination_node.
    Returns the candidate that minimizes the total simulated path cost.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_candidate = None
    best_total_cost = float('inf')

    # Pre-convert to list for indexing
    unvisited_list = list(unvisited_nodes)

    for candidate in unvisited_list:
        # Simulate path starting from current_node -> candidate
        visited_set = {current_node, candidate}
        remaining = set(unvisited_list)
        remaining.remove(candidate)

        total_cost = distance_matrix[current_node, candidate]
        current_sim = candidate

        # Greedy completion
        while remaining:
            # Find nearest unvisited node from current_sim
            nearest_node = None
            nearest_dist = float('inf')
            for node in remaining:
                dist = distance_matrix[current_sim, node]
                if dist < nearest_dist:
                    nearest_dist = dist
                    nearest_node = node
            # Move to that node
            total_cost += nearest_dist
            visited_set.add(nearest_node)
            remaining.remove(nearest_node)
            current_sim = nearest_node

        # Return to destination
        total_cost += distance_matrix[current_sim, destination_node]

        if total_cost < best_total_cost:
            best_total_cost = total_cost
            best_candidate = candidate

    # Fallback (should not happen)
    if best_candidate is None:
        best_candidate = min(unvisited_nodes, key=lambda n: distance_matrix[current_node, n])

    return best_candidate
