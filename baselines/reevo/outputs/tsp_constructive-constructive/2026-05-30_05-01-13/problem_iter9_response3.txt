import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit from the unvisited nodes using adaptive weights,
    regret, savings, multi-step lookahead, and minimal normalization."""
    
    # If only one node left, return it immediately
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    total_nodes = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    progress = 1.0 - remaining / total_nodes  # 0 early, 1 late
    
    candidates = list(unvisited_nodes)
    n_cand = len(candidates)
    
    # Precompute component values for each candidate
    d_current = np.zeros(n_cand)
    d_lookahead = np.zeros(n_cand)
    d_regret = np.zeros(n_cand)
    d_savings = np.zeros(n_cand)
    
    for i, node in enumerate(candidates):
        d_current[i] = distance_matrix[current_node][node]
        d_dest = distance_matrix[node][destination_node]
        
        # Lookahead: min distance from node to any other unvisited
        other_dists = [distance_matrix[node][k] for k in unvisited_nodes if k != node]
        if len(other_dists) > 0:
            d_lookahead[i] = min(other_dists)
        else:
            d_lookahead[i] = 0.0
        
        # Regret: difference between second best and best distance to other unvisited nodes
        if len(other_dists) >= 2:
            sorted_dists = sorted(other_dists)
            d_regret[i] = sorted_dists[1] - sorted_dists[0]
        else:
            d_regret[i] = 0.0
        
        # Savings: distance(current, destination) - (distance(current, node) + distance(node, destination))
        d_savings[i] = distance_matrix[current_node][destination_node] - (d_current[i] + d_dest)
    
    # Normalize each component to [0,1] using min-max across candidates
    def normalize(values):
        vmin = np.min(values)
        vmax = np.max(values)
        if vmax == vmin:
            return np.full_like(values, 0.5, dtype=float)
        return (values - vmin) / (vmax - vmin)
    
    norm_current = normalize(d_current)
    norm_lookahead = normalize(d_lookahead)
    norm_regret = normalize(d_regret)
    norm_savings = normalize(d_savings)
    
    # Adaptive weights (exploration early -> exploitation late)
    w_current = 0.2 + 0.2 * progress          # 0.2 -> 0.4
    w_lookahead = 0.5 - 0.2 * progress        # 0.5 -> 0.3
    w_regret = 0.5 - 0.3 * progress           # 0.5 -> 0.2
    w_savings = 0.4 - 0.3 * progress          # 0.4 -> 0.1
    
    # Final score (minimize): distance and lookahead are penalties,
    # regret and savings are rewards (negated)
    scores = (w_current * norm_current +
              w_lookahead * norm_lookahead -
              w_regret * norm_regret -
              w_savings * norm_savings)
    
    best_idx = np.argmin(scores)
    return candidates[best_idx]
