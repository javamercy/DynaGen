import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using adaptive weights, regret, lookahead, and savings.
    
    Parameters
    ----------
    current_node : int
        The node we are currently at.
    destination_node : int
        The final destination (starting node for TSP).
    unvisited_nodes : set
        Set of nodes not yet visited.
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.
    
    Returns
    -------
    int
        The node to visit next.
    """
    if not unvisited_nodes:
        raise ValueError("No unvisited nodes to select from.")
    
    candidates = list(unvisited_nodes)
    n = len(candidates)
    
    if n == 1:
        return candidates[0]
    
    # Compute metrics for each candidate
    d_cur = np.zeros(n)
    d_look = np.zeros(n)
    regret = np.zeros(n)
    savings = np.zeros(n)
    
    for i, node in enumerate(candidates):
        # Distance from current node
        d_cur[i] = distance_matrix[current_node, node]
        
        # Distance from node to destination
        d_dest = distance_matrix[node, destination_node]
        
        # Savings (Clarke-Wright)
        savings[i] = distance_matrix[current_node, destination_node] - (d_cur[i] + d_dest)
        
        # Lookahead: minimum distance from this node to any *other* unvisited node
        other_dists = [distance_matrix[node, other] 
                       for other in unvisited_nodes if other != node]
        if other_dists:
            d_look[i] = min(other_dists)
        else:
            d_look[i] = 0.0
        
        # Regret: difference between second best and best distance to other unvisited nodes
        if len(other_dists) >= 2:
            sorted_dists = sorted(other_dists)
            regret[i] = sorted_dists[1] - sorted_dists[0]
        else:
            regret[i] = 0.0
    
    # Min-max normalization to [0,1] (higher = better for regret and savings)
    def normalize(values):
        vmin, vmax = np.min(values), np.max(values)
        if vmax == vmin:
            return np.full_like(values, 0.5, dtype=float)
        return (values - vmin) / (vmax - vmin)
    
    norm_current = 1.0 - normalize(d_cur)          # invert: high distance is bad
    norm_lookahead = 1.0 - normalize(d_look)       # high lookahead is bad
    norm_regret = normalize(regret)                 # high regret is good
    norm_savings = normalize(savings)               # high savings is good
    
    # Adaptive weights based on progress (fraction of nodes visited)
    total_nodes = distance_matrix.shape[0]
    progress = 1.0 - n / total_nodes   # 0 at start, 1 near end
    
    w_current = 0.2 + 0.3 * progress    # 0.2 → 0.5
    w_lookahead = 0.4 - 0.2 * progress  # 0.4 → 0.2
    w_regret = 0.5 - 0.3 * progress     # 0.5 → 0.2
    w_savings = 0.5 - 0.3 * progress    # 0.5 → 0.2
    
    # Composite score (higher is better)
    score = (w_current * norm_current +
             w_lookahead * norm_lookahead +
             w_regret * norm_regret +
             w_savings * norm_savings)
    
    best_idx = int(np.argmax(score))
    return candidates[best_idx]
