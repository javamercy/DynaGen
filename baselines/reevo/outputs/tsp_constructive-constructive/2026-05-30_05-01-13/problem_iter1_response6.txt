import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node with a look-ahead heuristic that adapts to the remaining number of nodes."""
    n_remaining = len(unvisited_nodes)
    if n_remaining == 1:
        return next(iter(unvisited_nodes))

    # Adaptive weights based on remaining nodes
    # As the path nears its end (fewer unvisited), we emphasise reaching the destination.
    # When many nodes remain, we prefer nodes that are well-connected (small min-distance to others).
    beta = 0.3 + 0.5 * (n_remaining / (n_remaining + 5))
    gamma = 0.2 + 0.3 * (1.0 - n_remaining / (n_remaining + 5))

    best_node = None
    best_score = float('inf')

    for v in unvisited_nodes:
        d_curr_v = distance_matrix[current_node][v]
        d_v_dest = distance_matrix[v][destination_node]

        # Look-ahead: nearest unvisited neighbour from v (excluding v)
        other = [u for u in unvisited_nodes if u != v]
        if other:
            min_dist_to_other = min(distance_matrix[v][u] for u in other)
        else:
            min_dist_to_other = 0.0

        score = d_curr_v + beta * d_v_dest + gamma * min_dist_to_other
        if score < best_score:
            best_score = score
            best_node = v

    return best_node
