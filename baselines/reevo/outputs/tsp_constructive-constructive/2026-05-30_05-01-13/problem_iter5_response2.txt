import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a heuristic that combines:
    - proximity (distance from current node)
    - savings (Clarke-Wright style, wrt the destination node)
    - one-step look-ahead (minimum distance from candidate to another unvisited node)
    
    The score is minimized.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Weights (can be tuned)
    w1, w2, w3 = 0.4, 0.3, 0.3
    
    best_node = None
    best_score = float('inf')
    
    for node in unvisited_nodes:
        dist_cur = distance_matrix[current_node][node]
        # Savings: d(current,dest) + d(node,dest) - d(current,node)
        savings = (distance_matrix[current_node][destination_node] +
                   distance_matrix[node][destination_node] -
                   distance_matrix[current_node][node])
        
        # One-step look-ahead: distance to the nearest unvisited node from candidate
        look_ahead = np.min([distance_matrix[node][other] for other in unvisited_nodes if other != node])
        
        # Score (lower is better): combine proximity, negative savings, and look-ahead
        score = w1 * dist_cur - w2 * savings + w3 * look_ahead
        
        if score < best_score:
            best_score = score
            best_node = node
    
    return best_node
