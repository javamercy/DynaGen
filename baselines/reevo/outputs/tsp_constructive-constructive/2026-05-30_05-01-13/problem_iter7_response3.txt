import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    # If only one node left, it's the obvious choice
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Convert to array for vectorised operations
    unvisited = np.array(list(unvisited_nodes))
    m = len(unvisited)

    # Distances from current and to destination (depot)
    d_curr = distance_matrix[current_node, unvisited]
    d_dest = distance_matrix[destination_node, unvisited]

    max_dist = distance_matrix.max()
    if max_dist == 0:
        max_dist = 1.0  # fallback

    # Pairwise distances among unvisited nodes (m x m)
    d_ij = distance_matrix[unvisited[:, None], unvisited]  # includes diagonal (zero)

    # For each candidate i, compute best 2‑step cost from i:
    #   min_{j != i} ( d(i, j) + d(j, dest) )
    # Use broadcasting, set diagonal to infinity
    step_costs = d_ij + d_dest[None, :]          # (m, m)
    np.fill_diagonal(step_costs, np.inf)
    best_next = step_costs.min(axis=1)          # (m,)

    # ----- Normalised components (all to be minimised) -----
    prox = d_curr / max_dist                     # proximity (small is good)
    sav = (d_curr - d_dest) / max_dist           # negative savings? small is good (positive saving -> negative sav)
    look = best_next / max_dist                  # look‑ahead cost (small is good)

    # ----- Adaptive weights (depend on fraction remaining) -----
    frac_rem = m / distance_matrix.shape[0]
    # When many nodes remain: emphasise look‑ahead and savings
    # When few remain: emphasise proximity
    w_prox = 0.5 * (1.0 - frac_rem)              # ranges 0.0–0.5
    w_sav = 0.3
    w_look = 1.0 - w_prox - w_sav                # ensures sum = 1.0

    # Small adjust: ensure weights are non‑negative
    w_prox = max(0.0, w_prox)

    # ----- Final score (lower is better) -----
    scores = w_prox * prox + w_sav * sav + w_look * look

    # Select the node with minimal score
    best_idx = int(np.argmin(scores))
    return int(unvisited[best_idx])
