import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Select the next node to insert using an enhanced regret heuristic.

    Combines:
    - Clarke-Wright savings: dist(current, dest) - dist(current, u) - dist(u, dest)
    - Two-level regret: difference between the two smallest distances from u to any
      node in {current, destination} ∪ (unvisited \ {u})
    - Proximity: negative of the distance from current to u (prefer close nodes)

    Uses adaptive weights: savings weight decreases as the number of remaining
    nodes shrinks, allowing regret to dominate later to avoid poor choices.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_remaining = len(unvisited_nodes)

    # Adaptive weights: savings matters more early, regret matters more late
    # Weight for savings linearly decreases from 0.6 to 0.2 as n_remaining drops
    w_sav = max(0.2, 0.6 - 0.04 * (n_remaining - 1))
    w_reg = 1.0 - w_sav - 0.1  # reserve 0.1 for proximity
    w_prox = 0.1

    base_saving = distance_matrix[current_node][destination_node]
    best_score = -np.inf
    best_node = None

    # Set of nodes used for regret computation (excluding u)
    # This includes the endpoints of the current edge and all other unvisited nodes.
    reference_set = {current_node, destination_node} | unvisited_nodes

    for u in unvisited_nodes:
        # Savings: insertion cost reduction
        d_cur_u = distance_matrix[current_node][u]
        d_u_dest = distance_matrix[u][destination_node]
        saving = base_saving - d_cur_u - d_u_dest

        # Regret: difference between second‑best and best distance from u
        # to any node in the reference set (excluding u itself).
        # A large regret means u is very close to one node and far from others,
        # making it urgent to insert now.
        others = reference_set - {u}
        if len(others) >= 2:
            dists = sorted(distance_matrix[u][v] for v in others)
            regret = dists[1] - dists[0]
        elif len(others) == 1:
            # Only one other node → regret is 0 (no choice)
            regret = 0.0
        else:
            regret = 0.0

        # Proximity: prefer nodes close to current node
        proximity = -d_cur_u

        score = w_sav * saving + w_reg * regret + w_prox * proximity

        if score > best_score:
            best_score = score
            best_node = u

    return best_node
