```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a full two-step lookahead heuristic.
    
    For each candidate node n in unvisited_nodes, the heuristic evaluates the cost of the path
    current_node -> n -> k -> destination_node, where k is another unvisited node (k != n)
    that minimises this two-step cost. If only one candidate remains, the cost is simply
    current_node -> n -> destination_node. The candidate with the smallest total cost is chosen.
    
    This provides better decisions by considering not only the immediate step but also the
    effect of the choice on the subsequent step.
    """
    best_node = None
    best_score = float('inf')
    
    # Precompute distances from current and to destination for efficiency
    # (though we still need the full matrix for lookahead)
    for n in unvisited_nodes:
        d_cur_n = distance_matrix[current_node][n]
        
        if len(unvisited_nodes) == 1:
            # Only one candidate left; path is forced
            total = d_cur_n + distance_matrix[n][destination_node]
        else:
            # Two-step lookahead: find best next node k (different from n)
            min_two_step = float('inf')
            for k in unvisited_nodes:
                if k == n:
                    continue
                cost = d_cur_n + distance_matrix[n][k] + distance_matrix[k][destination_node]
                if cost < min_two_step:
                    min_two_step = cost
            total = min_two_step
        
        if total < best_score:
            best_score = total
            best_node = n
    
    return best_node
```
