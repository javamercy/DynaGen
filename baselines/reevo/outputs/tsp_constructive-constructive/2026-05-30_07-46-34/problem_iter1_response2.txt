import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a two-step look-ahead heuristic.

    For each candidate node u, compute the cost of the path:
        current -> u -> v -> w
    where v is chosen to minimize (distance[u][v] + min_{w != u,v} distance[v][w]).
    The candidate with the smallest total (d1 + best_two_step) is selected.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = np.inf

    for u in unvisited_nodes:
        d1 = distance_matrix[current_node][u]
        others = unvisited_nodes - {u}
        m = len(others)

        if m >= 2:
            # compute best two-step path: min_{v in others} (dist[u][v] + min_{w != u,v} dist[v][w])
            best_two_step = np.inf
            for v in others:
                others_v = others - {v}
                min_vw = min(distance_matrix[v][w] for w in others_v)
                two_step = distance_matrix[u][v] + min_vw
                if two_step < best_two_step:
                    best_two_step = two_step
        elif m == 1:
            v = next(iter(others))
            best_two_step = distance_matrix[u][v]
        else:  # should not happen because unvisited set non‑empty
            best_two_step = 0.0

        score = d1 + best_two_step
        if score < best_score:
            best_score = score
            best_node = u

    return best_node
