import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select the next node to visit using a normalized two‑step lookahead and
    regret heuristic with adaptive weights that shift from exploration
    (high two‑step weight) early on to exploitation (high destination weight)
    as the number of unvisited nodes decreases.

    Parameters
    ----------
    current_node : int
        Index of the current node.
    destination_node : int
        Index of the final destination (starting node in TSP).
    unvisited_nodes : set
        Set of unvisited node indices.
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.

    Returns
    -------
    int
        Index of the next node to visit.
    """
    # Trivial case: only one candidate
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_total = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    fraction = remaining / n_total  # fraction of nodes still unvisited

    candidates = list(unvisited_nodes)
    m = len(candidates)

    # Pre‑compute the three main components for every candidate
    direct_cost = np.zeros(m)
    dest_distance = np.zeros(m)
    two_step_cost = np.zeros(m)

    for i, node in enumerate(candidates):
        direct_cost[i] = distance_matrix[current_node, node]
        dest_distance[i] = distance_matrix[node, destination_node]
        # Best second step: minimum distance from candidate to any other unvisited node
        others = list(unvisited_nodes - {node})
        best_second = np.min(distance_matrix[node, others]) if others else 0.0
        two_step_cost[i] = direct_cost[i] + best_second

    # Regret: how much worse the candidate's two‑step cost is compared to the best
    best_two_step = np.min(two_step_cost)
    regret = two_step_cost - best_two_step  # always >= 0

    # Min‑max normalization to [0, 1] with safe handling of constant arrays
    def normalize(arr: np.ndarray) -> np.ndarray:
        mn, mx = np.min(arr), np.max(arr)
        if mx == mn:
            return np.full_like(arr, 0.5)
        return (arr - mn) / (mx - mn)

    norm_direct = normalize(direct_cost)
    norm_dest = normalize(dest_distance)
    norm_two = normalize(two_step_cost)
    norm_regret = normalize(regret)

    # Adaptive weights – emphasis shifts as more nodes are visited
    w_direct = 1.0
    w_dest = 2.0 * (1.0 - fraction)       # destination importance grows late in the tour
    w_two = 2.0 * fraction                 # two‑step lookahead matters more when many nodes remain
    w_regret = 1.5 * fraction              # regret plays a larger role early on

    # Composite score (lower is better)
    scores = (
        w_direct * norm_direct
        + w_dest * norm_dest
        + w_two * norm_two
        - w_regret * norm_regret
    )

    best_idx = int(np.argmin(scores))
    return candidates[best_idx]
