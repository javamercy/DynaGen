import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a look-ahead and regret-based heuristic."""
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]
    
    # Precompute distances from each node to destination
    dist_to_dest = {node: distance_matrix[node][destination_node] for node in unvisited}
    # Precompute distances from current node
    dist_from_cur = {node: distance_matrix[current_node][node] for node in unvisited}
    
    # For each candidate, compute a look-ahead cost for a 2-step path
    look_ahead_costs = {}
    for candidate in unvisited:
        # Direct step cost
        direct = dist_from_cur[candidate]
        # Remaining nodes after visiting candidate
        remaining = [n for n in unvisited if n != candidate]
        if remaining:
            # Find the best next node that minimizes (dist from candidate + dist from that node to destination)
            best_continuation = min(
                (distance_matrix[candidate][n] + dist_to_dest[n]) for n in remaining
            )
        else:
            best_continuation = dist_to_dest[candidate]
        look_ahead_costs[candidate] = direct + best_continuation
    
    # Compute regret: difference between best and second-best direct distance
    sorted_direct = sorted(dist_from_cur.values())
    regret = sorted_direct[1] - sorted_direct[0] if len(sorted_direct) > 1 else 0
    
    # Scoring: lower is better, combine look-ahead cost and regret
    # We give more weight to look-ahead cost, but also penalize nodes that increase regret
    scores = {}
    for candidate in unvisited:
        # Normalize look-ahead cost by its maximum
        look_ahead = look_ahead_costs[candidate]
        # Regret component: if this candidate is the best direct, it reduces regret; otherwise increase
        direct = dist_from_cur[candidate]
        if direct == sorted_direct[0]:
            regret_benefit = -regret  # benefit for being best
        else:
            regret_benefit = regret * (direct - sorted_direct[0]) / (sorted_direct[-1] - sorted_direct[0] + 1e-10)
        # Score: lower is better, combine look-ahead (main term) and regret penalty
        scores[candidate] = look_ahead + 0.5 * regret_benefit
    
    return min(scores, key=scores.get)
