import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select the next node using a regret-based look-ahead heuristic.
    Balances immediate cost, future decision robustness (regret), and
    proximity to the destination.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Normalization factor to keep weights meaningful across distance scales
    max_dist = distance_matrix.max()
    if max_dist == 0:
        max_dist = 1

    w1 = 0.4   # immediate cost weight
    w2 = 0.3   # regret weight (positive: we want high regret -> subtract)
    w3 = 0.3   # destination proximity weight

    best_node = None
    best_score = float("inf")

    for node in unvisited_nodes:
        # Immediate cost
        imm = distance_matrix[current_node][node] / max_dist

        # Look-ahead: regret from node to remaining unvisited nodes
        distances_from_node = [
            distance_matrix[node][i] for i in unvisited_nodes if i != node
        ]
        if len(distances_from_node) >= 2:
            sorted_dists = sorted(distances_from_node)
            best_future = sorted_dists[0]
            second_best_future = sorted_dists[1]
            regret = (second_best_future - best_future) / max_dist
        else:
            regret = 0.0

        # Distance to destination (return-path preservation)
        dest_cost = distance_matrix[node][destination_node] / max_dist

        # Combined score (lower is better):
        #   we want small immediate cost, small dest cost, and large regret.
        score = w1 * imm + w3 * dest_cost - w2 * regret

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
