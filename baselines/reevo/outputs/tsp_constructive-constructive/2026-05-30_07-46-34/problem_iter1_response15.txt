import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Look-ahead heuristic: choose the node that minimizes the sum of:
    - distance from current node to candidate,
    - distance from candidate to its nearest neighbour among remaining unvisited nodes (or to destination if none remain),
    - a small bias toward nodes closer to the destination.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node][candidate]

        remaining = unvisited_nodes - {candidate}
        if remaining:
            look_ahead = min(distance_matrix[candidate][j] for j in remaining)
        else:
            # last unvisited node; we will go from candidate directly back to destination
            look_ahead = distance_matrix[candidate][destination_node]

        total = immediate + look_ahead + 0.1 * distance_matrix[candidate][destination_node]

        if total < best_total:
            best_total = total
            best_node = candidate

    return best_node
