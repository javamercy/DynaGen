```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 3-step look-ahead with regret on the first step of the look-ahead.
    For each candidate u, we evaluate the possible next node v (from u) and then greedily choose
    the best third node w (from v). The total path cost for each v is computed.
    The score for candidate u is: dist(current, u) + best_cost - lambda * (second_best_cost - best_cost),
    where best_cost and second_best_cost are the smallest and second smallest path costs among all v.
    """
    lambda_regret = 0.5  # weight for the regret term
    best_score = float('inf')
    best_node = None

    for u in unvisited_nodes:
        remaining_u = unvisited_nodes - {u}
        if not remaining_u:
            # only one node left -> go directly to destination after u
            total = distance_matrix[current_node, u] + distance_matrix[u, destination_node]
        else:
            # evaluate all possible next nodes v from u
            path_costs = []
            for v in remaining_u:
                # cost for the segment: u -> v -> (possibly w) -> destination
                cost_uv = distance_matrix[current_node, u] + distance_matrix[u, v]
                remaining_v = remaining_u - {v}
                if not remaining_v:
                    # no node left after v -> go directly to destination
                    total_cost = cost_uv + distance_matrix[v, destination_node]
                else:
                    # choose the best third node w greedily (nearest neighbor from v)
                    w = min(remaining_v, key=lambda n: distance_matrix[v, n])
                    total_cost = cost_uv + distance_matrix[v, w] + distance_matrix[w, destination_node]
                path_costs.append(total_cost)
            # find best and second best path costs
            path_costs.sort()
            best_cost = path_costs[0]
            second_best_cost = path_costs[1] if len(path_costs) >= 2 else best_cost
            regret = second_best_cost - best_cost
            # total score for candidate u
            total = best_cost - lambda_regret * regret  # note: best_cost already includes dist(current, u)
            # Alternatively we could compute total as (dist(current,u)+...) but best_cost already includes it.
            # The formula above is consistent: best_cost is the best total path cost starting from current_node.
        if total < best_score:
            best_score = total
            best_node = u

    return best_node
```
