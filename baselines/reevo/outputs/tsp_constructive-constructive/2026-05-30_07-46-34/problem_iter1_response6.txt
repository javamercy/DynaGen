import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit with a look-ahead mechanism."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_node = None
    best_score = float('inf')
    alpha = 0.6
    beta = 0.2
    gamma = 0.3
    delta = 0.1
    
    for candidate in unvisited_nodes:
        remaining = unvisited_nodes - {candidate}
        # Look-ahead: distance from candidate to its closest unvisited neighbor
        if remaining:
            nearest_next = min(distance_matrix[candidate][u] for u in remaining)
            # Also compute second nearest for regret
            if len(remaining) >= 2:
                sorted_dists = sorted([distance_matrix[candidate][u] for u in remaining])
                regret = sorted_dists[1] - sorted_dists[0]
            else:
                regret = 0.0
        else:
            nearest_next = 0.0
            regret = 0.0
        
        # Ensure regret not negative
        regret = max(regret, 0.0)
        
        # Base cost: current -> candidate
        base_cost = distance_matrix[current_node][candidate]
        # Look-ahead cost: candidate -> nearest next
        lookahead_cost = alpha * nearest_next
        # Destination attraction: negative distance to destination encourages moving toward it
        dest_cost = - beta * distance_matrix[candidate][destination_node]
        # Regret term: higher regret makes node more urgent to pick now (lower score)
        regret_term = - gamma * regret
        # Add a small diversification term: average distance from candidate to all remaining nodes
        if remaining:
            avg_dist_to_remaining = np.mean([distance_matrix[candidate][u] for u in remaining])
        else:
            avg_dist_to_remaining = 0.0
        diversification = delta * avg_dist_to_remaining
        
        score = base_cost + lookahead_cost + dest_cost + regret_term + diversification
        
        if score < best_score:
            best_score = score
            best_node = candidate
    
    return best_node
