import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a two-step look-ahead with regret."""
    lambda_regret = 0.5  # weight for regret term
    best_score = float('inf')
    best_node = None
    
    for u in unvisited_nodes:
        remaining = unvisited_nodes - {u}
        if not remaining:
            # only one node left, must go to it then to destination
            total = distance_matrix[current_node, u] + distance_matrix[u, destination_node]
        else:
            # compute costs for each possible second node v
            costs_v = []
            for v in remaining:
                cost = distance_matrix[u, v] + distance_matrix[v, destination_node]
                costs_v.append(cost)
            # best and second best costs
            costs_v.sort()
            best_cost = costs_v[0]
            second_best_cost = costs_v[1] if len(costs_v) >= 2 else best_cost
            regret = second_best_cost - best_cost
            
            # look-ahead cost = immediate cost + best future cost - lambda * regret
            total = distance_matrix[current_node, u] + best_cost - lambda_regret * regret
        
        if total < best_score:
            best_score = total
            best_node = u
    
    return best_node
