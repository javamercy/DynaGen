import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead mechanism.
    For each candidate, compute the cost of a complete tour that would result
    if we choose that candidate now, using a quick nearest neighbour completion.
    Choose the candidate that minimises the total cost.
    """
    def nn_tour_cost(start: int, nodes: set, dest: int, dist: np.ndarray) -> float:
        """Return the total cost of a tour starting at 'start',
        visiting all nodes in 'nodes' (using nearest neighbour),
        and ending at 'dest'."""
        if not nodes:
            return dist[start][dest]
        current = start
        remaining = set(nodes)
        total = 0.0
        while remaining:
            # find nearest unvisited node
            nearest = min(remaining, key=lambda n: dist[current][n])
            total += dist[current][nearest]
            current = nearest
            remaining.remove(nearest)
        total += dist[current][dest]
        return total

    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # immediate edge cost from current to candidate
        immediate = distance_matrix[current_node][candidate]
        # remaining nodes after selecting candidate
        remaining = unvisited_nodes - {candidate}
        # cost to finish the tour from candidate
        future = nn_tour_cost(candidate, remaining, destination_node, distance_matrix)
        total = immediate + future
        if total < best_total:
            best_total = total
            best_node = candidate

    return best_node
