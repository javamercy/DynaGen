import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using an adaptive heuristic that combines:
    - two-step lookahead (more weight early)
    - regret (difference between nearest neighbor distance and current distance)
    - distance to destination
    - standard deviation of distances to other unvisited nodes
    All metrics are normalized across candidates.
    """
    # Edge case: only one unvisited node
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    n = len(candidates)

    # Compute raw metrics for each candidate
    d_cur = np.array([distance_matrix[current_node][i] for i in candidates])
    d_dest = np.array([distance_matrix[i][destination_node] for i in candidates])
    two_step = np.zeros(n)
    nearest = np.zeros(n)  # nearest to other unvisited
    std_dev = np.zeros(n)

    for idx, i in enumerate(candidates):
        # distances from i to all other unvisited nodes
        others = [j for j in candidates if j != i]
        dists = np.array([distance_matrix[i][j] for j in others])
        two_step[idx] = d_cur[idx] + np.min(dists) if len(others) > 0 else d_cur[idx] + d_dest[idx]
        nearest[idx] = np.min(dists) if len(others) > 0 else np.inf
        std_dev[idx] = np.std(dists) if len(others) > 0 else 0.0

    # Regret: nearest - d_cur (higher -> more urgent)
    regret = np.where(np.isfinite(nearest), nearest - d_cur, d_cur - d_dest)  # fallback for single neighbor case

    # Normalize each metric to [0,1] across candidates
    def normalize(arr):
        mn, mx = np.min(arr), np.max(arr)
        if mx - mn < 1e-12:
            return np.full_like(arr, 0.5, dtype=float)
        return (arr - mn) / (mx - mn)

    norm_two = normalize(two_step)       # lower is better -> later use 1-norm
    norm_regret = normalize(regret)      # higher is better
    norm_dest = normalize(d_dest)        # lower is better -> later use 1-norm
    norm_std = normalize(std_dev)        # higher is better (as in v1)

    # Adaptive weights based on number of unvisited nodes (proxy for early/late phase)
    # Early: many unvisited -> favour two-step lookahead
    # Late: few unvisited -> favour regret and destination
    n_unvisited = len(unvisited_nodes)
    # Assume total nodes is at least ~50, but we adapt based on n_unvisited
    # Use a smooth transition between n_unvisited=10 (early) and n_unvisited=3 (late)
    alpha = min(1.0, max(0.0, (n_unvisited - 3) / (10 - 3)))  # 0 when <=3, 1 when >=10
    w_lookahead = alpha * 0.6 + (1 - alpha) * 0.1
    w_regret    = alpha * 0.2 + (1 - alpha) * 0.4
    w_dest      = alpha * 0.1 + (1 - alpha) * 0.3
    w_std       = alpha * 0.1 + (1 - alpha) * 0.2

    # Compute score (higher better)
    scores = (w_lookahead * (1 - norm_two) +
              w_regret * norm_regret +
              w_dest * (1 - norm_dest) +
              w_std * norm_std)

    # Select candidate with highest score (ties broken by first occurrence)
    best_idx = np.argmax(scores)
    return candidates[best_idx]
