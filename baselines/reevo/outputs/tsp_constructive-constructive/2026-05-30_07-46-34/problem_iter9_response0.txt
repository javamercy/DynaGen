import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Select the next node to visit using adaptive weights, two-step lookahead,
    and normalized regret."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_total = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    fraction = remaining / n_total  # fraction of unvisited nodes

    candidates = list(unvisited_nodes)
    m = len(candidates)

    # Precompute cost, destination distance, and two-step lookahead cost
    costs = np.zeros(m)
    dest_dists = np.zeros(m)
    two_steps = np.zeros(m)
    for i, node in enumerate(candidates):
        costs[i] = distance_matrix[current_node, node]
        dest_dists[i] = distance_matrix[node, destination_node]
        # two-step: best immediate cost + best possible next step from node
        best_next = np.min(distance_matrix[node, list(unvisited_nodes - {node})])
        two_steps[i] = costs[i] + best_next

    # Regret: difference between this two-step cost and the global best two-step
    best_two_step = np.min(two_steps)
    regrets = two_steps - best_two_step

    # Min-max normalization of each component to [0,1]
    def normalize(arr: np.ndarray) -> np.ndarray:
        mn, mx = np.min(arr), np.max(arr)
        if mx == mn:
            return np.full_like(arr, 0.5)
        return (arr - mn) / (mx - mn)

    norm_costs = normalize(costs)
    norm_dest = normalize(dest_dists)
    norm_two = normalize(two_steps)
    norm_regret = normalize(regrets)

    # Adaptive weights (linear decay/w.r.t. remaining fraction)
    w_cost = 1.0
    w_dest = 2.0 * (1.0 - fraction)   # increases late in the tour
    w_two = 2.0 * fraction            # decreases late in the tour
    w_regret = 1.5 * fraction          # regret matters more early

    # Composite score (lower is better)
    scores = (
        w_cost * norm_costs
        + w_dest * norm_dest
        + w_two * norm_two
        - w_regret * norm_regret
    )

    best_idx = int(np.argmin(scores))
    return candidates[best_idx]
