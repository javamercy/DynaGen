import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved select_next_node using regret as a primary driver.
    Key improvements over v1:
    - Higher weight for regret (distance gap from best immediate neighbor)
    - Additional term for 'future regret' (gap from best future step)
    - Adaptive bias: near the end, emphasise destination more.
    """
    if not unvisited_nodes:
        return current_node

    candidates = list(unvisited_nodes)
    n_c = len(candidates)

    # 1. Immediate costs
    imm = np.array([distance_matrix[current_node][c] for c in candidates])

    # 2. Future costs (best second step)
    fut = np.zeros(n_c)
    for i, c in enumerate(candidates):
        remaining = unvisited_nodes - {c}
        if remaining:
            fut[i] = min(distance_matrix[c][n] for n in remaining)
        # else fut[i] stays 0 (no further nodes)

    # 3. Destination costs
    dest = np.array([distance_matrix[destination_node][c] for c in candidates])

    # 4. Immediate regret: gap from best immediate cost
    best_imm = imm.min()
    reg_imm = imm - best_imm

    # 5. Future regret: gap from best future cost
    best_fut = fut.min() if fut.size > 0 else 0
    reg_fut = fut - best_fut

    # Normalise each criterion to [0,1] (0 = best, 1 = worst)
    def norm(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    imm_norm = norm(imm)
    fut_norm = norm(fut)
    dest_norm = norm(dest)
    reg_imm_norm = norm(reg_imm)
    reg_fut_norm = norm(reg_fut)

    # Weights: regret gets highest share, destination weight increases near end
    remaining_frac = len(candidates) / (len(unvisited_nodes) + 1e-6)  # ~1 if few left
    w_imm = 0.20
    w_fut = 0.15
    w_dest = 0.30 * remaining_frac + 0.10  # 0.10 to 0.40
    w_reg_imm = 0.30
    w_reg_fut = 0.15

    # Lower score is better (minimisation)
    scores = (w_imm * imm_norm +
              w_fut * fut_norm +
              w_dest * dest_norm +
              w_reg_imm * reg_imm_norm +
              w_reg_fut * reg_fut_norm)

    best_idx = np.argmin(scores)
    return candidates[best_idx]
