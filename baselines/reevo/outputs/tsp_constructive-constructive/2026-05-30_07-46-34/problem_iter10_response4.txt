import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a hybrid scoring function that combines:
      - immediate distance from current node,
      - two-step lookahead (immediate + best next distance),
      - distance to the destination,
      - average distance to all other unvisited nodes (future connectivity),
    with min‑max normalization and continuous adaptive weights based on the
    fraction of remaining nodes.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    n = len(candidates)
    total_nodes = distance_matrix.shape[0]

    # ---- Raw criteria ----
    # Immediate distance
    d_cur = np.array([distance_matrix[current_node, u] for u in candidates])

    # Distance to destination
    d_dest = np.array([distance_matrix[u, destination_node] for u in candidates])

    # Two‑step lookahead: immediate + best next (best among other unvisited)
    best_next = np.array([
        min(distance_matrix[u, v] for v in candidates if v != u)
        for u in candidates
    ])
    two_step = d_cur + best_next

    # Future connectivity: average distance to all other unvisited nodes
    avg_future = np.array([
        np.mean([distance_matrix[u, v] for v in candidates if v != u])
        for u in candidates
    ])

    # ---- Min‑max normalization (add small epsilon to avoid division by zero) ----
    def minmax(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.full_like(arr, 0.5)
        return (arr - mn) / (mx - mn)

    n_cur = minmax(d_cur)
    n_dest = minmax(d_dest)
    n_two = minmax(two_step)
    n_avg = minmax(avg_future)

    # ---- Adaptive weights based on remaining fraction ----
    alpha = n / total_nodes          # 1 when all nodes remain, 0 when none
    # We adjust base weights with alpha: more exploration early, more exploitation late
    w_cur   = 0.25 * (0.5 + 0.5 * alpha)
    w_dest  = 0.15 * (1.0 - 0.5 * alpha)
    w_two   = 0.35 * (0.5 + 0.5 * alpha)
    w_avg   = 0.25 * (0.5 + 0.5 * alpha)

    # ---- Composite score (minimisation) ----
    scores = (w_cur * n_cur +
              w_dest * n_dest +
              w_two * n_two +
              w_avg * n_avg)

    return candidates[np.argmin(scores)]
