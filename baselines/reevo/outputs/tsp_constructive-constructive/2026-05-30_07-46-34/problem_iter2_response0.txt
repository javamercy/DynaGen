import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead heuristic with regret and destination proximity.

    For each candidate, computes:
        lookahead_cost = dist(current, candidate) + min_{n in remaining} (dist(candidate, n) + dist(n, destination))
        regret = lookahead_cost - min_lookahead_cost
        score = lookahead_cost + 0.5 * regret + 0.2 * dist(candidate, destination)

    Returns the candidate with the minimum score.
    """
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]

    # Precompute distances
    dist_from_cur = {node: distance_matrix[current_node, node] for node in unvisited}
    dist_to_dest = {node: distance_matrix[node, destination_node] for node in unvisited}
    # Also distance from current to destination (constant)
    # cur_to_dest = distance_matrix[current_node, destination_node]

    # Compute lookahead cost for each candidate
    lookahead = {}
    for candidate in unvisited:
        direct = dist_from_cur[candidate]
        remaining = [n for n in unvisited if n != candidate]
        if remaining:
            # Best continuation: candidate -> next -> destination
            best_future = min(
                distance_matrix[candidate, n] + dist_to_dest[n] for n in remaining
            )
        else:
            # No remaining nodes, go directly to destination
            best_future = dist_to_dest[candidate]
        lookahead[candidate] = direct + best_future

    # Determine best lookahead cost for regret calculation
    min_lookahead = min(lookahead.values())

    # Compute scores
    scores = {}
    for candidate in unvisited:
        la = lookahead[candidate]
        regret = la - min_lookahead
        dest_dist = dist_to_dest[candidate]
        # Weighted score: lookahead is primary, regret penalizes poor choices, dest_dist encourages proximity to end
        scores[candidate] = la + 0.5 * regret + 0.2 * dest_dist

    return min(scores, key=scores.get)
