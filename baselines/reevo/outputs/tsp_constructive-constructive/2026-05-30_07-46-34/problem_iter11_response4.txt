```python
import numpy as np

def select_next_node_v2(current_node: int,
                        destination_node: int,
                        unvisited_nodes: set,
                        distance_matrix: np.ndarray) -> int:
    """Select next node using two-step lookahead, regret, and adaptive weights.

    Steps:
    1. Normalize distances by row max.
    2. For each candidate:
       - d0: normalized distance from current.
       - d_lookahead: min normalized distance to another unvisited node.
       - d_avg: mean normalized distance to other unvisited nodes.
       - regret: gap between second nearest and nearest to candidate.
       - d_dest: normalized distance to destination.
    3. Compute weights via sigmoid over remaining count:
       - w_future = sigmoid(0.2*(remaining - 5))
       - w_dest = 1 - w_future
    4. Score = d0 + w_future * 0.5*(d_lookahead + d_avg) + w_dest * d_dest - 0.5*regret
    5. Return node with minimum score.
    """
    # Precompute row maxima for normalization
    row_max = np.max(distance_matrix, axis=1)
    remaining = len(unvisited_nodes)

    def sigmoid(x: float) -> float:
        return 1.0 / (1.0 + np.exp(-x))

    # Adaptive weights based on remaining nodes
    w_future = sigmoid(0.2 * (remaining - 5))
    w_dest = 1.0 - w_future

    best_score = float('inf')
    best_node = None

    for n in unvisited_nodes:
        # Normalized distance from current
        d0 = distance_matrix[current_node][n] / row_max[current_node]

        # Future connectivity (lookahead and average)
        other_nodes = [o for o in unvisited_nodes if o != n]
        if other_nodes:
            dists_to_other = [distance_matrix[n][o] / row_max[n] for o in other_nodes]
            d_lookahead = min(dists_to_other)
            d_avg = np.mean(dists_to_other)

            # Regret: gap between second best and best distance from n to others
            sorted_dists = sorted(dists_to_other)
            regret_val = sorted_dists[1] - sorted_dists[0] if len(sorted_dists) >= 2 else 0.0
        else:
            d_lookahead = 0.0
            d_avg = 0.0
            regret_val = 0.0

        # Normalized distance to destination
        d_dest = distance_matrix[n][destination_node] / row_max[n] if n != destination_node else 0.0

        # Composite score (lower is better)
        future = 0.5 * (d_lookahead + d_avg)
        score = d0 + w_future * future + w_dest * d_dest - 0.5 * regret_val

        if score < best_score:
            best_score = score
            best_node = n

    return best_node
```
