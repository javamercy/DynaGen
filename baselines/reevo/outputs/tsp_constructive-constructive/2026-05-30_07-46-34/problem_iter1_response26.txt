```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 2-step look-ahead heuristic.

    For each candidate node, evaluate the cost of moving to it and then
    the best possible move to a further unvisited node.  The score is
    a weighted sum of the immediate distance, the look-ahead distance,
    and the distance from the candidate to the destination (to bias toward
    keeping a good return path).
    """
    best_node = None
    best_score = float('inf')
    
    # Precompute distance from current to all candidates not needed now
    
    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node][candidate]
        remaining = unvisited_nodes - {candidate}
        
        if remaining:
            # Best distance to the next node from candidate
            next_min = min(distance_matrix[candidate][k] for k in remaining)
            # Distance from candidate to the final destination (starting node)
            dest_dist = distance_matrix[candidate][destination_node]
            # Score: combine immediate, look-ahead, and destination penalty
            score = immediate + 0.7 * next_min - 0.3 * dest_dist
        else:
            # Last node: we must return directly to destination from here
            score = immediate + distance_matrix[candidate][destination_node]
        
        if score < best_score:
            best_score = score
            best_node = candidate
    
    return best_node
```
