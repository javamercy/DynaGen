import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using look-ahead with MST lower bound."""
    def mst_cost(nodes):
        """Compute MST cost for a list of node indices."""
        n = len(nodes)
        if n <= 1:
            return 0.0
        visited = [False] * n
        min_edge = [float('inf')] * n
        min_edge[0] = 0.0
        total = 0.0
        for _ in range(n):
            # select unvisited node with smallest min_edge
            u = -1
            best = float('inf')
            for i in range(n):
                if not visited[i] and min_edge[i] < best:
                    best = min_edge[i]
                    u = i
            if u == -1:
                break
            visited[u] = True
            total += best
            for v in range(n):
                if not visited[v]:
                    d = distance_matrix[nodes[u]][nodes[v]]
                    if d < min_edge[v]:
                        min_edge[v] = d
        return total

    best_node = None
    best_total = float('inf')
    # Iterate over candidate next nodes
    for candidate in unvisited_nodes:
        # cost to move to candidate
        cost_current = distance_matrix[current_node][candidate]
        # build set of nodes that must be visited including candidate,
        # all other unvisited nodes, and the final destination
        remaining_nodes = list(unvisited_nodes - {candidate}) + [destination_node]
        # add candidate itself to the MST set (start of remaining tour)
        nodes_for_mst = [candidate] + remaining_nodes
        # total estimate = step cost + MST lower bound for remainder
        total_estimate = cost_current + mst_cost(nodes_for_mst)
        if total_estimate < best_total:
            best_total = total_estimate
            best_node = candidate
    return best_node
