import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a regret-based look-ahead heuristic.
    For each candidate, compute the regret (difference between second-best and best distance
    to any other unvisited node) and combine it with the immediate travel cost.
    """
    alpha = 0.5  # weight for regret
    best_score = float('inf')
    best_node = None
    
    for node in unvisited_nodes:
        # Compute distances from candidate node to all other unvisited nodes
        other_nodes = [v for v in unvisited_nodes if v != node]
        if other_nodes:
            dists = [distance_matrix[node][v] for v in other_nodes]
            sorted_dists = sorted(dists)
            # Regret = second best - best
            if len(sorted_dists) >= 2:
                regret = sorted_dists[1] - sorted_dists[0]
            else:
                regret = 0.0
        else:
            regret = 0.0
        
        # Score: want to minimize immediate distance but also favor high regret
        score = distance_matrix[current_node][node] - alpha * regret
        
        if score < best_score:
            best_score = score
            best_node = node
    
    return best_node
