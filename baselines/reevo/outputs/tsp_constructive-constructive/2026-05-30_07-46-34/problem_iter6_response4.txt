import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using an adaptive multi‑criteria scoring function that
    incorporates:
      - immediate distance from current node
      - one‑step lookahead (best distance from candidate to any other unvisited node)
      - average future distance (mean distance to all other unvisited nodes)
      - look‑ahead regret (difference between best and second best future distance)
      - distance to the destination (starting node)
    Weights are adjusted dynamically based on the number of remaining nodes.
    All raw distances are normalized per candidate set to allow consistent weighting.
    """
    candidates = list(unvisited_nodes)
    n = len(candidates)

    if n == 0:
        return destination_node
    if n == 1:
        return candidates[0]

    # Dynamic weight adjustment based on problem phase
    w_imm = 0.30          # immediate distance
    w_best = 0.20         # one‑step lookahead (best future distance)
    w_avg = 0.15          # average future distance
    w_reg = 0.20          # regret (maximize)
    w_dest = 0.15         # distance to destination

    if n <= 5:
        # Few nodes left – focus on returning to destination
        w_dest += 0.10
        w_imm -= 0.05
        w_best -= 0.05
    elif n > 10:
        # Many nodes – emphasize immediate and regret to avoid poor early decisions
        w_imm += 0.05
        w_reg += 0.05
        w_avg -= 0.05
        w_dest -= 0.05

    # Precompute distances to destination for all candidates
    dest_dist = {c: distance_matrix[c, destination_node] for c in candidates}

    # Will store raw scores (unnormalized), then normalize
    scores_raw = np.zeros(n)

    # For normalization we collect individual components for all candidates
    imm_list = np.array([distance_matrix[current_node, c] for c in candidates])
    best_list = np.zeros(n)
    avg_list = np.zeros(n)
    reg_list = np.zeros(n)
    dest_list = np.array([dest_dist[c] for c in candidates])

    for i, c in enumerate(candidates):
        other = [x for x in candidates if x != c]
        if other:
            other_dists = np.array([distance_matrix[c, x] for x in other])
            best_list[i] = other_dists.min()
            avg_list[i] = other_dists.mean()
            if len(other_dists) >= 2:
                sorted_dists = np.sort(other_dists)
                best = sorted_dists[0]
                second_best = sorted_dists[1]
                reg_list[i] = second_best - best
            else:
                reg_list[i] = 0.0
        else:
            best_list[i] = 0.0
            avg_list[i] = 0.0
            reg_list[i] = 0.0

    # Min‑max normalization for each component (0 = best, 1 = worst)
    def normalize(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    imm_norm = normalize(imm_list)
    best_norm = normalize(best_list)
    avg_norm = normalize(avg_list)
    reg_norm = normalize(reg_list)
    dest_norm = normalize(dest_list)

    # Scoring: lower is better.
    # For regret, we want to maximize it, so we subtract its normalized value.
    scores = (w_imm * imm_norm +
              w_best * best_norm +
              w_avg * avg_norm -
              w_reg * reg_norm +
              w_dest * dest_norm)

    best_idx = np.argmin(scores)
    return candidates[best_idx]
