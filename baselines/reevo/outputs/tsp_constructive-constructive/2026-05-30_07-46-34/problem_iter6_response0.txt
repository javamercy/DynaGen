import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using multiple normalized criteria, adaptive weights,
    and statistical dispersion.
    """
    unvisited = list(unvisited_nodes)
    n = len(unvisited)
    if n == 0:
        return destination_node
    if n == 1:
        return unvisited[0]

    # Precompute distances to destination for all candidates
    dest_dists = {node: distance_matrix[node, destination_node] for node in unvisited}

    # Prepare containers for criteria values across candidates
    d_curr_vals = []
    d_dest_vals = []
    avg_future_vals = []
    regret_vals = []
    std_vals = []

    for node in unvisited:
        # Immediate distance
        d_curr = distance_matrix[current_node, node]
        d_curr_vals.append(d_curr)

        # Distance to destination
        d_dest = dest_dists[node]
        d_dest_vals.append(d_dest)

        # Distances to other unvisited nodes
        others = [v for v in unvisited if v != node]
        if others:
            other_dists = [distance_matrix[node, v] for v in others]
            avg_future = np.mean(other_dists)
            std_future = np.std(other_dists) if len(other_dists) > 1 else 0.0
            sorted_dists = sorted(other_dists)
            best = sorted_dists[0]
            second_best = sorted_dists[1] if len(sorted_dists) >= 2 else best
            regret = second_best - best
        else:
            avg_future = 0.0
            std_future = 0.0
            regret = 0.0

        avg_future_vals.append(avg_future)
        regret_vals.append(regret)
        std_vals.append(std_future)

    # Normalize each criterion between 0 and 1 (higher is worse, except regret)
    def normalize(arr):
        arr = np.array(arr, dtype=float)
        mn, mx = arr.min(), arr.max()
        if mx - mn == 0:
            return np.full_like(arr, 0.5)
        return (arr - mn) / (mx - mn)

    norm_d_curr = normalize(d_curr_vals)
    norm_d_dest = normalize(d_dest_vals)
    norm_avg_future = normalize(avg_future_vals)
    norm_regret_raw = normalize(regret_vals)
    norm_regret = 1.0 - norm_regret_raw          # high regret becomes low cost
    norm_std = normalize(std_vals)               # lower std preferred

    # Adaptive weights
    w_immediate = 0.30
    w_avg_future = 0.20
    w_regret = 0.20
    w_std = 0.10
    w_dest = 0.20

    # Phase adaptation: fewer nodes -> emphasize destination, less future
    if n <= 5:
        w_dest += 0.10
        w_avg_future -= 0.05
        w_immediate -= 0.05
    elif n > 10:
        w_regret += 0.05
        w_avg_future += 0.05
        w_immediate -= 0.05
        w_dest -= 0.05

    # CV-based adaptation: if immediate distances vary a lot, give them more weight
    d_curr_arr = np.array(d_curr_vals)
    cv_immediate = d_curr_arr.std() / (d_curr_arr.mean() + 1e-9)
    if cv_immediate > 0.5:
        w_immediate += 0.10
        # reduce others proportionally
        reduction = 0.10 * (w_avg_future + w_regret + w_std + w_dest)
        scale = 1 - reduction / (w_avg_future + w_regret + w_std + w_dest + 1e-9)
        w_avg_future *= scale
        w_regret *= scale
        w_std *= scale
        w_dest *= scale

    # Compute combined score (lower is better)
    scores = []
    for i in range(n):
        score = (w_immediate * norm_d_curr[i] +
                 w_dest * norm_d_dest[i] +
                 w_avg_future * norm_avg_future[i] +
                 w_regret * norm_regret[i] +
                 w_std * norm_std[i])
        scores.append(score)

    # Return node with lowest score
    return unvisited[np.argmin(scores)]
