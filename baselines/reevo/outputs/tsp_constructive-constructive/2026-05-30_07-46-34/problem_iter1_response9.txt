import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using look-ahead nearest neighbor heuristic.
    For each candidate, simulate a greedy nearest neighbor tour from that candidate 
    through all remaining unvisited nodes ending at the destination.
    Choose the candidate that minimizes the total predicted tour length.
    """
    best_node = None
    best_cost = float('inf')
    for candidate in unvisited_nodes:
        # Simulate NN from candidate
        remaining = set(unvisited_nodes) - {candidate}
        curr = candidate
        tour_cost = distance_matrix[current_node][candidate]
        while remaining:
            # nearest neighbor among remaining
            nn = min(remaining, key=lambda x: distance_matrix[curr][x])
            tour_cost += distance_matrix[curr][nn]
            curr = nn
            remaining.remove(nn)
        # Add return to destination
        tour_cost += distance_matrix[curr][destination_node]
        if tour_cost < best_cost:
            best_cost = tour_cost
            best_node = candidate
    return best_node
