import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Select next node using normalized two‑step lookahead, regret, and statistics.

    Combines immediate cost, destination distance, two‑step lookahead,
    average and standard deviation of distances to other unvisited nodes,
    and regret (second‑best minus best distance from candidate).
    All components are min‑max normalized and combined with tuned weights.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    nodes = list(unvisited_nodes)
    n = len(nodes)

    # Preallocate arrays
    imm = np.empty(n)
    dest = np.empty(n)
    two_step = np.empty(n)
    regret = np.empty(n)
    avg_other = np.empty(n)
    std_other = np.empty(n)

    for i, node in enumerate(nodes):
        imm[i] = distance_matrix[current_node, node]
        dest[i] = distance_matrix[destination_node, node]

        # Distances from candidate to all other unvisited nodes
        others = [v for v in unvisited_nodes if v != node]
        if others:
            dists = distance_matrix[node, others]
            avg_other[i] = np.mean(dists)
            std_other[i] = np.std(dists)
            min_dist = np.min(dists)
            two_step[i] = imm[i] + min_dist
            if len(dists) >= 2:
                sorted_dists = np.sort(dists)
                regret[i] = sorted_dists[1] - sorted_dists[0]
            else:
                regret[i] = 0.0
        else:
            avg_other[i] = 0.0
            std_other[i] = 0.0
            two_step[i] = imm[i]
            regret[i] = 0.0

    # Min‑max normalization to [0,1] (eps avoids division by zero)
    eps = 1e-9

    def normalize(arr: np.ndarray) -> np.ndarray:
        mn, mx = np.min(arr), np.max(arr)
        if mx - mn < eps:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    imm_norm = normalize(imm)
    dest_norm = normalize(dest)
    two_step_norm = normalize(two_step)
    avg_norm = normalize(avg_other)
    std_norm = normalize(std_other)
    regret_norm = normalize(regret)

    # Weights (tuned for balance)
    w_imm = 0.30
    w_dest = 0.15
    w_two_step = 0.20
    w_avg = 0.10
    w_std = 0.10
    w_regret = 0.15

    # Score – higher is better. Costs are negated, benefits (std, regret) are added.
    scores = (
        - w_imm * imm_norm
        - w_dest * dest_norm
        - w_two_step * two_step_norm
        - w_avg * avg_norm
        + w_std * std_norm
        + w_regret * regret_norm
    )

    best_idx = np.argmax(scores)
    return nodes[best_idx]
