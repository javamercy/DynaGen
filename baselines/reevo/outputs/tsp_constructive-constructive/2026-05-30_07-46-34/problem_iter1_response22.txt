import numpy as np
from typing import Set

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: Set[int], distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead mechanism.
    For each candidate, simulate a complete greedy path from that node to the destination
    covering all remaining nodes, and choose the candidate that minimizes the total cost.
    """
    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # start with edge from current to candidate
        total = distance_matrix[current_node][candidate]
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        current = candidate

        # greedy path covering all remaining nodes
        while remaining:
            # nearest neighbor
            next_node = min(remaining, key=lambda n: distance_matrix[current][n])
            total += distance_matrix[current][next_node]
            current = next_node
            remaining.remove(next_node)

        # return to destination
        total += distance_matrix[current][destination_node]

        if total < best_total:
            best_total = total
            best_node = candidate

    return best_node
