import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node with adaptive weights and combined look-ahead."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n = len(unvisited_nodes)
    # Base weights – will be adapted by remaining number of nodes
    alpha = 0.5 * (1 - (1 / (n + 1)))   # look-ahead importance decreases as we progress
    beta = 0.3 * (1 + (1 / (n + 1)))    # destination attraction increases near the end
    gamma = 0.2                          # regret weight
    delta = 0.1                          # diversification (anti-clustering) weight

    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        remaining = unvisited_nodes - {candidate}

        # Immediate cost
        base_cost = distance_matrix[current_node][candidate]

        # Look-ahead: nearest neighbor from candidate among remaining
        if remaining:
            nearest_next = min(distance_matrix[candidate][u] for u in remaining)
            # Regret: difference between second best and best from candidate to remaining
            if len(remaining) >= 2:
                dists = [distance_matrix[candidate][u] for u in remaining]
                dists.sort()
                regret = dists[1] - dists[0]
            else:
                regret = 0.0
        else:
            nearest_next = 0.0
            regret = 0.0

        # Destination attraction (negative because we want small distance to destination)
        dest_attraction = -beta * distance_matrix[candidate][destination_node]

        # Regret term: higher regret → more urgent to pick now → lower score
        regret_term = -gamma * max(regret, 0.0)

        # Diversification: penalize candidates that are far from remaining nodes
        if remaining:
            avg_dist = np.mean([distance_matrix[candidate][u] for u in remaining])
        else:
            avg_dist = 0.0
        diversification = delta * avg_dist

        score = base_cost + alpha * nearest_next + dest_attraction + regret_term + diversification

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
