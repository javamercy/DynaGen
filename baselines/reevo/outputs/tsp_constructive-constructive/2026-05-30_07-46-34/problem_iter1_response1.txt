```python
import numpy as np
from typing import Set

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: Set[int], distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 2-step look-ahead with nearest neighbor completion.
    For each candidate node, compute the total estimated cost:
        cost(current->candidate) + estimated completion cost from candidate to destination
    (where completion uses greedy nearest neighbor to visit all remaining nodes and end at destination).
    The candidate with the smallest total cost is chosen.
    """
    unvisited = list(unvisited_nodes)  # convert to list for indexing

    def completion_cost(start: int, remaining: Set[int], dest: int) -> float:
        """greedy nearest neighbor completion cost from start to dest visiting all remaining nodes."""
        if not remaining:
            return distance_matrix[start][dest]
        current = start
        unvisited_rem = set(remaining)
        total = 0.0
        while unvisited_rem:
            # Find nearest
            best_node = min(unvisited_rem, key=lambda n: distance_matrix[current][n])
            total += distance_matrix[current][best_node]
            current = best_node
            unvisited_rem.remove(current)
        total += distance_matrix[current][dest]
        return total

    best_score = float('inf')
    best_node = None
    for node in unvisited:
        remaining = set(n for n in unvisited if n != node)
        # Estimated total cost if we go to node now
        total_cost = distance_matrix[current_node][node] + completion_cost(node, remaining, destination_node)
        if total_cost < best_score:
            best_score = total_cost
            best_node = node
    # Tie-breaking: if multiple nodes have same cost, choose the one with smaller distance to current (optional)
    # But we already break ties by first encountered, which is fine.
    return best_node
```
