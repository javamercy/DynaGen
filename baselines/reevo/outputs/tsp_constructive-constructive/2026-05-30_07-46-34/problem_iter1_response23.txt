```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Selects the next node to visit using a look-ahead mechanism with greedy completion.
    For each candidate node, it simulates a nearest-neighbor tour over the remaining unvisited
    nodes (including the final return to the destination) and picks the candidate that minimizes
    the total estimated tour length.
    """
    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # immediate distance from current to candidate
        immediate = distance_matrix[current_node][candidate]

        # simulate greedy nearest-neighbor completion from the candidate
        visited_sim = {candidate}
        remaining = set(unvisited_nodes) - {candidate}
        current_sim = candidate
        simulated_dist = 0.0

        while remaining:
            # find nearest unvisited node
            next_sim = min(remaining, key=lambda x: distance_matrix[current_sim][x])
            simulated_dist += distance_matrix[current_sim][next_sim]
            visited_sim.add(next_sim)
            remaining.remove(next_sim)
            current_sim = next_sim

        # return to destination
        simulated_dist += distance_matrix[current_sim][destination_node]

        total_estimate = immediate + simulated_dist

        if total_estimate < best_total:
            best_total = total_estimate
            best_node = candidate

    return best_node
```
