```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 2-step look-ahead heuristic with stronger emphasis on keeping a short return path.
    
    For each candidate, evaluate the immediate cost, the best next step from the candidate, and the distance from
    the candidate to the destination. The score is a weighted combination where the destination distance is 
    penalized (subtracted) to favor nodes that keep the return path short. The weights are chosen to balance 
    immediate and future costs while prioritizing the return path.
    """
    best_node = None
    best_score = float('inf')
    
    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node][candidate]
        remaining = unvisited_nodes - {candidate}
        
        if remaining:
            # Look-ahead: best next hop from candidate to another unvisited node
            next_min = min(distance_matrix[candidate][k] for k in remaining)
            # Distance from candidate back to the starting node (destination)
            dest_dist = distance_matrix[candidate][destination_node]
            # Score: immediate + 0.6 * look-ahead - 0.4 * dest_dist
            # The negative coefficient on dest_dist encourages selecting nodes with shorter return route.
            score = immediate + 0.6 * next_min - 0.4 * dest_dist
        else:
            # Last node: must go directly to destination
            score = immediate + distance_matrix[candidate][destination_node]
        
        if score < best_score:
            best_score = score
            best_node = candidate
    
    return best_node
```
