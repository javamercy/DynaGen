import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using look-ahead with regret and weighted multi-criteria scoring.
    
    Criteria:
    - Immediate distance: cost to go to the candidate now.
    - Average future distance: encourages visiting isolated nodes early.
    - Regret (second smallest - smallest future distance): added as positive term (higher regret penalized).
    - Standard deviation (diversification): spreads choices.
    - Distance to destination: preserves return path.
    """
    unvisited = list(unvisited_nodes)
    n_unvisited = len(unvisited)
    if n_unvisited == 0:
        return destination_node
    if n_unvisited == 1:
        return unvisited[0]

    # Fixed weights
    w1, w2, w3, w4, w5 = 0.3, 0.3, 0.2, 0.1, 0.1

    scores = {}
    for node in unvisited:
        # Immediate distance
        d_curr = distance_matrix[current_node][node]

        # Distances from node to all other unvisited nodes
        other_distances = [distance_matrix[node][i] for i in unvisited if i != node]

        if other_distances:
            avg_future = np.mean(other_distances)
            std_future = np.std(other_distances)
            sorted_dist = sorted(other_distances)
            if len(sorted_dist) >= 2:
                d1, d2 = sorted_dist[0], sorted_dist[1]
                regret_future = d2 - d1   # look-ahead regret measure
            else:
                regret_future = 0.0
        else:
            avg_future = 0.0
            std_future = 0.0
            regret_future = 0.0

        # Distance to destination (return path preservation)
        d_dest = distance_matrix[node][destination_node]

        # Weighted score: we minimize this score
        # Regret is added as a positive term (reflection: "add regret as positive term")
        score = (w1 * d_curr
                 - w2 * avg_future
                 + w3 * regret_future    # Changed from - to +
                 + w4 * std_future
                 - w5 * d_dest)
        scores[node] = score

    # Node with minimum score is selected
    return min(scores, key=scores.get)
