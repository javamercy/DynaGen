import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                         unvisited_nodes: set,
                         distance_matrix: np.ndarray) -> int:
    """
    Improved node selection using adaptive weights, two-step lookahead,
    regret from second-best distance, and hubness (average distance).
    Weights are adjusted based on progress and variance of each metric.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    total_nodes = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    progress = remaining / (total_nodes - 1)  # 1 when all unvisited, 0 when one left

    nodes = list(unvisited_nodes)
    n = len(nodes)

    # Precompute arrays for metrics
    d_curr_arr = np.array([distance_matrix[current_node, node] for node in nodes])
    d_dest_arr = np.array([distance_matrix[node, destination_node] for node in nodes])
    lookahead_arr = np.empty(n)
    regret_arr = np.empty(n)
    hubness_arr = np.empty(n)

    for i, node in enumerate(nodes):
        # Two-step lookahead: current + best next step
        other_unvisited = unvisited_nodes - {node}
        next_dists = []
        if other_unvisited:
            next_dists.extend(distance_matrix[node, j] for j in other_unvisited)
        # Also consider direct to destination if not yet visited
        if destination_node not in unvisited_nodes:
            next_dists.append(distance_matrix[node, destination_node])
        best_next = min(next_dists) if next_dists else 0.0
        lookahead_arr[i] = d_curr_arr[i] + best_next

        # Regret: second best distance from current to other unvisited minus d_curr
        other_dists = [distance_matrix[current_node, k] for k in unvisited_nodes if k != node]
        if len(other_dists) >= 2:
            sorted_dists = sorted(other_dists)
            second_best = sorted_dists[1]
            regret_arr[i] = max(0.0, second_best - d_curr_arr[i])
        else:
            regret_arr[i] = 0.0

        # Hubness: average distance to other unvisited nodes
        if other_unvisited:
            hubness_arr[i] = np.mean([distance_matrix[node, j] for j in other_unvisited])
        else:
            hubness_arr[i] = 0.0

    # Normalise each metric to z-scores (0 mean, 1 std) across candidates
    # For regret we want larger values to be better (so we will subtract it)
    # For all others, smaller is better.
    metrics = {
        'lookahead': lookahead_arr,
        'dest': d_dest_arr,
        'regret': regret_arr,   # larger is better → will be subtracted
        'hubness': hubness_arr
    }
    normed = {}
    for name, arr in metrics.items():
        std = np.std(arr)
        if std < 1e-12 or np.isclose(std, 0.0):
            normed[name] = np.zeros(n)
        else:
            normed[name] = (arr - np.mean(arr)) / std

    # Compute dynamic weights based on progress and variance of each metric
    # Base weights (progress‑dependent)
    w_look_base = 0.3 + 0.2 * (1 - progress)
    w_dest_base = 0.1 + 0.4 * progress
    w_regret_base = 0.1 + 0.3 * (1 - progress)
    w_hub_base = 0.1 + 0.2 * progress

    # Supplement weights by variance (more variance → more informative)
    var_look = np.var(lookahead_arr) if n > 1 else 0.0
    var_dest = np.var(d_dest_arr) if n > 1 else 0.0
    var_regret = np.var(regret_arr) if n > 1 else 0.0
    var_hub = np.var(hubness_arr) if n > 1 else 0.0
    # Combine variances with base weights (avoid dominance)
    eps = 1e-8
    w_look = w_look_base * (1.0 + var_look) + eps
    w_dest = w_dest_base * (1.0 + var_dest) + eps
    w_regret = w_regret_base * (1.0 + var_regret) + eps
    w_hub = w_hub_base * (1.0 + var_hub) + eps
    total = w_look + w_dest + w_regret + w_hub
    w_look /= total
    w_dest /= total
    w_regret /= total
    w_hub /= total

    # Compute combined score (lower is better)
    # regret is subtracted because higher regret should be preferred (avoid missing good connections)
    scores = (w_look * normed['lookahead'] +
              w_dest * normed['dest'] -
              w_regret * normed['regret'] +
              w_hub * normed['hubness'])

    best_idx = int(np.argmin(scores))
    return nodes[best_idx]
