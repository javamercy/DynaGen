import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using look-ahead, regret, and weighted multi-criteria scoring.
    Balances immediate cost, future cost (via look-ahead), regret (criticality of the choice),
    and distance to the destination (return-path preservation).
    """
    # Trivial case: only one candidate left
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Weights for the scoring components (tuned for simplicity and balance)
    w_imm   = 0.4   # immediate distance from current node
    w_look  = 0.2   # look‑ahead: minimum distance from candidate to remaining nodes
    w_regret= 0.3   # regret: difference between best and second‑best next step from candidate
    w_dest  = 0.1   # distance from candidate to the destination node

    scores = {}
    for candidate in unvisited_nodes:
        immediate_cost = distance_matrix[current_node][candidate]

        # Remaining nodes after choosing this candidate
        remaining = unvisited_nodes - {candidate}

        if remaining:
            # Look‑ahead: best possible next edge from the candidate
            look_ahead_cost = min(distance_matrix[candidate][m] for m in remaining)

            # Regret: second‑best minus best edge from candidate (zero if only one option)
            if len(remaining) >= 2:
                distances_to_remaining = [distance_matrix[candidate][m] for m in remaining]
                distances_to_remaining.sort()
                regret = distances_to_remaining[1] - distances_to_remaining[0]
            else:
                regret = 0.0
        else:
            # No nodes left: next step is directly to the destination
            look_ahead_cost = distance_matrix[candidate][destination_node]
            regret = 0.0

        destination_cost = distance_matrix[candidate][destination_node]

        # Composite score (minimize)
        score = (w_imm * immediate_cost
                 + w_look * look_ahead_cost
                 - w_regret * regret
                 + w_dest * destination_cost)

        scores[candidate] = score

    # Return the candidate with the smallest score
    return min(scores, key=scores.get)
