import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved version focusing on regret terms and adaptive destination weighting.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    n = len(candidates)

    # Immediate distances from current node
    imm = np.array([distance_matrix[current_node][c] for c in candidates])

    # Future distances: minimum distance from candidate to another unvisited node
    fut = np.zeros(n)
    for i, c in enumerate(candidates):
        others = unvisited_nodes - {c}
        if others:
            fut[i] = min(distance_matrix[c][o] for o in others)
        # else stays 0

    # Distances to destination (starting node, return point)
    dest = np.array([distance_matrix[destination_node][c] for c in candidates])

    # Best immediate and future costs (global minima)
    best_imm = np.min(imm)
    best_fut = np.min(fut)

    # Per-candidate regrets: how much worse than the best possible
    per_regret_imm = imm - best_imm                     # range [0, ...]
    per_regret_fut = fut - best_fut

    # Normalize each component to [0,1] (0 = best, 1 = worst)
    def normalize(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.full_like(arr, 0.5, dtype=float)
        return (arr - mn) / (mx - mn)

    imm_norm = normalize(imm)
    fut_norm = normalize(fut)
    dest_norm = normalize(dest)
    reg_imm_norm = normalize(per_regret_imm)
    reg_fut_norm = normalize(per_regret_fut)

    # Adaptive weight for destination: increases as fewer nodes remain
    remaining_frac = len(unvisited_nodes) / (len(unvisited_nodes) + 1.0)
    w_imm = 0.3
    w_fut = 0.15
    w_dest = 0.2 + 0.3 * (1.0 - remaining_frac)        # ranges from 0.2 to 0.5
    w_reg_imm = 0.2
    w_reg_fut = 0.15

    # Combined score: lower is better.
    # Regret terms are subtracted because higher regret (cost of skipping) should be preferred.
    scores = (w_imm * imm_norm +
              w_fut * fut_norm +
              w_dest * dest_norm -
              w_reg_imm * reg_imm_norm -
              w_reg_fut * reg_fut_norm)

    best_idx = int(np.argmin(scores))
    return candidates[best_idx]
