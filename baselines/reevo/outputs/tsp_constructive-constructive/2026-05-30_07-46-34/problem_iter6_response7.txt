def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Improved constructive heuristic for TSP. Normalizes distances per row,
    uses two-step lookahead, and adapts weights with a sigmoid progress curve.
    Higher regret is rewarded (subtracted), immediate/lookahead/destination penalized.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_remaining = len(unvisited_nodes)
    total_nodes = distance_matrix.shape[0]
    # progress: 0 at start, 1 at end (when only destination left)
    progress = 1.0 - (n_remaining / total_nodes)

    # Sigmoid-like weight adaptation: more aggressive shift at middle
    # w_imm decreases, w_dest increases, w_reg increases, w_lookahead decreases
    sig = 1.0 / (1.0 + np.exp(-5.0 * (progress - 0.5)))  # 0->0.07, 0.5->0.5, 1->0.93
    w_imm = 0.5 - 0.3 * sig          # 0.5 -> 0.2
    w_look = 0.3 - 0.2 * sig         # 0.3 -> 0.1
    w_dest = 0.2 + 0.4 * sig         # 0.2 -> 0.6
    w_reg = 0.2 + 0.2 * sig          # 0.2 -> 0.4

    best_node = None
    best_score = float("inf")

    current_row = distance_matrix[current_node, :]
    max_from_current = current_row.max()
    if max_from_current == 0:
        max_from_current = 1

    for node in unvisited_nodes:
        # Immediate cost normalized per row of current node
        imm = distance_matrix[current_node, node] / max_from_current

        # Remaining nodes (excluding current candidate)
        remaining = unvisited_nodes - {node}
        if remaining:
            # distances from candidate to all remaining
            cand_dist_list = [distance_matrix[node, u] for u in remaining]
            min_future = min(cand_dist_list)
            if len(cand_dist_list) >= 2:
                sorted_dists = sorted(cand_dist_list)
                regret = (sorted_dists[1] - sorted_dists[0])
                # normalize regret by max distance from candidate
                max_from_cand = max(cand_dist_list)
                if max_from_cand > 0:
                    regret /= max_from_cand
                else:
                    regret = 0.0
            else:
                regret = 0.0
            # Two-step lookahead: immediate + min future, normalized by sum of max possible
            # Use per-row normalization: divide by (max_from_current + max_from_cand)
            max_from_cand = max(cand_dist_list) if cand_dist_list else 1
            look = (distance_matrix[current_node, node] + min_future) / (max_from_current + max_from_cand)
        else:
            # Only destination remains: go directly
            min_future = distance_matrix[node, destination_node]
            regret = 0.0
            look = (distance_matrix[current_node, node] + min_future) / (max_from_current + distance_matrix[node, destination_node])

        # Destination cost normalized by max distance from candidate
        dest_cost = distance_matrix[node, destination_node]
        # Use per‑node max for destination normalization
        node_max = distance_matrix[node, :].max()
        if node_max > 0:
            dest_cost /= node_max
        else:
            dest_cost = 0.0

        # Score (minimize)
        score = (w_imm * imm +
                 w_look * look +
                 w_dest * dest_cost -
                 w_reg * regret)

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
