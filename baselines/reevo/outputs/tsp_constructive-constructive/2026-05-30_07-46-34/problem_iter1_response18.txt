import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a regret-based look-ahead heuristic."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Precompute distances from current node to all unvisited
    cur_to_unvisited = {u: distance_matrix[current_node][u] for u in unvisited_nodes}

    # For each candidate, compute look-ahead metrics
    scores = {}
    for u in unvisited_nodes:
        # Immediate cost
        immediate = cur_to_unvisited[u]

        # Distances from u to all other unvisited nodes and to destination
        other_distances = [distance_matrix[u][v] for v in unvisited_nodes if v != u]
        if len(other_distances) >= 2:
            sorted_dist = sorted(other_distances)
            nearest = sorted_dist[0]
            second_nearest = sorted_dist[1]
            regret = second_nearest - nearest
        elif len(other_distances) == 1:
            nearest = other_distances[0]
            regret = 0.0  # no alternative
        else:
            nearest = 0.0
            regret = 0.0

        dest_distance = distance_matrix[u][destination_node]

        # Weighted score (minimization)
        score = (1.0 * immediate +
                 0.5 * nearest -
                 0.3 * regret +
                 0.2 * dest_distance)
        scores[u] = score

    # Return the node with minimum score
    return min(scores, key=scores.get)
