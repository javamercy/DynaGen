import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using two-step lookahead and regret, with sigmoid-weighted emphasis."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_nodes = len(distance_matrix)
    # Precompute row-wise maximum distances for normalization
    row_max = np.max(distance_matrix, axis=1)

    # Sigmoid weight: high weight for future aspects when many nodes remain
    remaining = len(unvisited_nodes)
    fraction = remaining / n_nodes
    # steepness 10, midpoint 0.5: w_close to 1 for fraction>0.5, close to 0 for fraction<0.5
    w = 1.0 / (1.0 + np.exp(10 * (fraction - 0.5)))
    beta = 0.5  # weight for regret penalty

    best_candidate = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        # Normalized distance from current to candidate
        norm_current_dist = distance_matrix[current_node][candidate] / row_max[current_node]

        # Consider next step from candidate (excluding candidate itself)
        other_unvisited = [u for u in unvisited_nodes if u != candidate]
        n_other = len(other_unvisited)

        # Two-step lookahead: best next normalized distance and regret
        best_next = 0.0
        regret = 0.0
        if n_other >= 2:
            # Normalize distances from candidate to others using its own row max
            dists = [distance_matrix[candidate][u] / row_max[candidate] for u in other_unvisited]
            sorted_dists = sorted(dists)
            best_next = sorted_dists[0]
            second_best = sorted_dists[1]
            regret = second_best - best_next
        elif n_other == 1:
            # Only one other node
            best_next = distance_matrix[candidate][other_unvisited[0]] / row_max[candidate]

        # Future component (current step + best next, penalized by regret)
        future = norm_current_dist + best_next - beta * regret

        # Normalized distance from candidate to destination
        dest_dist = distance_matrix[candidate][destination_node] / row_max[candidate]

        # Combined score
        score = w * future + (1 - w) * dest_dist

        if score < best_score:
            best_score = score
            best_candidate = candidate

    return best_candidate
