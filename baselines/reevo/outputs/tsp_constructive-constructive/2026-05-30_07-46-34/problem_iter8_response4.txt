import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using adaptive multi-criteria: two-step lookahead, regret, destination proximity, and average distance to other unvisited nodes."""
    n = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    if remaining == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    d_cur_vals = []
    two_step_vals = []
    regret_vals = []
    d_dest_vals = []
    avg_vals = []

    for u in candidates:
        d_cur = distance_matrix[current_node][u]
        d_dest = distance_matrix[u][destination_node]
        other = unvisited_nodes - {u}
        if other:
            dists_to_others = [distance_matrix[u][v] for v in other]
            min_next = min(dists_to_others)
            two_step = d_cur + min_next
            sorted_dists = sorted(dists_to_others)
            regret = sorted_dists[1] - sorted_dists[0] if len(sorted_dists) >= 2 else 0.0
            avg_other = np.mean(dists_to_others)
        else:
            two_step = d_cur + d_dest
            regret = 0.0
            avg_other = 0.0

        d_cur_vals.append(d_cur)
        two_step_vals.append(two_step)
        regret_vals.append(regret)
        d_dest_vals.append(d_dest)
        avg_vals.append(avg_other)

    # Min-max normalization to [0,1]
    def normalize(vals):
        arr = np.array(vals, dtype=float)
        lo, hi = arr.min(), arr.max()
        if hi - lo < 1e-12:
            return np.zeros_like(arr)
        return (arr - lo) / (hi - lo)

    d_cur_norm = normalize(d_cur_vals)
    two_step_norm = normalize(two_step_vals)
    regret_norm = normalize(regret_vals)
    d_dest_norm = normalize(d_dest_vals)
    avg_norm = normalize(avg_vals)

    # Adaptive weights based on fraction of remaining nodes
    alpha = remaining / n  # 1 when all unvisited, -> 0 when few left
    w_immediate = 0.2 + 0.1 * (1 - alpha)          # slightly grows as we progress
    w_lookahead = 0.4 * alpha                      # strong early, weak late
    w_regret = 0.2 + 0.3 * (1 - alpha)              # stronger late (few options, regret more decisive)
    w_dest = 0.1 + 0.2 * (1 - alpha)                # destination more important near end
    w_avg = 0.1 * (1 - alpha)                       # average distance to others matters more when many nodes

    # Score: minimize immediate, two-step, destination distance, average distance; maximize regret -> subtract
    scores = (w_immediate * d_cur_norm +
              w_lookahead * two_step_norm -
              w_regret * regret_norm +
              w_dest * d_dest_norm +
              w_avg * avg_norm)

    best_idx = int(np.argmin(scores))
    return candidates[best_idx]
