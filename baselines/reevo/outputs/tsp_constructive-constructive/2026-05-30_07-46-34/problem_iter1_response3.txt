import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using a look-ahead heuristic that considers immediate cost,
    distance to destination, best next step from the candidate, and regret."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Weights for the composite score (tuned to balance factors)
    w1 = 0.35  # immediate travel cost
    w2 = 0.20  # distance to destination (encourages heading towards end)
    w3 = 0.30  # best next step from candidate (look-ahead)
    w4 = 0.15  # regret (difference between best and second best next step)
    
    scores = {}
    candidate_list = list(unvisited_nodes)
    
    for u in candidate_list:
        d_cur = distance_matrix[current_node][u]
        d_dest = distance_matrix[u][destination_node]
        
        # Remaining set after visiting u (excluding u)
        remaining = [v for v in candidate_list if v != u]
        
        if len(remaining) > 0:
            # Distances from u to all remaining unvisited nodes
            next_distances = [distance_matrix[u][v] for v in remaining]
            next_distances.sort()
            next_best = next_distances[0]
            # Regret: difference between second best and best (if available)
            if len(next_distances) > 1:
                regret = next_distances[1] - next_distances[0]
            else:
                regret = 0.0
        else:
            next_best = 0.0
            regret = 0.0
        
        # Composite score (lower is better)
        score = (w1 * d_cur + 
                 w2 * d_dest + 
                 w3 * next_best + 
                 w4 * regret)
        scores[u] = score
    
    # Return node with minimum score
    next_node = min(scores, key=scores.get)
    return next_node
