import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a 2-step look-ahead that balances immediate cost,
    future flexibility, and destination proximity."""
    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        # Distances from the candidate to other unvisited nodes
        others = [n for n in unvisited_nodes if n != node]
        if others:
            dists_to_others = np.array([distance_matrix[node][n] for n in others])
            min_next = np.min(dists_to_others)
            max_next = np.max(dists_to_others)
            avg_next = np.mean(dists_to_others)
            spread = max_next - min_next
        else:
            # Last node: must go to destination after this
            min_next = distance_matrix[node][destination_node]
            max_next = min_next
            avg_next = min_next
            spread = 0.0

        immediate = distance_matrix[current_node][node]
        to_dest = distance_matrix[node][destination_node]

        # Weights for components
        a, b, c, d, e = 0.35, 0.20, 0.10, 0.20, 0.15
        score = (a * immediate +
                 b * min_next +
                 c * spread +
                 d * avg_next +
                 e * to_dest)

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
