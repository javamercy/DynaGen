import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a look-ahead mechanism.
    For each candidate, the total tour cost is estimated as:
        edge(current, candidate) + greedy_TSP_cost(candidate, remaining, destination)
    The candidate with the smallest estimated cost is chosen.
    """
    def greedy_tour_cost(start: int, end: int, nodes: list) -> float:
        """Compute cost of a greedy nearest-neighbour tour from start through
        all nodes (in any order) and finally to end."""
        if not nodes:
            return distance_matrix[start][end]
        current = start
        total = 0.0
        unvisited = set(nodes)  # mutable copy
        while unvisited:
            next_node = min(unvisited, key=lambda n: distance_matrix[current][n])
            total += distance_matrix[current][next_node]
            current = next_node
            unvisited.remove(next_node)
        total += distance_matrix[current][end]
        return total

    best_node = None
    best_estimated_cost = float('inf')

    for candidate in unvisited_nodes:
        edge_cost = distance_matrix[current_node][candidate]
        remaining = [n for n in unvisited_nodes if n != candidate]
        # look-ahead: greedy tour from candidate through remaining to destination
        future_cost = greedy_tour_cost(candidate, destination_node, remaining)
        estimated_total = edge_cost + future_cost

        if estimated_total < best_estimated_cost:
            best_estimated_cost = estimated_total
            best_node = candidate

    return best_node
