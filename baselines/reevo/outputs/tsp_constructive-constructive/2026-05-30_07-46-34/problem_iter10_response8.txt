import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using two-step lookahead, regret, and adaptive weights based on remaining fraction."""
    if not unvisited_nodes:
        return current_node
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    num_unvisited = len(candidates)
    total_nodes = distance_matrix.shape[0]
    alpha = num_unvisited / total_nodes  # fraction remaining

    # Precompute lists
    immediate = []
    dest = []
    two_step = []
    regret = []

    for node in candidates:
        imm = distance_matrix[current_node, node]
        immediate.append(imm)

        dest.append(distance_matrix[node, destination_node])

        remaining = unvisited_nodes - {node}
        if remaining:
            dists = [distance_matrix[node, j] for j in remaining]
            two_step.append(imm + min(dists))
            if len(dists) >= 2:
                sorted_d = sorted(dists)
                regret.append(sorted_d[1] - sorted_d[0])
            else:
                regret.append(0.0)
        else:
            two_step.append(imm)
            regret.append(0.0)

    # Adaptive weights based on alpha (smooth transition)
    # Early exploration: emphasize look-ahead and regret; later: emphasize immediate and destination
    w_imm = 0.2 + 0.2 * (1 - alpha)  # increases as alpha decreases
    w_dest = 0.2 + 0.1 * (1 - alpha)
    w_two = 0.4 - 0.2 * (1 - alpha)  # decreases as alpha decreases
    w_reg = 0.2 - 0.1 * (1 - alpha)

    # Normalize each component to [0,1]
    def normalize(vals):
        arr = np.array(vals, dtype=float)
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    n_imm = normalize(immediate)
    n_dest = normalize(dest)
    n_two = normalize(two_step)
    n_reg = normalize(regret)

    # Combined score (lower is better)
    scores = w_imm * n_imm + w_dest * n_dest + w_two * n_two + w_reg * (1.0 - n_reg)

    # Tie-breaking: if multiple nodes have same score, choose the one with smallest immediate distance
    min_score = np.min(scores)
    best_indices = np.where(np.isclose(scores, min_score))[0]
    if len(best_indices) > 1:
        best = min(best_indices, key=lambda idx: immediate[idx])
    else:
        best = best_indices[0]

    return candidates[best]
