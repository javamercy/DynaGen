import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Improved constructive heuristic for TSP.
    - Normalizes distances per row.
    - Uses two-step lookahead and regret.
    - Sigmoid‑adapted weights based on remaining progress.
    
    Returns the next node to visit.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n_remaining = len(unvisited_nodes)
    total_nodes = distance_matrix.shape[0]
    # progress goes from 0 (many nodes left) to 1 (only destination left)
    progress = 1.0 - (n_remaining / total_nodes)

    # Sigmoid activation: steepest around progress=0.5
    sig = 1.0 / (1.0 + np.exp(-5.0 * (progress - 0.5)))

    # Weight adaptation (sum ≈ 1.0)
    w_imm = 0.5 - 0.3 * sig       # immediate cost weight
    w_look = 0.3 - 0.2 * sig      # two-step lookahead weight
    w_dest = 0.2 + 0.4 * sig      # destination proximity weight
    w_reg = 0.2 + 0.2 * sig       # regret weight (subtracted)

    best_node = None
    best_score = float("inf")

    current_row = distance_matrix[current_node, :]
    max_current = current_row.max()
    if max_current == 0:
        max_current = 1.0

    for node in unvisited_nodes:
        # ----- 1. Immediate cost (normalised by current node's row max) -----
        imm = distance_matrix[current_node, node] / max_current

        # ----- 2. Two‑step lookahead and regret -----
        remaining = unvisited_nodes - {node}
        if remaining:
            # distances from candidate to all remaining nodes
            cand_dists = [distance_matrix[node, u] for u in remaining]
            cand_dists.sort()
            min_future = cand_dists[0]
            if len(cand_dists) >= 2:
                regret = (cand_dists[1] - cand_dists[0])
            else:
                regret = 0.0

            # normalise regret by candidate's row max among remaining
            max_future = max(cand_dists)
            if max_future > 0:
                regret /= max_future
            else:
                regret = 0.0

            # two‑step lookahead: current edge + min onward, normalised
            # by sum of current row max and candidate's onward max
            look = (distance_matrix[current_node, node] + min_future) / \
                   (max_current + max_future)
        else:
            # only destination remains
            min_future = distance_matrix[node, destination_node]
            regret = 0.0
            look = (distance_matrix[current_node, node] + min_future) / \
                   (max_current + distance_matrix[node, destination_node])

        # ----- 3. Destination proximity (normalised by candidate's row max) -----
        node_max = distance_matrix[node, :].max()
        if node_max == 0:
            node_max = 1.0
        dest_cost = distance_matrix[node, destination_node] / node_max

        # ----- 4. Final score (minimise) -----
        score = (w_imm * imm +
                 w_look * look +
                 w_dest * dest_cost -
                 w_reg * regret)

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
