import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit from the unvisited nodes.
    Uses a weighted score of immediate distance, average and standard deviation of distances to other unvisited nodes,
    and distance to the final destination. Lower scores are preferred.
    """
    scores = {}
    # Coefficients for the heuristic
    c1, c2, c3, c4 = 0.4, 0.3, 0.2, 0.1

    # Precompute list representation for faster distance lookups
    unvisited_list = list(unvisited_nodes)

    for node in unvisited_list:
        # Immediate distance from current node to candidate
        immediate = distance_matrix[current_node, node]

        # Distances from candidate to other unvisited nodes (excluding itself)
        other_nodes = [n for n in unvisited_list if n != node]
        if other_nodes:
            distances_to_others = np.array([distance_matrix[node, n] for n in other_nodes])
            avg_dist = np.mean(distances_to_others)
            std_dist = np.std(distances_to_others)
        else:
            # Last node: no other unvisited nodes; set stats to zero
            avg_dist = 0.0
            std_dist = 0.0

        # Distance from candidate to final destination
        dest_dist = distance_matrix[node, destination_node]

        # Score: lower is better
        score = (c1 * immediate
                 - c2 * avg_dist
                 + c3 * std_dist
                 - c4 * dest_dist)

        scores[node] = score

    # Return the node with the lowest score
    return min(scores, key=scores.get)
