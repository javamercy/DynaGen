import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Improved constructive heuristic for TSP next‑node selection.

    Normalizes distances by the global maximum, uses linear weight decay
    based on the fraction of remaining nodes, and includes regret as a
    beneficial term. The composite score balances immediate cost, future
    connectivity, dispersion, and destination pressure.

    Parameters
    ----------
    current_node : int
        Index of the current node.
    destination_node : int
        Index of the final destination (starting node in TSP).
    unvisited_nodes : set
        Set of unvisited node indices.
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.

    Returns
    -------
    int
        Index of the next node to visit.
    """
    # Convert set to list for indexing
    unvisited = list(unvisited_nodes)
    n_unvisited = len(unvisited)
    if n_unvisited == 0:
        return destination_node
    if n_unvisited == 1:
        return unvisited[0]

    total_nodes = distance_matrix.shape[0]
    alpha = n_unvisited / total_nodes  # fraction of nodes still unvisited

    # Dynamic base weights (fine‑tuned for both exploration and exploitation)
    # All weights are positive; regret term carries a negative sign in the score.
    base_w_imm = 0.30   # immediate distance
    base_w_fut = 0.25   # average future distance (connectivity)
    base_w_reg = 0.20   # regret (urgency)
    base_w_std = 0.10   # standard deviation of future distances (diversity)
    base_w_dest = 0.15  # distance to final destination

    # Linear weight decay: exploration (high alpha) favours immediate and future terms,
    # exploitation (low alpha) gives more weight to destination.
    w_imm = base_w_imm * (0.6 + 0.4 * alpha)
    w_fut = base_w_fut * (0.6 + 0.4 * alpha)
    w_reg = base_w_reg * (0.6 + 0.4 * alpha)          # regret still benefits when many remain
    w_std = base_w_std * (0.5 + 0.5 * alpha)          # slight increase with alpha for diversity
    w_dest = base_w_dest * (1.0 - 0.5 * alpha)        # grows as fewer nodes remain

    # Global maximum distance for normalization (avoids min‑max instability)
    max_dist = np.max(distance_matrix)
    if max_dist < 1e-12:
        max_dist = 1.0

    # Pre‑compute distance from each candidate to the destination
    dest_dists = distance_matrix[unvisited, destination_node] / max_dist

    # Pre‑compute array of candidate indices for fast indexing
    cand_indices = np.array(unvisited, dtype=int)

    # Score for every candidate
    scores = np.empty(n_unvisited)
    for i, node in enumerate(unvisited):
        # Immediate distance from current node
        d_curr = distance_matrix[current_node, node] / max_dist

        # Distances from candidate to all other unvisited nodes
        other_mask = np.ones(n_unvisited, dtype=bool)
        other_mask[i] = False
        other_nodes = cand_indices[other_mask]

        if len(other_nodes) > 0:
            future_dists = distance_matrix[node, other_nodes] / max_dist
            avg_fut = future_dists.mean()
            std_fut = future_dists.std(ddof=0)
            # Regret = second smallest - smallest (urgency to take this candidate)
            sorted_dists = np.sort(future_dists)
            best = sorted_dists[0]
            second_best = sorted_dists[1] if len(sorted_dists) > 1 else best
            regret_fut = second_best - best
        else:
            avg_fut = 0.0
            std_fut = 0.0
            regret_fut = 0.0

        d_dest = dest_dists[i]

        # Composite score (minimisation)
        score = (w_imm * d_curr
                 + w_fut * avg_fut
                 + w_std * std_fut
                 - w_reg * regret_fut          # regret lowers the score (beneficial)
                 + w_dest * d_dest)
        scores[i] = score

    # Return node with minimal score
    return unvisited[int(np.argmin(scores))]
