```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using immediate and destination costs minus a weighted regret.
    Regret is the difference between the second-best and best distance from the candidate
    to other unvisited nodes. A small emphasis on regret (alpha > 1) can improve tour quality.
    """
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]

    best_score = float('inf')
    best_node = None
    alpha = 1.2  # slightly emphasize regret

    for node in unvisited:
        d_cur = distance_matrix[current_node][node]
        d_dest = distance_matrix[node][destination_node]

        # Compute regret: second-best minus best distance to other unvisited nodes
        others = [n for n in unvisited if n != node]
        if len(others) >= 2:
            dists = sorted([distance_matrix[node][n] for n in others])
            best = dists[0]
            second_best = dists[1]
            regret = second_best - best
        else:
            regret = 0.0

        score = d_cur + d_dest - alpha * regret

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
