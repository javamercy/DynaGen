import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node with a 2-step lookahead and regret mechanism."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    alpha = 0.3   # weight for regret
    beta = 0.1    # weight for destination proximity
    
    unvisited = list(unvisited_nodes)
    best_score = float('inf')
    best_node = None
    
    for u in unvisited:
        # immediate distance from current to u
        immediate = distance_matrix[current_node, u]
        
        # distances from u to the remaining unvisited (excluding u)
        remaining = [v for v in unvisited if v != u]
        if len(remaining) == 0:
            # only one node left (shouldn't happen because len>1)
            score = immediate
        else:
            dists = distance_matrix[u, remaining]
            # find two smallest distances
            sorted_idx = np.argsort(dists)
            best_dist = dists[sorted_idx[0]]
            second_best_dist = dists[sorted_idx[1]] if len(sorted_idx) > 1 else best_dist
            regret = second_best_dist - best_dist
        
            # distance to destination
            dest_dist = distance_matrix[destination_node, u]
        
            # composite score (minimize)
            score = immediate + best_dist - alpha * regret - beta * dest_dist
        
        if score < best_score:
            best_score = score
            best_node = u
    
    return best_node
