import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Two-step lookahead heuristic for TSP path construction.
    For each candidate, computes cost of current->candidate + best next edge from candidate,
    plus a small term to encourage proximity to the destination.
    """
    best_score = float('inf')
    best_node = None
    # Precompute distances from current node and to destination for all candidates
    for candidate in unvisited_nodes:
        # Immediate cost
        d_curr_cand = distance_matrix[current_node][candidate]
        # Lookahead: best next step from candidate (excluding candidate itself)
        d_cand_next = min(
            distance_matrix[candidate][other]
            for other in unvisited_nodes
            if other != candidate
        ) if len(unvisited_nodes) > 1 else 0.0  # if last node, no next step cost
        # Proximity to destination (small weight)
        d_cand_dest = distance_matrix[candidate][destination_node]
        # Total score: immediate + lookahead + 0.5 * destination distance
        score = d_curr_cand + d_cand_next + 0.5 * d_cand_dest
        if score < best_score:
            best_score = score
            best_node = candidate
    return best_node
