```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Two-step look-ahead heuristic with regret and dynamic weighting.

    For each candidate u in unvisited_nodes:
        - direct = distance from current to u
        - Among remaining unvisited (excluding u), compute:
            - best_future = smallest distance from u to any other unvisited node
            - second_best = second smallest (or same if only one)
            - regret = second_best - best_future   (positive or zero)
        - final_penalty = distance from u to destination, scaled by remaining count
        - dynamic weight alpha = 0.5 * (1 - exp(-len(unvisited)/10))  # grows with more nodes
        - score = direct + best_future - alpha * regret + 0.1 * final_penalty

    The node with the smallest score is selected.
    """
    n_remaining = len(unvisited_nodes)
    best_score = float('inf')
    best_node = None

    # Dynamic weight for regret: higher when more nodes remain
    alpha = 0.5 * (1.0 - np.exp(-n_remaining / 10.0))

    for u in unvisited_nodes:
        direct = distance_matrix[current_node, u]

        # Remaining nodes except u
        others = [v for v in unvisited_nodes if v != u]
        if others:
            dists = [distance_matrix[u, v] for v in others]
            dists.sort()
            best_future = dists[0]
            second_best = dists[1] if len(dists) >= 2 else dists[0]
            regret = second_best - best_future  # non-negative
        else:
            # Only one node left: after u, go directly to destination
            best_future = distance_matrix[u, destination_node]
            regret = 0.0

        # Mild penalty for distance to final destination (scaled by remaining nodes)
        final_penalty = distance_matrix[u, destination_node] / (n_remaining + 1.0)

        score = direct + best_future - alpha * regret + 0.1 * final_penalty

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
```
