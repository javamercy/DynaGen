```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a weighted combination of the distance from
    the current node and the distance to the destination. The weight on the
    distance to the destination increases as fewer nodes remain unvisited.
    """
    if not unvisited_nodes:
        return current_node

    # Total number of nodes in the problem
    total_nodes = distance_matrix.shape[0]
    remaining_frac = len(unvisited_nodes) / total_nodes

    # Dynamic weight for the destination term: emphasize more towards the end
    alpha = max(0.1, 1.0 - remaining_frac)  # ranges from 0.1 to 1.0

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        score = distance_matrix[current_node, node] + alpha * distance_matrix[node, destination_node]
        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
