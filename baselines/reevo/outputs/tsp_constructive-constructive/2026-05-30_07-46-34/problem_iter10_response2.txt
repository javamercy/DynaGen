import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select the next node using a simple, adaptive regret‑based heuristic.

    The score for each candidate is:
        score = d(current, candidate) + d(candidate, destination) - alpha * regret

    where regret = second_best_distance - best_distance from the candidate
    to all *other* unvisited nodes. The weight `alpha` increases with the
    fraction of unvisited nodes, encouraging the algorithm to visit nodes
    with high regret early (when many options remain) and to focus more on
    immediate and destination distances later.

    Parameters
    ----------
    current_node : int
        Index of the current node.
    destination_node : int
        Index of the final destination (starting node in TSP).
    unvisited_nodes : set
        Set of unvisited node indices.
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.

    Returns
    -------
    int
        Index of the next node to visit.
    """
    unvisited = list(unvisited_nodes)
    n = len(unvisited)

    # Trivial cases
    if n == 0:
        return destination_node
    if n == 1:
        return unvisited[0]

    total_nodes = distance_matrix.shape[0]
    # Regret weight: higher when many nodes remain
    alpha = 1.0 + 0.5 * (n / total_nodes)

    best_score = float("inf")
    best_node = unvisited[0]

    for node in unvisited:
        d_cur = distance_matrix[current_node, node]
        d_dest = distance_matrix[node, destination_node]

        # Compute regret (difference between second‑best and best distance
        # to other unvisited nodes)
        others = [v for v in unvisited if v != node]
        if len(others) >= 2:
            dists = sorted(distance_matrix[node, other] for other in others)
            regret = dists[1] - dists[0]
        elif len(others) == 1:
            regret = distance_matrix[node, others[0]]  # only one alternative, regret = that distance
        else:
            regret = 0.0

        score = d_cur + d_dest - alpha * regret

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
