import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using adaptive multi‑criteria scoring with look‑ahead regret,
    statistical dispersion measures, and dynamic weights.
    """
    unvisited = list(unvisited_nodes)
    n_unvisited = len(unvisited)
    if n_unvisited == 0:
        return destination_node
    if n_unvisited == 1:
        return unvisited[0]

    scores = {}
    # Dynamic weight adjustment based on problem phase (remaining nodes)
    w_immediate = 0.3
    w_avg_future = 0.25
    w_regret = 0.2
    w_std = 0.1
    w_dest = 0.15

    # Scale weights to give more importance to destination when few nodes remain
    if n_unvisited <= 5:
        w_dest += 0.1
        w_avg_future -= 0.05
        w_immediate -= 0.05
    elif n_unvisited > 10:
        w_regret += 0.05
        w_avg_future += 0.05
        w_immediate -= 0.05
        w_dest -= 0.05

    # Precompute distances to speed up multiple candidate evaluations
    dist_to_dest = {n: distance_matrix[n, destination_node] for n in unvisited}

    for node in unvisited:
        d_curr = distance_matrix[current_node, node]

        # Distances from candidate to all other unvisited nodes
        other = [n for n in unvisited if n != node]
        if other:
            other_dists = [distance_matrix[node, n] for n in other]
            avg_future = np.mean(other_dists)
            std_future = np.std(other_dists) if len(other_dists) > 1 else 0.0
            # Look‑ahead regret: difference between best and second best continuation
            sorted_dists = sorted(other_dists)
            best = sorted_dists[0]
            second_best = sorted_dists[1] if len(sorted_dists) >= 2 else best
            future_regret = second_best - best
        else:
            avg_future = 0.0
            std_future = 0.0
            future_regret = 0.0

        d_dest = dist_to_dest[node]

        # Scoring: minimize weighted sum. Negative coefficients encourage larger values.
        score = (w_immediate * d_curr
                 - w_avg_future * avg_future
                 - w_regret * future_regret
                 + w_std * std_future
                 - w_dest * d_dest)

        scores[node] = score

    return min(scores, key=scores.get)
