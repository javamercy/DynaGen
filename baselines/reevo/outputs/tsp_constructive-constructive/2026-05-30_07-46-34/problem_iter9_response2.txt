import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using adaptive two-step lookahead and regret."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Precompute values for each candidate
    candidates = list(unvisited_nodes)
    imm = np.array([distance_matrix[current_node, i] for i in candidates])
    to_dest = np.array([distance_matrix[i, destination_node] for i in candidates])
    # Two-step lookahead: best next after candidate
    best_next = np.array([
        min(distance_matrix[i, j] for j in unvisited_nodes if j != i)
        if len(unvisited_nodes) > 1 else 0.0
        for i in candidates
    ])
    combined = imm + best_next

    # Normalize each component to [0,1] across candidates
    norm_imm = (imm - imm.min()) / (imm.max() - imm.min() + 1e-10)
    norm_dest = (to_dest - to_dest.min()) / (to_dest.max() - to_dest.min() + 1e-10)
    norm_combined = (combined - combined.min()) / (combined.max() - combined.min() + 1e-10)
    regret = norm_combined  # high regret -> poor two-step choice

    # Adaptive weights based on number of remaining nodes
    remaining = len(unvisited_nodes)
    if remaining > 10:
        w_imm, w_dest, w_reg = 0.3, 0.1, 0.6
    elif remaining > 5:
        w_imm, w_dest, w_reg = 0.2, 0.3, 0.5
    else:
        w_imm, w_dest, w_reg = 0.2, 0.6, 0.2

    # Score: minimize (immediate + avoid high regret + approach destination)
    scores = w_imm * norm_imm + w_dest * (1 - norm_dest) + w_reg * regret
    next_node = candidates[np.argmin(scores)]
    return next_node
