import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using:
    - normalized immediate cost
    - two-step lookahead (greedy one-step future cost)
    - regret (difference between best and second best future distances)
    - destination proximity
    - tuned weights for balance and diversification
    """
    if len(unvisited_nodes) == 0:
        raise ValueError("No unvisited nodes")
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    n_candidates = len(candidates)

    # Precompute distances from each candidate to all unvisited (including self, but we'll exclude self later)
    dists_from_cand = np.array([[distance_matrix[c][u] for u in unvisited_nodes] for c in candidates])  # shape (n_cand, n_unvisited)

    # 1. Immediate cost: distance from current node to candidate
    immediate = np.array([distance_matrix[current_node][c] for c in candidates])

    # 2. Future lookahead (one-step greedy): for each candidate, compute the best next node among remaining unvisited (excluding candidate itself)
    #    The cost to that best next node is the lookahead cost.
    lookahead = np.zeros(n_candidates)
    for i, c in enumerate(candidates):
        # distances from candidate c to all other unvisited nodes (excluding c itself)
        others = [u for u in unvisited_nodes if u != c]
        if others:
            best = min(distance_matrix[c][u] for u in others)
            lookahead[i] = best
        else:
            lookahead[i] = 0  # no future nodes left

    # 3. Regret: for each candidate, difference between best and second-best future distance from candidate to remaining unvisited
    regret = np.zeros(n_candidates)
    for i, c in enumerate(candidates):
        others = [distance_matrix[c][u] for u in unvisited_nodes if u != c]
        if len(others) >= 2:
            sorted_others = np.sort(others)
            regret[i] = sorted_others[1] - sorted_others[0]
        else:
            regret[i] = 0

    # 4. Destination closeness: distance from candidate to destination_node (smaller is better)
    dest_dist = np.array([distance_matrix[c][destination_node] for c in candidates])

    # 5. Summary statistics (mean and std) of distances from candidate to all unvisited (including candidate? but we exclude candidate)
    mean_dist = np.array([np.mean([distance_matrix[c][u] for u in unvisited_nodes if u != c]) for c in candidates])
    std_dist = np.array([np.std([distance_matrix[c][u] for u in unvisited_nodes if u != c]) for c in candidates])

    # Normalize each component to [0,1] (min-max across candidates)
    def normalize(arr):
        min_val = np.min(arr)
        max_val = np.max(arr)
        if max_val - min_val == 0:
            return np.zeros_like(arr)
        return (arr - min_val) / (max_val - min_val)

    imm_norm = normalize(immediate)
    look_norm = normalize(lookahead)
    regret_norm = normalize(regret)
    dest_norm = normalize(dest_dist)
    mean_norm = normalize(mean_dist)
    std_norm = normalize(std_dist)

    # Weights: tuned for balancing immediate, future, regret, destination, and diversification (std)
    w1 = 0.3  # immediate cost (want small)
    w2 = 0.25 # lookahead cost (small)
    w3 = 0.2  # regret (large regret is good, means current choice is clearly better)
    w4 = 0.15 # destination proximity (small is good)
    w5 = 0.05 # mean distance to unvisited (small is good, future flexibility)
    w6 = 0.05 # std distance (larger std -> more diversity, small weight)

    # Score: we want to minimize weighted sum of costs that should be small,
    # but regret and std are positive attributes (larger better), so we negate them.
    scores = (w1 * imm_norm
              + w2 * look_norm
              - w3 * regret_norm   # negative because larger regret is better
              + w4 * dest_norm
              + w5 * mean_norm
              - w6 * std_norm)     # negative because larger std adds diversity

    best_idx = np.argmin(scores)
    return candidates[best_idx]
