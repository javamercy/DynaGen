import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using adaptive weights, regret, and two-step lookahead."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    total_nodes = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    # progress: fraction of unvisited nodes remaining (0 = none left, 1 = all unvisited)
    progress = remaining / (total_nodes - 1)

    # Adaptive weights – tuned heuristically
    w_look = 0.3 + 0.2 * (1 - progress)      # lookahead important early
    w_dest = 0.1 + 0.4 * progress             # destination important late
    w_regret = 0.1 + 0.3 * (1 - progress)     # regret critical early
    w_avg = 0.1 + 0.2 * progress              # hubness matters late

    # Normalise weights to sum to 1
    total = w_look + w_dest + w_regret + w_avg
    w_look /= total
    w_dest /= total
    w_regret /= total
    w_avg /= total

    nodes = list(unvisited_nodes)
    n = len(nodes)
    two_step = [0.0] * n
    d_dest = [0.0] * n
    regret = [0.0] * n
    avg = [0.0] * n

    for i, node in enumerate(nodes):
        d_curr = distance_matrix[current_node, node]
        d_dest[i] = distance_matrix[node, destination_node]

        # Two‑step lookahead: current + best next step (to another unvisited or destination)
        next_dists = [distance_matrix[node, j] for j in unvisited_nodes if j != node]
        if destination_node not in unvisited_nodes:
            next_dists.append(distance_matrix[node, destination_node])
        min_next = min(next_dists) if next_dists else 0.0
        two_step[i] = d_curr + min_next

        # Regret: second best distance from current to other unvisited minus d_curr (≥0)
        other_dists = [distance_matrix[current_node, k] for k in unvisited_nodes if k != node]
        if len(other_dists) >= 2:
            sorted_dists = sorted(other_dists)
            second_best = sorted_dists[1]
            regret[i] = max(0.0, second_best - d_curr)
        else:
            regret[i] = 0.0

        # Average distance to other unvisited nodes (hubness)
        dists_to_others = [distance_matrix[node, j] for j in unvisited_nodes if j != node]
        avg[i] = np.mean(dists_to_others) if dists_to_others else 0.0

    # Min‑max normalisation to [0,1] (0 = best, 1 = worst) for all metrics
    def normalize(values):
        mn = min(values)
        mx = max(values)
        if mx == mn:
            return [0.5] * len(values)   # neutral value
        return [(v - mn) / (mx - mn) for v in values]

    norm_two_step = normalize(two_step)   # smaller is better
    norm_d_dest   = normalize(d_dest)     # smaller is better
    norm_regret   = normalize(regret)     # larger is better (will be subtracted)
    norm_avg      = normalize(avg)        # smaller is better

    best_node = None
    best_score = float('inf')
    for i, node in enumerate(nodes):
        score = (w_look * norm_two_step[i] +
                 w_dest * norm_d_dest[i] -
                 w_regret * norm_regret[i] +
                 w_avg * norm_avg[i])
        if score < best_score:
            best_score = score
            best_node = node

    return best_node
