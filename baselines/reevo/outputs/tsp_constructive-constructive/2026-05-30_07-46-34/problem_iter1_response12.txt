import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a two-step look-ahead heuristic.
    
    For each candidate node, the score is the minimum cost of a path
    current -> candidate -> next -> destination, where 'next' is a different
    unvisited node. If only one candidate remains, the score reduces to
    current -> candidate -> destination.
    The candidate with the smallest score is chosen.
    """
    best_node = None
    best_score = float('inf')
    
    for n in unvisited_nodes:
        # Compute the cost of going from current to n
        d_cur_n = distance_matrix[current_node][n]
        
        # If only one node left, the path is current->n->destination
        if len(unvisited_nodes) == 1:
            total = d_cur_n + distance_matrix[n][destination_node]
        else:
            # Two-step look-ahead: find the best next node k (different from n)
            # that minimises d_cur_n + d(n,k) + d(k, destination)
            min_two_step = float('inf')
            for k in unvisited_nodes:
                if k == n:
                    continue
                cost = d_cur_n + distance_matrix[n][k] + distance_matrix[k][destination_node]
                if cost < min_two_step:
                    min_two_step = cost
            total = min_two_step
        
        if total < best_score:
            best_score = total
            best_node = n
    
    return best_node
