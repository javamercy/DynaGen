import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved constructive heuristic for TSP.
    Combines additive criteria, regret, summary statistics, two-step lookahead,
    and normalized distances with tuned weights.
    """
    # Handle trivial case
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Normalization factor
    max_dist = np.max(distance_matrix)
    # Weights (tuned for diversification and return-path preservation)
    w_imm = 0.3
    w_fut = 0.2
    w_dest = 0.2
    w_reg = 0.15
    w_look = 0.1
    w_std = 0.05

    candidates = list(unvisited_nodes)
    scores = {}

    for node in candidates:
        # --- Immediate cost ---
        imm = distance_matrix[current_node, node] / max_dist

        # --- Future average distance to other unvisited nodes ---
        others = [j for j in unvisited_nodes if j != node]
        if others:
            fut = np.mean([distance_matrix[node, j] for j in others]) / max_dist
        else:
            fut = 0.0

        # --- Standard deviation of distances to other unvisited nodes ---
        if len(others) >= 2:
            std = np.std([distance_matrix[node, j] for j in others]) / max_dist
        else:
            std = 0.0

        # --- Destination distance ---
        dest = distance_matrix[node, destination_node] / max_dist

        # --- Regret (difference between second-best and best outgoing edge to others) ---
        if len(others) >= 2:
            dst = [distance_matrix[node, j] for j in others]
            dst_sorted = sorted(dst)
            best = dst_sorted[0]
            second_best = dst_sorted[1]
            regret = (second_best - best) / max_dist
        else:
            regret = 0.0

        # --- Two-step lookahead: imm + best next step ---
        if others:
            best_next = min([distance_matrix[node, j] for j in others]) / max_dist
            lookahead = imm + best_next
        else:
            lookahead = imm  # if no others, two-step degenerates

        # Composite score (minimization)
        score = (w_imm * imm +
                 w_fut * fut +
                 w_dest * dest -
                 w_reg * regret +
                 w_look * lookahead +
                 w_std * std)

        scores[node] = score

    # Return candidate with minimal score
    return min(scores, key=scores.get)
