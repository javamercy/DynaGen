```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Select the next node using a simple additive score with a regret term.

    The regret term is the difference between the second-best and best distance
    from the candidate to other unvisited nodes. A high regret indicates that
    postponing the node would likely increase cost, so we subtract regret from
    the immediate + destination cost. The node with the smallest score is chosen.
    """
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]

    best_score = float("inf")
    best_node = None

    for node in unvisited:
        d_cur = distance_matrix[current_node][node]
        d_dest = distance_matrix[node][destination_node]

        # Distances from candidate to other unvisited nodes
        others = [n for n in unvisited if n != node]
        if len(others) == 0:
            # Only one node left (handled above), but keep safe
            regret = 0.0
        else:
            dists = sorted(distance_matrix[node, n] for n in others)
            best = dists[0]
            if len(dists) >= 2:
                second_best = dists[1]
                regret = second_best - best
            else:
                regret = 0.0

        score = d_cur + d_dest - regret

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
