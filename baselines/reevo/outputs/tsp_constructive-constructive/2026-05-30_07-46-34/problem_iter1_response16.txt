```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead mechanism.
    For each candidate, it simulates a greedy nearest-neighbor tour from that candidate
    over the remaining unvisited nodes and returns to the destination.
    The candidate that minimizes the total estimated tour cost is chosen.
    """
    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # Cost from current node to candidate
        total = distance_matrix[current_node][candidate]
        
        # Simulate greedy nearest-neighbor from candidate
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        current = candidate
        
        while remaining:
            # Find nearest neighbor
            nearest = min(remaining, key=lambda n: distance_matrix[current][n])
            total += distance_matrix[current][nearest]
            current = nearest
            remaining.remove(nearest)
        
        # Return to destination node
        total += distance_matrix[current][destination_node]
        
        if total < best_total:
            best_total = total
            best_node = candidate
    
    return best_node
```
