```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """Improved version: normalize distances, balance immediate cost, regret,
    and destination proximity, with adaptive weights based on problem size."""

    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    n = len(unvisited_nodes)
    total_nodes = distance_matrix.shape[0]

    # Normalization factor
    max_dist = distance_matrix.max()
    if max_dist == 0:
        max_dist = 1

    # Adaptive weights: early stages prioritize immediate cost and regret,
    # later stages emphasize destination proximity.
    progress = 1.0 - (n / total_nodes)  # 0 at start, near 1 at end
    w_imm = 0.5 - 0.2 * progress        # immediate cost weight (0.5 -> 0.3)
    w_reg = 0.3 + 0.1 * progress        # regret weight (0.3 -> 0.4)
    w_dest = 0.2 + 0.2 * progress       # destination weight (0.2 -> 0.4)

    best_node = None
    best_score = float("inf")

    for node in unvisited_nodes:
        # Immediate cost
        imm = distance_matrix[current_node][node] / max_dist

        # Regret: difference between best and second best from candidate to remaining
        remaining = unvisited_nodes - {node}
        if remaining:
            dists = [distance_matrix[node][u] for u in remaining]
            if len(dists) >= 2:
                sorted_dists = sorted(dists)
                regret = (sorted_dists[1] - sorted_dists[0]) / max_dist
            else:
                regret = 0.0
        else:
            regret = 0.0

        # Destination proximity
        dest_cost = distance_matrix[node][destination_node] / max_dist

        # Score: lower is better
        score = w_imm * imm + w_dest * dest_cost - w_reg * regret

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
