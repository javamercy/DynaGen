import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using adaptive multi‑criteria scoring with
    per‑criterion min‑max normalization and dynamic weights that depend on
    the fraction of remaining nodes.

    The score for each candidate integrates:
      - immediate distance from the current node (penalty)
      - average distance to all other unvisited nodes (penalty, future connectivity)
      - future regret: difference between second‑best and best distance from
        the candidate to any other unvisited node (benefit, high regret lowers score)
      - distance to the destination node (penalty)

    All terms are normalised to [0, 1] over the current candidate set so that
    weights are comparable. Weights shift from exploration (emphasising immediate
    cost, future connectivity and regret) to exploitation (emphasising the
    destination) as the number of unvisited nodes shrinks.

    Parameters
    ----------
    current_node : int
        Index of the current node.
    destination_node : int
        Index of the final destination (starting node in TSP).
    unvisited_nodes : set
        Set of unvisited node indices.
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.

    Returns
    -------
    int
        Index of the next node to visit.
    """
    unvisited = list(unvisited_nodes)
    n = len(unvisited)

    # Trivial cases
    if n == 0:
        return destination_node
    if n == 1:
        return unvisited[0]

    # --------------------------------------------------
    # 1. Compute raw criteria for each candidate
    # --------------------------------------------------
    d_curr = np.array([distance_matrix[current_node, u] for u in unvisited])
    d_dest = np.array([distance_matrix[u, destination_node] for u in unvisited])

    avg_fut = np.zeros(n)
    regret_fut = np.zeros(n)
    for i, u in enumerate(unvisited):
        other = [v for j, v in enumerate(unvisited) if j != i]
        if len(other) >= 2:
            dists = np.sort(distance_matrix[u, other])
            avg_fut[i] = np.mean(dists)
            regret_fut[i] = dists[1] - dists[0]
        elif len(other) == 1:
            avg_fut[i] = distance_matrix[u, other[0]]
            regret_fut[i] = 0.0
        else:
            avg_fut[i] = 0.0
            regret_fut[i] = 0.0

    # --------------------------------------------------
    # 2. Min‑max normalisation per criterion
    # --------------------------------------------------
    def minmax(arr):
        mn, mx = arr.min(), arr.max()
        if mx == mn:
            return np.zeros_like(arr)
        return (arr - mn) / (mx - mn)

    d_curr_norm = minmax(d_curr)
    avg_fut_norm = minmax(avg_fut)
    regret_fut_norm = minmax(regret_fut)
    d_dest_norm = minmax(d_dest)

    # --------------------------------------------------
    # 3. Adaptive weights based on remaining fraction
    # --------------------------------------------------
    total_nodes = distance_matrix.shape[0]
    alpha = n / total_nodes                      # 1 when all nodes remain, 0 when none

    # Base weights (tuned for balance)
    w_imm = 0.35 * (0.5 + 0.5 * alpha)          # higher when many nodes remain
    w_fut = 0.25 * (0.5 + 0.5 * alpha)          # higher when many nodes remain
    w_reg = 0.20 * (0.5 + 0.5 * alpha)          # higher when many nodes remain
    w_dest = 0.10 * (1.0 - 0.5 * alpha)         # higher when few nodes remain

    # --------------------------------------------------
    # 4. Composite score (minimisation)
    # --------------------------------------------------
    scores = (w_imm * d_curr_norm
              + w_fut * avg_fut_norm
              - w_reg * regret_fut_norm          # high regret → lower score
              + w_dest * d_dest_norm)

    best_index = int(np.argmin(scores))
    return unvisited[best_index]
