import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node with adaptive multi-criteria scoring including regret and diversification."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    total_nodes = distance_matrix.shape[0]
    n_remaining = len(unvisited_nodes)
    max_unvisited = total_nodes - 1  # current_node is already visited
    progress = 1.0 - n_remaining / max_unvisited if max_unvisited > 0 else 1.0

    # Adaptive weights
    base_w = 0.5 * (0.5 + 0.5 * progress)           # more important near end
    look_w = 0.3 * (1.0 - progress)                  # short-term future, important early
    regret_w = 0.4 * progress                        # urgency matters when few remain
    div_w = 0.3 * (1.0 - progress)                   # global coverage, important early
    dest_w = 0.1                                     # always a slight pull to destination

    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        remaining = unvisited_nodes - {candidate}
        base_cost = distance_matrix[current_node][candidate]

        # Look-ahead: distance to nearest unvisited neighbour
        if remaining:
            dists_to_remaining = [distance_matrix[candidate][u] for u in remaining]
            nearest_dist = min(dists_to_remaining)
            # Regret: difference between second best and best
            if len(remaining) >= 2:
                sorted_dists = sorted(dists_to_remaining)
                second_nearest = sorted_dists[1]
                regret = max(second_nearest - nearest_dist, 0.0)
            else:
                regret = 0.0
            # Diversification: average distance to all remaining (higher => more isolated)
            avg_dist = np.mean(dists_to_remaining)
        else:
            nearest_dist = 0.0
            regret = 0.0
            avg_dist = 0.0

        # Destination attraction (negative to favour closeness)
        dest_attraction = -dest_w * distance_matrix[candidate][destination_node]

        score = (base_w * base_cost +
                 look_w * nearest_dist +
                 regret_w * (-regret) +               # negative: urgent nodes get lower score
                 div_w * (-avg_dist) +                # negative: isolated nodes get lower score
                 dest_attraction)

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
