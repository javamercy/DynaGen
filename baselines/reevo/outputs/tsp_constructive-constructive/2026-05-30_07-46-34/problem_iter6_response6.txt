import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """Adaptive multi-criteria selection for TSP constructive heuristic."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    total_nodes = distance_matrix.shape[0]
    n_remaining = len(unvisited_nodes)
    max_unvisited = total_nodes - 1  # current node already visited
    # progress: 0 when all nodes remain, 1 when only one left
    progress = 1.0 - n_remaining / max_unvisited if max_unvisited > 0 else 1.0

    # Adaptive weights: immediate cost is prioritized early; lookahead and regret later
    w_imm = 0.5 * (1.0 - progress)       # decreases as we progress
    w_look = 0.4 * progress              # increases later
    w_regret = 0.3 * progress            # increases later
    w_avg = 0.2 * (1.0 - progress)       # smaller average distance is better, more important early
    w_dest = 0.1                         # destination proximity, constant small weight
    w_std = 0.1 * (1.0 - progress)       # diversity reward, more important early

    candidates = list(unvisited_nodes)
    imm_vals = []
    look_vals = []
    regret_vals = []
    avg_vals = []
    dest_vals = []
    std_vals = []

    for cand in candidates:
        remaining = unvisited_nodes - {cand}
        imm_vals.append(distance_matrix[current_node, cand])
        dest_vals.append(distance_matrix[cand, destination_node])

        if remaining:
            dists = [distance_matrix[cand, u] for u in remaining]
            look_vals.append(min(dists))
            sorted_dists = sorted(dists)
            if len(remaining) >= 2:
                regret_vals.append(sorted_dists[1] - sorted_dists[0])
            else:
                regret_vals.append(0.0)
            avg_vals.append(np.mean(dists))
            std_vals.append(np.std(dists))
        else:
            # Only candidate left; after visiting, go directly to destination
            look_vals.append(distance_matrix[cand, destination_node])
            regret_vals.append(0.0)
            avg_vals.append(0.0)
            std_vals.append(0.0)

    eps = 1e-9
    def normalize(vals):
        mn = min(vals)
        mx = max(vals)
        if mx - mn < eps:
            return [0.0 for _ in vals]
        return [(v - mn) / (mx - mn) for v in vals]

    imm_norm = normalize(imm_vals)
    look_norm = normalize(look_vals)
    regret_norm = normalize(regret_vals)
    avg_norm = normalize(avg_vals)
    dest_norm = normalize(dest_vals)
    std_norm = normalize(std_vals)

    best_score = float('inf')
    best_node = None
    for i, cand in enumerate(candidates):
        # Lower composite score is better
        score = (w_imm * imm_norm[i] +
                 w_look * look_norm[i] -
                 w_regret * regret_norm[i] +   # larger regret reduces score
                 w_avg * avg_norm[i] +          # larger average increases score (bad)
                 w_dest * dest_norm[i] -        # larger diversity reduces score (good)
                 w_std * std_norm[i])           # larger std reduces score (good)
        if score < best_score:
            best_score = score
            best_node = cand

    return best_node
