import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using adaptive weights and two-step lookahead."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    total_nodes = distance_matrix.shape[0]
    remaining = len(unvisited_nodes)
    # progress: fraction of unvisited nodes remaining (0 if all visited, 1 if none visited yet)
    progress = remaining / (total_nodes - 1)

    # Adaptive weights (with small constant to avoid zeros)
    w_look   = 0.4 * progress + 0.05
    w_dest   = 0.2 * (1 - progress) + 0.05
    w_regret = 0.25 * (1 - progress) + 0.05
    w_avg    = 0.15 * progress + 0.05

    # Precompute candidate lists
    nodes = list(unvisited_nodes)
    two_step_vals = []
    d_dest_vals = []
    regret_vals = []
    avg_vals = []

    for node in nodes:
        d_curr = distance_matrix[current_node, node]
        d_dest = distance_matrix[node, destination_node]
        d_dest_vals.append(d_dest)

        # Two-step: current distance + min next step (including destination)
        next_candidates = [distance_matrix[node, j] for j in unvisited_nodes if j != node]
        if destination_node not in unvisited_nodes:
            next_candidates.append(distance_matrix[node, destination_node])
        min_next = min(next_candidates) if next_candidates else 0
        two_step = d_curr + min_next
        two_step_vals.append(two_step)

        # Regret: difference between best alternative distance from current and d_curr
        alt_distances = [distance_matrix[current_node, k] for k in unvisited_nodes if k != node]
        if alt_distances:
            best_alt = min(alt_distances)
            regret = best_alt - d_curr
        else:
            regret = 0  # only one node left (handled earlier)
        regret_vals.append(regret)

        # Average distance to other unvisited nodes (hubness)
        dists_to_others = [distance_matrix[node, j] for j in unvisited_nodes if j != node]
        avg = np.mean(dists_to_others) if dists_to_others else 0
        avg_vals.append(avg)

    # Normalize each metric to [0,1] across candidates
    def normalize(values):
        min_v = min(values)
        max_v = max(values)
        if max_v > min_v:
            return [(v - min_v) / (max_v - min_v) for v in values]
        else:
            return [0.0] * len(values)

    norm_two_step = normalize(two_step_vals)
    norm_d_dest   = normalize(d_dest_vals)
    norm_regret   = normalize(regret_vals)
    norm_avg      = normalize(avg_vals)

    # Compute combined score (minimization)
    best_node = None
    best_score = float('inf')
    for i, node in enumerate(nodes):
        score = (w_look * norm_two_step[i] +
                 w_dest * norm_d_dest[i] -
                 w_regret * norm_regret[i] +
                 w_avg * norm_avg[i])
        if score < best_score:
            best_score = score
            best_node = node

    return best_node
