```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 3-step look‑ahead heuristic.
    For each candidate, it greedily builds a small path of up to three edges
    (including the edge from current_node to the candidate) and uses the total
    cost of that path as the score. The candidate with the smallest total cost
    is chosen.
    """
    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        # 1st step: current -> candidate
        cost = distance_matrix[current_node, candidate]
        remaining1 = unvisited_nodes - {candidate}

        if not remaining1:
            # no more nodes -> go directly to destination after candidate
            cost += distance_matrix[candidate, destination_node]
        else:
            # 2nd step: best next node from candidate (nearest neighbour)
            next2 = min(remaining1, key=lambda n: distance_matrix[candidate, n])
            cost += distance_matrix[candidate, next2]
            remaining2 = remaining1 - {next2}

            if not remaining2:
                # only one node after candidate -> then go to destination from next2
                cost += distance_matrix[next2, destination_node]
            else:
                # 3rd step: best next node from next2
                next3 = min(remaining2, key=lambda n: distance_matrix[next2, n])
                cost += distance_matrix[next2, next3]
                # after three steps, finally add distance to destination
                cost += distance_matrix[next3, destination_node]

        if cost < best_score:
            best_score = cost
            best_node = candidate

    return best_node
```
