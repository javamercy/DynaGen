```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a composite heuristic that balances
    immediate travel cost, future connectivity, and return preparation.

    For each candidate node the score is:
        total = w1 * immediate_cost
              + w2 * (average distance to other unvisited nodes)
              + w3 * (distance to the nearest other unvisited node)
              + w4 * (distance to destination)

    where w > 0 means a penalty. The node with the smallest total is chosen.
    """
    w1, w2, w3, w4 = 0.5, 0.3, 0.2, 0.1

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        # Immediate cost from current to candidate
        immediate = distance_matrix[current_node, node]

        # Other unvisited nodes (excluding candidate itself)
        others = list(unvisited_nodes - {node})
        n_others = len(others)

        if n_others > 0:
            # Average distance from candidate to other unvisited nodes
            avg_dist = np.mean([distance_matrix[node, n] for n in others])
            # Distance to the nearest other unvisited node
            nearest_dist = np.min([distance_matrix[node, n] for n in others])
        else:
            # Last node: must return to destination
            avg_dist = 0.0
            nearest_dist = distance_matrix[node, destination_node]

        # Distance to final destination
        dest_dist = distance_matrix[node, destination_node]

        # Composite score (all terms are penalties)
        score = (w1 * immediate +
                 w2 * avg_dist +
                 w3 * nearest_dist +
                 w4 * dest_dist)

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
