```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a two-step look-ahead with regret.
    The heuristic evaluates each candidate based on:
        - immediate distance from current node
        - best next step from the candidate (look-ahead)
        - regret: difference between the best and second-best next step
    Lower composite score = better choice.
    """
    alpha = 0.6      # weight for immediate distance
    beta = 0.4       # weight for look-ahead cost
    gamma = 0.3      # weight for regret (promotes flexibility)
    
    best_score = float('inf')
    best_node = None
    
    for candidate in unvisited_nodes:
        # immediate cost
        cost_curr_cand = distance_matrix[current_node][candidate]
        
        remaining = unvisited_nodes - {candidate}
        if remaining:
            # distances from candidate to all other unvisited nodes
            dists = [distance_matrix[candidate][j] for j in remaining]
            sorted_dists = sorted(dists)
            best_next = sorted_dists[0]
            second_best_next = sorted_dists[1] if len(sorted_dists) > 1 else best_next
            regret = second_best_next - best_next
            look_ahead_cost = best_next
        else:
            # last node before returning to destination
            look_ahead_cost = distance_matrix[candidate][destination_node]
            regret = 0.0
        
        score = alpha * cost_curr_cand + beta * look_ahead_cost - gamma * regret
        
        if score < best_score:
            best_score = score
            best_node = candidate
    
    return best_node
```
