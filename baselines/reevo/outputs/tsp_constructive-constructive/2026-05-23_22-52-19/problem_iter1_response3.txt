import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a greedy look-ahead.
    For each candidate, simulate a nearest-neighbor completion of the remaining
    tour (including the final move to destination_node) and choose the candidate
    that minimizes the total estimated tour cost.
    """
    # Helper: simulate nearest-neighbor path from start through all nodes in remaining,
    # ending at destination_node. Returns total distance.
    def simulate_greedy(start: int, remaining: set, dest: int) -> float:
        current = start
        cost = 0.0
        to_visit = set(remaining)
        while to_visit:
            # Find nearest unvisited node
            next_node = min(to_visit, key=lambda n: distance_matrix[current, n])
            cost += distance_matrix[current, next_node]
            current = next_node
            to_visit.remove(next_node)
        cost += distance_matrix[current, dest]
        return cost

    best_candidate = None
    best_estimated_cost = float('inf')

    for j in unvisited_nodes:
        # Remaining nodes after choosing j (all unvisited except j)
        remaining = unvisited_nodes - {j}
        # Estimated total tour cost: move to j + greedy completion from j
        estimated_cost = distance_matrix[current_node, j] + simulate_greedy(j, remaining, destination_node)
        if estimated_cost < best_estimated_cost:
            best_estimated_cost = estimated_cost
            best_candidate = j

    return best_candidate
