```python
import numpy as np
from typing import Set

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: Set[int], distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a look-ahead mechanism.
    For each candidate, simulate a nearest-neighbor completion from that candidate
    through all remaining unvisited nodes and back to the destination.
    Choose the candidate that minimizes the total estimated tour cost.
    """
    best_candidate = None
    best_total_estimate = float('inf')
    # Cache for look-ahead computations (optional, helps if same unvisited set recurs)
    # Not implemented for simplicity, but can be added.

    for candidate in unvisited_nodes:
        # Simulation: tour so far: current -> candidate
        total_estimate = distance_matrix[current_node][candidate]
        # Remaining nodes after picking candidate
        remaining = unvisited_nodes - {candidate}
        if remaining:
            # Greedy nearest-neighbor from candidate through all remaining nodes
            current_sim = candidate
            while remaining:
                # Find nearest unvisited node from current_sim
                min_dist = float('inf')
                next_sim = None
                for node in remaining:
                    d = distance_matrix[current_sim][node]
                    if d < min_dist:
                        min_dist = d
                        next_sim = node
                total_estimate += min_dist
                remaining.remove(next_sim)
                current_sim = next_sim
            # Finally return to destination from last node
            total_estimate += distance_matrix[current_sim][destination_node]
        else:
            # No remaining nodes: go directly to destination from candidate
            total_estimate += distance_matrix[candidate][destination_node]

        if total_estimate < best_total_estimate:
            best_total_estimate = total_estimate
            best_candidate = candidate

    return best_candidate
```
