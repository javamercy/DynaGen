import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using a look-ahead heuristic."""
    best_score = float('inf')
    best_node = None
    # Greedy baseline: minimum distance from current to any unvisited node
    min_local_distance = min(distance_matrix[current_node][node] for node in unvisited_nodes)
    
    for node in unvisited_nodes:
        # 1. Local edge cost
        local_cost = distance_matrix[current_node][node]
        
        # 2. Regret: how much worse than the greedy choice
        regret = local_cost - min_local_distance
        
        # 3. Look-ahead: simulate next best step from candidate node
        remaining = unvisited_nodes - {node}
        if remaining:
            next_best_cost = min(distance_matrix[node][n] for n in remaining)
            avg_dist = np.mean([distance_matrix[node][n] for n in remaining])
            std_dist = np.std([distance_matrix[node][n] for n in remaining])
        else:
            next_best_cost = distance_matrix[node][destination_node]
            avg_dist = 0.0
            std_dist = 0.0
        
        # 4. Destination cost
        dest_cost = distance_matrix[node][destination_node]
        
        # Weighted combination (sums to 1)
        score = (0.3 * local_cost +
                 0.2 * regret +
                 0.3 * next_best_cost +
                 0.1 * avg_dist +
                 0.05 * std_dist +
                 0.05 * dest_cost)
        
        if score < best_score:
            best_score = score
            best_node = node
    
    return best_node
