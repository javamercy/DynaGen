```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """Select next node using a dynamic look‑ahead heuristic.

    For each candidate, the total tour cost is estimated as:
        immediate_cost = dist(current, candidate)
        remaining_cost = greedy path cost from candidate to destination
                         through all other unvisited nodes.
    The final score is a weighted sum, where the weight alpha gives more
    importance to the look‑ahead when many nodes remain.
    """
    total_nodes = distance_matrix.shape[0]
    n_unvisited = len(unvisited_nodes)
    # alpha grows with the remaining fraction of nodes (excluding current and destination)
    # when few nodes are left, the immediate cost matters more.
    total_remaining_steps = total_nodes - 2  # current and destination are fixed
    if total_remaining_steps > 0:
        alpha = n_unvisited / total_remaining_steps
    else:
        alpha = 0.0

    best_candidate = None
    best_score = float('inf')

    def greedy_path_cost(start, end, nodes, dist):
        """Compute cost of a greedy nearest‑neighbour path from start to end
        visiting all nodes in the set `nodes` (excluding start and end)."""
        visited = {start}
        current = start
        cost = 0.0
        remaining = set(nodes)
        while remaining:
            # nearest neighbour
            next_node = min(remaining, key=lambda n: dist[current][n])
            cost += dist[current][next_node]
            current = next_node
            visited.add(current)
            remaining.remove(current)
        cost += dist[current][end]  # final edge to destination
        return cost

    for candidate in unvisited_nodes:
        imm = distance_matrix[current_node][candidate]
        others = unvisited_nodes - {candidate}
        lookahead = greedy_path_cost(candidate, destination_node, others, distance_matrix)
        score = (1 - alpha) * imm + alpha * lookahead
        if score < best_score:
            best_score = score
            best_candidate = candidate

    return best_candidate
```
