import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a regret-based heuristic with look-ahead.
    For each candidate node, compute:
        immediate_cost = distance from current_node to candidate
        best_next = distance from candidate to the nearest other unvisited node (or to destination if none)
        second_best_next = distance from candidate to the second nearest other unvisited node
        regret = second_best_next - best_next (0 if only one other unvisited)
    The candidate with the largest regret (urgency) is chosen, breaking ties by lowest immediate_cost.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_regret = -float('inf')
    best_immediate = float('inf')

    for node in unvisited_nodes:
        immediate = distance_matrix[current_node][node]

        # Collect distances from candidate to all other unvisited nodes
        others = [k for k in unvisited_nodes if k != node]
        if others:
            distances = [distance_matrix[node][k] for k in others]
            distances.sort()
            best = distances[0]
            second_best = distances[1] if len(distances) > 1 else distances[0]
            regret = second_best - best
        else:
            # Last unvisited node, next step must go to destination
            regret = 0.0  # No choice left

        # Incorporate destination proximity as a soft factor (optional)
        # Not used directly in regret, but can be added as tie-breaker

        # Choose node with largest regret; tie-break by smallest immediate cost
        if regret > best_regret or (regret == best_regret and immediate < best_immediate):
            best_regret = regret
            best_immediate = immediate
            best_node = node

    return best_node
