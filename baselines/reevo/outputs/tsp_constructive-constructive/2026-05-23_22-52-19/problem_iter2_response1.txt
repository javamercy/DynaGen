```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a lookahead heuristic with regret.
    For each candidate:
      - immediate cost from current to candidate
      - best second step (lookahead) from candidate to any remaining unvisited node (or to destination)
      - regret: difference between the best and second-best second step (urgency to take the best next edge)
      - distance to the final destination (bias for early return planning)
    The composite score minimizes immediate + lookahead + destination bias - regret
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        # Immediate cost
        first_cost = distance_matrix[current_node][candidate]

        # Lookahead: best second step from candidate
        remaining = unvisited_nodes - {candidate}
        if remaining:
            second_costs = [distance_matrix[candidate][j] for j in remaining]
            second_best = min(second_costs)
            # Regret: how much worse is the second best
            if len(second_costs) >= 2:
                sorted_second = sorted(second_costs)
                regret = sorted_second[1] - sorted_second[0]
            else:
                regret = 0.0
        else:
            # Last node: must go to destination
            second_best = distance_matrix[candidate][destination_node]
            regret = 0.0

        # Proximity to final destination
        dest_cost = distance_matrix[candidate][destination_node]

        # Composite score (weights tuned to balance immediate, lookahead, regret, and destination)
        score = first_cost + 0.5 * second_best + 0.1 * dest_cost - 0.2 * regret

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
```
