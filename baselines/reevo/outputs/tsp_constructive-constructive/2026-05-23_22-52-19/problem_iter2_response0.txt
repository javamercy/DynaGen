```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using statistical features of candidate distances to unvisited nodes.
    The score for each candidate combines:
    - Immediate distance from current node (greedy)
    - Mean and standard deviation of distances from candidate to all other unvisited nodes (robust lookahead)
    - Minimum distance from candidate to any other unvisited node (best immediate future step)
    - Distance from candidate to destination (early return planning)
    All components penalize larger values; lower total score is better.
    """
    best_node = None
    best_score = float('inf')
    remaining_set = unvisited_nodes

    for node in remaining_set:
        # Immediate cost
        imm = distance_matrix[current_node][node]

        # Distances from candidate to all other unvisited nodes
        others = [n for n in remaining_set if n != node]
        if others:
            dists = [distance_matrix[node][n] for n in others]
            mean_dist = np.mean(dists)
            std_dist = np.std(dists)
            min_future = np.min(dists)
        else:
            # No other unvisited nodes: must return to destination
            mean_dist = 0.0
            std_dist = 0.0
            min_future = distance_matrix[node][destination_node]

        # Distance to final destination
        dest_dist = distance_matrix[node][destination_node]

        # Weighted sum (all positive coefficients)
        # Weights can be tuned; these are chosen to balance greedy vs lookahead
        score = (1.0 * imm +
                 0.6 * mean_dist +
                 0.2 * std_dist +
                 0.5 * min_future +
                 0.3 * dest_dist)

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
