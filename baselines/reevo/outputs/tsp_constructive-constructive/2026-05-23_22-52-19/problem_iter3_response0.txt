import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit with look-ahead: combines direct edge cost,
    statistical features of the candidate's unvisited neighbors, distance to
    destination, and the minimum distance from candidate to any other
    unvisited node (look-ahead).
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Adaptive weights: more look-ahead when many nodes remain
    n = len(unvisited_nodes)
    w1 = 0.4          # current edge cost
    w2 = 0.2          # average distance to unvisited
    w3 = 0.1          # std dev distance to unvisited
    w4 = 0.1          # distance to destination
    w5 = 0.2 * (1 + 0.5 * (1 - n / 10.0))  # look-ahead (min edge to another unvisited)
    w5 = max(0.05, min(0.5, w5))  # clamp

    scores = {}
    for node in unvisited_nodes:
        # distances from candidate to other unvisited nodes
        neighbors = [distance_matrix[node][i] for i in unvisited_nodes if i != node]
        if not neighbors:
            avg_dist = 0.0
            std_dist = 0.0
            min_nb = 0.0
        else:
            avg_dist = np.mean(neighbors)
            std_dist = np.std(neighbors)
            min_nb = np.min(neighbors)

        # score: lower is better
        score = (w1 * distance_matrix[current_node][node]
                 - w2 * avg_dist
                 + w3 * std_dist
                 - w4 * distance_matrix[destination_node][node]
                 + w5 * min_nb)   # positive look-ahead term (minimizing cost)
        scores[node] = score

    return min(scores, key=scores.get)
