import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using a full greedy look-ahead from each candidate."""
    best_node = None
    best_total = float('inf')
    
    for candidate in unvisited_nodes:
        # Build the greedy path starting from candidate
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        
        # Simulate nearest neighbor from candidate covering all remaining nodes
        path_cost = 0.0
        cur = candidate
        while remaining:
            # Find nearest unvisited node
            best_next = min(remaining, key=lambda n: distance_matrix[cur][n])
            path_cost += distance_matrix[cur][best_next]
            cur = best_next
            remaining.remove(best_next)
        # Return to destination
        path_cost += distance_matrix[cur][destination_node]
        
        # Total cost from current node
        total_cost = distance_matrix[current_node][candidate] + path_cost
        
        if total_cost < best_total:
            best_total = total_cost
            best_node = candidate
            
    return best_node
