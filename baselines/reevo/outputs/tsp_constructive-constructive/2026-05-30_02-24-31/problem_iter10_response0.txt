import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved heuristic: uses regret relative to direct edge and balanced weights.
    Two-step regret = best indirect path - direct edge.
    Future regret = second best outgoing - best outgoing.
    Score = w_imm * imm - w_two * regret_two - w_future * regret_future - w_dest * dest_dist.
    Weights are tuned to emphasise regret terms.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for tie-breaking
    candidates = sorted(unvisited_nodes)

    # Balanced weights: regret terms get higher weight
    w_imm, w_two, w_future, w_dest = 0.15, 0.35, 0.35, 0.15

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # Immediate distance from current node to i
        imm = distance_matrix[current_node, i]

        # Two-step regret: best indirect path via any other j, relative to direct edge
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        if best_two_step != float('inf'):
            regret_two = best_two_step - imm  # positive if indirect worse
        else:
            regret_two = 0.0

        # Future regret: difference between second-best and best outgoing edge from i
        future_costs = [distance_matrix[i, v] for v in unvisited_nodes if v != i]
        if len(future_costs) >= 2:
            sorted_future = sorted(future_costs)
            regret_future = sorted_future[1] - sorted_future[0]
        else:
            regret_future = 0.0

        # Distance to final destination
        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better)
        score = w_imm * imm - w_two * regret_two - w_future * regret_future - w_dest * dest_dist

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and (best_node is None or i < best_node)):
            best_score = score
            best_node = i

    return best_node
