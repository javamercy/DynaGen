```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost, two-step lookahead regret,
    and a lookahead savings term that measures the benefit of an intermediate
    node before reaching the destination. Weights are tuned to emphasize regret.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for tie-breaking
    candidates = sorted(unvisited_nodes)

    # Weights (emphasize regret, include savings, no destination distance)
    w_imm, w_two, w_sav = 0.2, 0.5, 0.3

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # 1. Immediate cost
        imm = distance_matrix[current_node, i]

        # 2. Two-step regret: best indirect path from current to i via another node
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret_two = best_two_step - imm if best_two_step != float('inf') else 0.0

        # 3. Lookahead savings: benefit of going to another node j before destination
        # Compute best(i -> j -> destination)
        best_continuation = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[i, j] + distance_matrix[j, destination_node]
            if cost < best_continuation:
                best_continuation = cost
        direct_to_dest = distance_matrix[i, destination_node]
        savings = max(0.0, direct_to_dest - best_continuation) if best_continuation != float('inf') else 0.0

        # Combined score (lower is better)
        score = w_imm * imm - w_two * regret_two - w_sav * savings

        # Deterministic tie-breaking
        if score < best_score or (np.isclose(score, best_score) and (best_node is None or i < best_node)):
            best_score = score
            best_node = i

    return best_node
```
