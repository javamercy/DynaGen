import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a combination of:
    - immediate cost,
    - two-step regret (best vs second-best successor path from current through another node to candidate),
    - future edge regret (second-best vs best outgoing edge from candidate to remaining unvisited nodes),
    - distance to the final destination (penalizing far nodes).
    Weights are tuned to emphasize regret, and tie-breaking is deterministic by node index.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for tie-breaking
    candidates = sorted(unvisited_nodes)

    # Weights: immediate cost, two-step regret, future regret, destination distance
    w_imm, w_two, w_future, w_dest = 0.2, 0.4, 0.3, 0.1

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # Immediate distance from current node to i
        imm = distance_matrix[current_node, i]

        # Two-step lookahead regret: compare best two paths that end at i
        best_two_step = float('inf')
        second_best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                second_best_two_step = best_two_step
                best_two_step = cost
            elif cost < second_best_two_step:
                second_best_two_step = cost
        if second_best_two_step == float('inf'):  # only one other node
            second_best_two_step = best_two_step
        regret_two = second_best_two_step - best_two_step

        # Future regret: difference between second-best and best edge from i to other unvisited nodes
        future_costs = [distance_matrix[i, v] for v in unvisited_nodes if v != i]
        if len(future_costs) >= 2:
            sorted_future = sorted(future_costs)
            regret_future = sorted_future[1] - sorted_future[0]
        else:
            regret_future = 0.0

        # Distance to the final destination (penalized)
        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better): penalize high immediate cost and large destination distance,
        # reward high regret (i.e., subtract regret terms)
        score = w_imm * imm - w_two * regret_two - w_future * regret_future + w_dest * dest_dist

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
