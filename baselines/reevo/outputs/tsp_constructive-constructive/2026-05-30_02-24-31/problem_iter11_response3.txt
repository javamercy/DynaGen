import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved version: combine immediate cost, two-step lookahead regret (relative to direct edge),
    and distance to final destination. Simpler weights, no future regret.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering
    candidates = sorted(unvisited_nodes)

    # Tuned weights – prioritize regret (two-step) and immediate cost, use destination distance
    w_imm, w_two, w_dest = 0.3, 0.5, 0.2

    best_score = float('inf')
    best_node = None

    for i in candidates:
        imm = distance_matrix[current_node, i]

        # Two-step lookahead: best indirect path via any other unvisited node j
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret_two = best_two_step - imm if best_two_step != float('inf') else 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Score: lower is better (negative weights for regret and destination make them desirable)
        score = w_imm * imm - w_two * regret_two - w_dest * dest_dist

        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
