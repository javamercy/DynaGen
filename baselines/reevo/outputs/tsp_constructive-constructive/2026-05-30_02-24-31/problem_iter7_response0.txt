import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a combination of immediate cost, two-step regret (best vs second-best successor),
    future edge regret (from candidate to other unvisited nodes), and distance to final destination.
    Emphasizes regret to avoid costly future connections; uses deterministic tie-breaking.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = sorted(unvisited_nodes)  # deterministic ordering for tie-breaking

    # Weights tuned to emphasise regret: reduce immediate cost, increase regret/impact
    w_imm, w_two, w_future, w_dest = 0.2, 0.4, 0.3, 0.1

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # Immediate distance
        imm = distance_matrix[current_node, i]

        # Two-step regret: best vs second-best path via an intermediate node j
        paths = []
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            paths.append(cost)
        if len(paths) >= 2:
            sorted_paths = sorted(paths)
            regret_two = sorted_paths[1] - sorted_paths[0]
        elif len(paths) == 1:
            # Only one alternative, regret is the difference to direct edge (though direct not used)
            regret_two = paths[0] - imm
        else:
            regret_two = 0.0

        # Future regret (from i to other unvisited nodes)
        future_costs = [distance_matrix[i, v] for v in unvisited_nodes if v != i]
        if len(future_costs) >= 2:
            sorted_future = sorted(future_costs)
            regret_future = sorted_future[1] - sorted_future[0]
        elif len(future_costs) == 1:
            regret_future = future_costs[0]  # only one edge, no regret (treat as cost)
        else:
            regret_future = 0.0

        # Distance to final destination
        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better)
        score = w_imm * imm - w_two * regret_two - w_future * regret_future - w_dest * dest_dist

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
