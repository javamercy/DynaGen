```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost, two-step regret (relative to direct edge),
    and distance to destination. Regret is prioritized, no normalization, deterministic tie-breaking.
    Simpler than v1, removing future regret component.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for tie-breaking
    candidates = sorted(unvisited_nodes)

    # Weights: prioritize two-step regret
    w_imm = 0.2
    w_two = 0.5
    w_dest = 0.3

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # Immediate distance from current to i
        imm = distance_matrix[current_node, i]

        # Two-step lookahead regret: best indirect path via any other j minus direct edge
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret_two = best_two_step - imm if best_two_step != float('inf') else 0.0

        # Distance to the final destination
        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better); regret contributes negatively (prefer high regret)
        score = w_imm * imm - w_two * regret_two - w_dest * dest_dist

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
```
