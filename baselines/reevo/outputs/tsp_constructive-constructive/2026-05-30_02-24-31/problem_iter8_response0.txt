def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost, two-step regret, a greedy nearest‑neighbour lookahead that
    estimates the total remaining path cost, and distance to destination. Weights are tuned to give
    more importance to lookahead. Deterministic tie‑breaking by smallest node index.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = sorted(unvisited_nodes)          # deterministic order

    # weights: immediate, regret, lookahead, dest_dist
    w1, w2, w3, w4 = 0.2, 0.3, 0.3, 0.2

    scores = {}
    for i in unvisited:
        immediate = distance_matrix[current_node, i]

        # ----- regret -----
        best_alternative = float('inf')
        for j in unvisited:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_alternative:
                best_alternative = cost
        regret = best_alternative - immediate if best_alternative != float('inf') else 0.0

        # ----- lookahead: greedy nearest‑neighbour path from i to destination through all other unvisited nodes -----
        remaining = [node for node in unvisited if node != i]
        curr = i
        lookahead_cost = 0.0
        while remaining:
            # find nearest unvisited node from current
            nearest = min(remaining, key=lambda n: distance_matrix[curr, n])
            lookahead_cost += distance_matrix[curr, nearest]
            curr = nearest
            remaining.remove(nearest)
        # finally go to destination
        lookahead_cost += distance_matrix[curr, destination_node]

        dest_dist = distance_matrix[i, destination_node]

        # lower score is better
        score = w1 * immediate - w2 * regret - w3 * lookahead_cost - w4 * dest_dist
        scores[i] = score

    best_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == best_score]
    return min(candidates)
