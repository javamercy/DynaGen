import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a regret-based look-ahead heuristic that also considers
    the final destination to avoid costly future connections.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Weights emphasizing regret: high beta promotes high-regret nodes (postponing them would be costly)
    alpha = 0.3   # weight for immediate edge cost
    beta  = 0.6   # weight for regret
    gamma = 0.1   # weight for distance to destination

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        dist_curr = distance_matrix[current_node][node]
        dist_dest = distance_matrix[node][destination_node]

        # Compute regret over future nodes: other unvisited nodes plus the final destination
        other_unvisited = [n for n in unvisited_nodes if n != node]
        future_nodes = other_unvisited + [destination_node]

        if len(future_nodes) >= 2:
            # Sort distances from current candidate to all future nodes
            dists = sorted(distance_matrix[node][n] for n in future_nodes)
            regret = dists[1] - dists[0]   # difference between second best and best
        else:
            regret = 0.0

        # Score: lower is better (small distance, high regret, moderate destination distance)
        score = alpha * dist_curr - beta * regret + gamma * dist_dest

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
