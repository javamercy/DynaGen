import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit by simulating a greedy tour from each candidate (lookahead)."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_node = None
    best_total_cost = float('inf')
    
    for candidate in unvisited_nodes:
        # Simulate a greedy nearest-neighbor tour starting from this candidate
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        
        current = candidate
        total_cost = distance_matrix[current_node][candidate]
        
        while remaining:
            next_node = min(remaining, key=lambda n: distance_matrix[current][n])
            total_cost += distance_matrix[current][next_node]
            current = next_node
            remaining.remove(current)
        
        # Return to destination
        total_cost += distance_matrix[current][destination_node]
        
        if total_cost < best_total_cost:
            best_total_cost = total_cost
            best_node = candidate
    
    return best_node
