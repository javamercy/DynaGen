import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Regret-based look-ahead heuristic.
    For each candidate node, compute the best and second-best direct connection cost
    from the candidate to any other unvisited node or to the destination.
    The total insertion cost is distance[current_node][candidate] plus that best connection.
    Regret is the difference between second-best and best connection cost.
    Choose the candidate with maximum regret (most urgent to visit now).
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_regret = -float('inf')

    for candidate in unvisited_nodes:
        # cost to reach candidate from current
        base_cost = distance_matrix[current_node][candidate]

        # collect distances from candidate to all other possible next steps
        # other unvisited nodes (excluding candidate) and the destination
        neighbors = []
        for other in unvisited_nodes:
            if other != candidate:
                neighbors.append(distance_matrix[candidate][other])
        # also consider going directly to destination
        neighbors.append(distance_matrix[candidate][destination_node])

        # sort to get best and second best connection cost
        neighbors.sort()
        best_conn = neighbors[0]
        # second best exists because len(neighbors) >= 2 when unvisited_nodes size >= 2?
        # Actually if unvisited_nodes size = 2, then candidate + one other unvisited + destination => 2 neighbors
        # So at least 2 neighbors are present.
        second_best_conn = neighbors[1]

        regret = second_best_conn - best_conn

        if regret > best_regret:
            best_regret = regret
            best_node = candidate

    return best_node
