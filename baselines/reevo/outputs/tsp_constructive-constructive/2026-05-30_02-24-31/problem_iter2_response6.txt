import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select next node by combining immediate cost, look-ahead distance,
    proximity to destination, and regret (urgency to visit now).
    Uses weighted scoring: lower score is better.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Weights for the composite score
    alpha = 0.5   # weight for look-ahead distance
    beta = 0.3    # weight for proximity to destination (negative sign)
    gamma = 0.2   # weight for regret (negative sign, since high regret is good)

    best_node = None
    best_score = float('inf')

    for u in unvisited_nodes:
        d_current_u = distance_matrix[current_node][u]

        # Collect distances from u to all possible next nodes
        # (other unvisited nodes and the destination)
        neighbors = []
        for v in unvisited_nodes:
            if v != u:
                neighbors.append(distance_matrix[u][v])
        neighbors.append(distance_matrix[u][destination_node])

        # Sort to get best and second-best connection costs
        neighbors.sort()
        best_conn = neighbors[0]
        second_best_conn = neighbors[1]  # At least two entries due to early exit

        look_ahead = best_conn
        regret = second_best_conn - best_conn
        d_u_dest = distance_matrix[u][destination_node]

        # Composite score (lower is better)
        score = (
            d_current_u
            + alpha * look_ahead
            - beta * d_u_dest
            - gamma * regret
        )

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
