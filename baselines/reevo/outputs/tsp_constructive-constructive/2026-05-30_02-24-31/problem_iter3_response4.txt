import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node to visit using immediate cost, regret look-ahead, and destination distance."""
    # If only one unvisited node, return it directly
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Weights (tuned)
    w_immediate = 0.4
    w_regret = 0.3
    w_dest = 0.1
    
    scores = {}
    for node in unvisited_nodes:
        # Immediate cost
        immediate = distance_matrix[current_node][node]
        
        # Regret look-ahead: difference between best and second best future cost from this node
        future_costs = [distance_matrix[node][m] for m in unvisited_nodes if m != node]
        if len(future_costs) >= 2:
            sorted_costs = sorted(future_costs)
            best = sorted_costs[0]
            second_best = sorted_costs[1]
            regret = second_best - best
        else:
            regret = 0.0
        
        # Destination distance
        dest_dist = distance_matrix[node][destination_node]
        
        # Score (lower is better)
        score = w_immediate * immediate - w_regret * regret + w_dest * dest_dist
        scores[node] = score
    
    # Deterministic tie-breaking: min score, then smallest node index
    best_node = min(scores, key=lambda n: (scores[n], n))
    return best_node
