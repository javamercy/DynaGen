import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead greedy strategy.
    For each candidate, simulate a greedy tour of the remaining nodes and pick
    the candidate that minimizes the total estimated cost to complete the tour.
    """
    best_next = None
    best_cost = float('inf')
    
    for candidate in unvisited_nodes:
        immediate_cost = distance_matrix[current_node][candidate]
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        
        # Simulate nearest-neighbor from candidate to cover all remaining nodes
        current = candidate
        tour_cost = 0.0
        while remaining:
            # Find nearest node in remaining
            next_node = min(remaining, key=lambda n: distance_matrix[current][n])
            tour_cost += distance_matrix[current][next_node]
            current = next_node
            remaining.remove(next_node)
        # Finally go to destination
        tour_cost += distance_matrix[current][destination_node]
        
        total_estimated = immediate_cost + tour_cost
        if total_estimated < best_cost:
            best_cost = total_estimated
            best_next = candidate
    
    return best_next
