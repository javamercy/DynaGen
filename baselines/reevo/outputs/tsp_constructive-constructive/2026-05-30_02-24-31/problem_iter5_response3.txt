import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost, regret-based two-step lookahead,
    and distance to destination. Prioritizes nodes with high regret (second-best minus best)
    and those far from the destination by subtracting these components from the score.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for tie-breaking
    candidates = sorted(unvisited_nodes)

    # Weights tuned for balanced performance (immediate, regret, destination distance)
    w1, w2, w3 = 0.4, 0.3, 0.3

    scores = {}
    for i in candidates:
        # Immediate distance from current node to i
        immediate = distance_matrix[current_node, i]

        # Regret = best alternative two-step path to reach i minus the direct path.
        # Equivalent to second-best minus best (here best is the direct path).
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            # Indirect path: current -> j -> i
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret = best_two_step - immediate if best_two_step != float('inf') else 0.0

        # Distance from candidate i to the final destination
        dest_dist = distance_matrix[i, destination_node]

        # Combined score: lower is better.
        # High regret and high dest_dist reduce the score (favored).
        score = w1 * immediate - w2 * regret - w3 * dest_dist
        scores[i] = score

    # Deterministic tie-breaking: smallest index among those with minimal score
    min_score = min(scores.values())
    best_candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(best_candidates)
