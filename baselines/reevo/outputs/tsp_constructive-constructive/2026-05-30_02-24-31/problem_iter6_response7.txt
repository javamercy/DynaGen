import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a weighted combination of immediate cost, regret, 
    and destination distance. Weights are tuned to emphasize regret and 
    destination distance over immediate cost.

    For each candidate node i:
      - immediate = distance from current to i
      - regret = best alternative cost to visit i via another node minus immediate
      - dest_dist = distance from i to destination

    Score = w1 * immediate - w2 * regret - w3 * dest_dist   (lower is better)

    Parameters
    ----------
    current_node : int
        The current node index.
    destination_node : int
        The final node that must be reached after visiting all nodes.
    unvisited_nodes : set of int
        Set of node indices not yet visited.
    distance_matrix : np.ndarray
        Square matrix of distances.

    Returns
    -------
    int
        The next node to visit. Assumes unvisited is non-empty.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Tuned weights: immediate less important, regret and dest distance more
    w1, w2, w3 = 0.2, 0.4, 0.4

    unvisited = sorted(unvisited_nodes)  # deterministic order
    scores = {}

    for i in unvisited:
        immediate = distance_matrix[current_node, i]

        # Compute regret: best alternative cost to visit i via a different node j
        best_alternative = float('inf')
        for j in unvisited:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_alternative:
                best_alternative = cost
        regret = best_alternative - immediate if best_alternative != float('inf') else 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Combined score: lower is better, high regret and dest dist lower the score
        score = w1 * immediate - w2 * regret - w3 * dest_dist
        scores[i] = score

    # Deterministic tie-breaking: smallest index among minimal score
    min_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(candidates)
