import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a 2-step look-ahead with regret and destination awareness.
    """
    unvisited = list(unvisited_nodes)
    n = len(unvisited)
    if n == 1:
        return unvisited[0]

    scores = {}
    for node in unvisited:
        d_cur = distance_matrix[current_node, node]
        # Compute distances from node to all other unvisited nodes
        others = [o for o in unvisited if o != node]
        if not others:
            # Only one node left in unvisited (but n>1 means others not empty)
            scores[node] = d_cur
            continue

        dists = [distance_matrix[node, o] for o in others]
        dists.sort()
        best_next = dists[0]
        second_best = dists[1] if len(dists) >= 2 else best_next  # if only one other, regret=0
        regret = second_best - best_next  # non-negative

        # Distance to the final destination
        d_dest = distance_matrix[node, destination_node]

        # Weights for the composite score (minimized)
        w1, w2, w3, w4 = 0.35, 0.35, 0.20, 0.10
        # Score: lower is better
        # - regret is subtracted because high regret (large difference) means we should capture the good edge now
        score = (w1 * d_cur +
                 w2 * best_next -
                 w3 * regret +
                 w4 * d_dest)

        scores[node] = score

    # Return node with minimum score
    return min(scores, key=scores.get)
