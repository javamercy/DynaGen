import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved version incorporating lookahead savings and two-step regret,
    with weights tuned to emphasize future benefits.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Sort for deterministic tie-breaking
    unvisited = sorted(unvisited_nodes)

    # Weights: emphasis on regret and lookahead savings; small weight on immediate cost
    w_imm = 0.15
    w_reg = 0.40
    w_dest = 0.20
    w_sav = 0.25

    best_score = float('inf')
    best_node = None

    for u in unvisited:
        # Immediate cost from current node
        imm = distance_matrix[current_node, u]

        # Two-step regret: difference between best two-step cost and immediate cost
        # best two-step: min_{j != u} (distance[current_node, j] + distance[j, u])
        best_alternative = float('inf')
        for j in unvisited:
            if j == u:
                continue
            alt_cost = distance_matrix[current_node, j] + distance_matrix[j, u]
            if alt_cost < best_alternative:
                best_alternative = alt_cost
        regret = best_alternative - imm if best_alternative != float('inf') else 0.0

        # Distance to final destination (positive weight to prefer closeness)
        dest_dist = distance_matrix[u, destination_node]

        # Lookahead savings: best forward edge from current to any unvisited minus
        # best forward edge from candidate u to any other unvisited
        best_from_current = min(distance_matrix[current_node, j] for j in unvisited)
        best_from_u = min(distance_matrix[u, j] for j in unvisited if j != u)
        lookahead_savings = best_from_current - best_from_u

        # Composite score (lower is better)
        score = w_imm * imm - w_reg * regret + w_dest * dest_dist - w_sav * lookahead_savings

        # Deterministic tie-breaking: prefer smaller node index
        if score < best_score - 1e-12:
            best_score = score
            best_node = u
        elif abs(score - best_score) < 1e-12 and u < best_node:
            best_node = u

    return best_node
