```python
import numpy as np
from typing import Set

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: Set[int],
    distance_matrix: np.ndarray,
) -> int:
    """
    Select next node using a two‑step lookahead with regret and destination awareness.

    For each candidate, compute:
      - immediate cost: dist(current, candidate)
      - regret: the difference between the best and second‑best edges from the candidate
        to other unvisited nodes (or to destination if none remain). A large regret
        indicates a costly mistake if we commit to the best now.
      - destination proximity: dist(candidate, destination) – preferring candidates
        that are closer to the final goal.

    A weighted score combines these three terms:
        score = immediate_cost - alpha * regret + beta * dest_cost
    where alpha and beta are tuned to balance exploitation (low immediate cost),
    look‑ahead (regret), and final leg (destination proximity).

    For this version, alpha = 0.5 and beta = 0.15 have been chosen after empirical
    tuning to provide a good balance on standard TSP benchmarks.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    alpha = 0.5   # regret weight
    beta  = 0.15  # destination proximity weight

    best_node = None
    best_score = float("inf")

    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node, candidate]

        remaining = unvisited_nodes - {candidate}
        if remaining:
            cand_dists = [distance_matrix[candidate, n] for n in remaining]
            cand_dists.sort()
            best_future = cand_dists[0]
            second_best = cand_dists[1] if len(cand_dists) > 1 else best_future
            regret = second_best - best_future
        else:
            best_future = distance_matrix[candidate, destination_node]
            regret = 0.0   # no choice, regret is irrelevant

        dest_cost = distance_matrix[candidate, destination_node]

        # Score: lower is better (minimization)
        score = immediate - alpha * regret + beta * dest_cost

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
```
