```python
import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node using a composite score with fixed, well-tuned weights.
    Lower scores are preferred.

    Score = w1 * immediate_cost - w2 * regret + w3 * look_ahead - w4 * dest_dist

    - immediate_cost: distance from current_node to candidate
    - regret: best alternative two-step cost (via another unvisited node) minus
      immediate cost (high regret means it is costly to postpone this node)
    - look_ahead: minimum distance from candidate to any other unvisited node,
      or to destination if no other unvisited nodes remain
    - dest_dist: distance from candidate to destination_node

    Weights: w1=0.35, w2=0.35, w3=0.15, w4=0.15

    Tie-breaking: smallest index among nodes with the minimum score.
    """
    if len(unvisited_nodes) == 0:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = list(unvisited_nodes)
    w1, w2, w3, w4 = 0.35, 0.35, 0.15, 0.15

    best_score = float('inf')
    best_node = None

    for u in unvisited:
        # Immediate cost
        imm = distance_matrix[current_node, u]

        # Regret: best two-step alternative
        best_alt = float('inf')
        for v in unvisited:
            if v == u:
                continue
            alt = distance_matrix[current_node, v] + distance_matrix[v, u]
            if alt < best_alt:
                best_alt = alt
        regret = best_alt - imm if best_alt != float('inf') else 0.0

        # Look-ahead: best future edge from candidate
        look_candidates = [distance_matrix[u, v] for v in unvisited if v != u]
        if look_candidates:
            look_ahead = min(look_candidates)
        else:
            look_ahead = distance_matrix[u, destination_node]

        # Distance to final destination
        dest_dist = distance_matrix[u, destination_node]

        # Composite score (lower is better)
        score = w1 * imm - w2 * regret + w3 * look_ahead - w4 * dest_dist

        # Update best with deterministic tie-breaking (smaller index)
        if score < best_score - 1e-12:
            best_score = score
            best_node = u
        elif abs(score - best_score) <= 1e-12 and (best_node is None or u < best_node):
            best_node = u

    return best_node
```
