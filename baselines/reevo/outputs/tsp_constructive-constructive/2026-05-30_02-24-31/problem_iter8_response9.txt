import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Improved selection heuristic with adaptive weighting based on remaining nodes,
    incorporating two-step regret, lookahead savings, and destination proximity.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = sorted(unvisited_nodes)
    n_remaining = len(unvisited)

    # Adaptive weights: when many nodes remain, favour lookahead and regret;
    # when few remain, favour direct cost and destination proximity.
    # Coefficients determined by remaining fraction (relative to total? Here we just use n_remaining).
    # But we don't know total number of nodes. Use a simple exponential:
    alpha = 0.2 + 0.3 * (1 - 1.0 / (1 + n_remaining))   # increases with n_remaining
    beta  = 0.3 - 0.1 * (1 - 1.0 / (1 + n_remaining))   # decreases
    gamma = 0.2 + 0.1 * (1.0 / (1 + n_remaining))       # increases when few left
    delta = 0.3 - 0.2 * (1.0 / (1 + n_remaining))       # decreases when few left

    best_score = float('inf')
    best_node = None

    # Precompute best outgoing edge from current to any unvisited node (for lookahead savings)
    best_from_current = min(distance_matrix[current_node, j] for j in unvisited)

    for u in unvisited:
        # Immediate cost
        imm = distance_matrix[current_node, u]

        # Two-step regret: best alternative indirect cost via another node
        best_alt = float('inf')
        for j in unvisited:
            if j == u:
                continue
            alt_cost = distance_matrix[current_node, j] + distance_matrix[j, u]
            if alt_cost < best_alt:
                best_alt = alt_cost
        regret = best_alt - imm if best_alt != float('inf') else 0.0

        # Distance to final destination
        dest_dist = distance_matrix[u, destination_node]

        # Lookahead savings: best outgoing edge from candidate to others (not itself)
        best_from_u = min(distance_matrix[u, j] for j in unvisited if j != u)
        lookahead_savings = best_from_current - best_from_u

        # Composite score (lower is better)
        score = (
            alpha * imm
            - beta * regret
            + gamma * dest_dist
            - delta * lookahead_savings
        )

        # Deterministic tie-breaking
        if score < best_score - 1e-12:
            best_score = score
            best_node = u
        elif abs(score - best_score) < 1e-12 and (best_node is None or u < best_node):
            best_node = u

    return best_node
