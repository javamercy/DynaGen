```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved heuristic: balances immediate cost, regret (detour penalty),
    and distance to the final destination.

    Regret is defined as the extra cost incurred if the candidate is not taken now
    but reached later via some other unvisited node (detour). This measure is more
    meaningful than simple outgoing-edge regret.

    Lower score is better. Deterministic tie-breaking by node index.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = sorted(unvisited_nodes)          # deterministic order
    n_candidates = len(candidates)

    # Weights: emphasise regret and destination proximity
    w_imm, w_reg, w_dest = 0.2, 0.4, 0.4

    best_score = float('inf')
    best_node = candidates[0]

    # Precompute distances from current node to all candidates for efficiency
    dist_from_current = distance_matrix[current_node, candidates]

    for idx, i in enumerate(candidates):
        immediate = dist_from_current[idx]

        # Compute regret: best alternative cost to reach i via a detour j → i
        # where j is another unvisited node (j != i)
        best_alt = float('inf')
        for j in candidates:
            if j == i:
                continue
            # cost = go current → j (direct) then j → i
            alt_cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if alt_cost < best_alt:
                best_alt = alt_cost

        regret = best_alt - immediate if best_alt != float('inf') else 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Score: lower is better. Negative weights for regret and dest_dist
        # because larger regret/distance should increase score (worse).
        score = w_imm * immediate - w_reg * regret - w_dest * dest_dist

        # Deterministic tie-breaking (smaller node index wins)
        if score < best_score - 1e-12:
            best_score = score
            best_node = i
        elif abs(score - best_score) <= 1e-12 and i < best_node:
            best_score = score
            best_node = i

    return best_node
```
