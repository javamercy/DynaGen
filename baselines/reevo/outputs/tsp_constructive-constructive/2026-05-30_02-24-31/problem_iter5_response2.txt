import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost, two-step lookahead regret (from candidate),
    and destination distance. Regret is the penalty of not picking the best successor
    of the candidate node.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = sorted(unvisited_nodes)  # deterministic order
    w1, w2, w3 = 0.35, 0.40, 0.25  # tuned weights: emphasise regret

    scores = {}
    for i in unvisited:
        immediate = distance_matrix[current_node, i]

        # Regret based on best and second-best next step from i
        distances_from_i = [distance_matrix[i, j] for j in unvisited if j != i]
        if distances_from_i:
            sorted_dists = sorted(distances_from_i)
            best_next = sorted_dists[0]
            second_best = sorted_dists[1] if len(sorted_dists) > 1 else best_next
            regret = second_best - best_next
        else:
            regret = 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Lower score is better: subtract regret and dest_dist to favour high-regret and far nodes
        score = w1 * immediate - w2 * regret - w3 * dest_dist
        scores[i] = score

    # Deterministic tie-breaking (smallest index)
    min_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(candidates)
