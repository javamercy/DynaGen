import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using two-step lookahead with regret and destination distance."""
    if not unvisited_nodes:
        return destination_node  # safety, should not happen
    unvisited = list(unvisited_nodes)
    if len(unvisited) == 1:
        return unvisited[0]

    # Precompute two-step cost, destination cost, and immediate cost for each candidate
    data = {}
    for u in unvisited:
        immediate = distance_matrix[current_node, u]
        # best next from u
        remaining = [v for v in unvisited if v != u]
        best_next = min(distance_matrix[u, v] for v in remaining)
        two_step = immediate + best_next
        dest_cost = distance_matrix[destination_node, u]
        data[u] = (two_step, dest_cost, immediate)

    # Find best and second best two-step costs for regret calculation
    two_step_sorted = sorted((two_step, u) for u, (two_step, _, _) in data.items())
    best_two_step, best_u = two_step_sorted[0]
    second_best_two_step = two_step_sorted[1][0] if len(two_step_sorted) > 1 else best_two_step

    # Compute regret for each candidate
    regret = {}
    for u in unvisited:
        two_step_u, _, _ = data[u]
        if u == best_u:
            best_other = second_best_two_step
        else:
            best_other = best_two_step
        regret_u = best_other - two_step_u
        regret[u] = regret_u

    # Weighted score: lower is better
    alpha, beta, gamma = 0.5, 0.3, 0.2
    scores = {}
    for u in unvisited:
        two_step, dest_cost, _ = data[u]
        score = alpha * two_step + beta * dest_cost - gamma * regret[u]
        scores[u] = score

    # Deterministic tie-breaking: pick the one with smallest immediate cost
    best_score = min(scores.values())
    best_candidates = [u for u, s in scores.items() if abs(s - best_score) < 1e-12]
    if len(best_candidates) > 1:
        return min(best_candidates, key=lambda u: data[u][2])
    else:
        return best_candidates[0]
