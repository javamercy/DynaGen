import numpy as np
from typing import Set

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: Set[int],
                        distance_matrix: np.ndarray) -> int:
    """
    Improved selection with tuned weights: immediate cost, regret, and
    distance to destination.  Weights are set to β = 0.5, γ = 0.15 to
    give slightly more influence to regret and the final leg than the
    baseline version.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = float('inf')
    beta = 0.5
    gamma = 0.15

    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node, candidate]

        remaining = unvisited_nodes - {candidate}
        if remaining:
            cand_dists = sorted(distance_matrix[candidate, n] for n in remaining)
            best_future = cand_dists[0]
            second_best = cand_dists[1] if len(cand_dists) > 1 else best_future
            regret = second_best - best_future
        else:
            # no other unvisited node, must go to destination
            regret = 0.0

        to_dest = distance_matrix[candidate, destination_node]

        # Lower score is better: combine immediate cost, negative regret,
        # and distance to destination.
        score = immediate - beta * regret + gamma * to_dest

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
