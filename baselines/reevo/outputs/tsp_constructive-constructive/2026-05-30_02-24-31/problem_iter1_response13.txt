import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a 2-step lookahead heuristic.
    For each candidate, compute:
        score = distance(current, candidate) + alpha * min_next_cost(candidate)
    where min_next_cost is the minimum distance from candidate to any remaining unvisited node,
    or to the destination if no nodes remain. Choose the candidate with smallest score.
    """
    alpha = 1.0  # weight for lookahead
    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        immediate = distance_matrix[current_node, candidate]
        remaining = unvisited_nodes - {candidate}
        if remaining:
            # Look one step ahead: best next move from candidate among remaining nodes
            future_min = min(distance_matrix[candidate, n] for n in remaining)
        else:
            # Last node: must go to destination
            future_min = distance_matrix[candidate, destination_node]

        score = immediate + alpha * future_min

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
