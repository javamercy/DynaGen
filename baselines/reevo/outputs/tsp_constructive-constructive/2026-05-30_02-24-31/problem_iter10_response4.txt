import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Improved selection heuristic with fraction-based adaptive weights,
    average two-step regret, and average future cost savings.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = sorted(unvisited_nodes)
    n_remaining = len(unvisited)
    total_nodes = distance_matrix.shape[0]
    fraction = n_remaining / total_nodes

    # Adaptive weights based on remaining fraction
    w_imm = 0.1 + 0.4 * (1 - fraction)   # higher when few remain
    w_dest = 0.2 + 0.5 * (1 - fraction)  # higher when few remain
    w_regret = 0.3 + 0.3 * fraction      # higher when many remain
    w_future = 0.2 + 0.2 * fraction      # higher when many remain

    best_score = float('inf')
    best_node = None

    # Precompute current node distances to all unvisited (for efficiency)
    dist_current = {j: distance_matrix[current_node, j] for j in unvisited}

    for u in unvisited:
        imm = dist_current[u]

        # Average two-step regret: mean of (dist(current,j) + dist(j,u)) for j != u
        alt_costs = []
        for j in unvisited:
            if j == u:
                continue
            alt_costs.append(dist_current[j] + distance_matrix[j, u])
        if alt_costs:
            avg_alt = np.mean(alt_costs)
            regret = avg_alt - imm
        else:
            regret = 0.0

        # Average future cost: mean distance from u to all other unvisited nodes
        future_distances = [distance_matrix[u, v] for v in unvisited if v != u]
        if future_distances:
            avg_future = np.mean(future_distances)
        else:
            avg_future = 0.0

        # Distance to final destination
        dest_dist = distance_matrix[u, destination_node]

        # Composite score (lower is better)
        score = (
            w_imm * imm
            + w_dest * dest_dist
            - w_regret * regret
            - w_future * avg_future
        )

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score - 1e-12:
            best_score = score
            best_node = u
        elif abs(score - best_score) < 1e-12 and (best_node is None or u < best_node):
            best_node = u

    return best_node
