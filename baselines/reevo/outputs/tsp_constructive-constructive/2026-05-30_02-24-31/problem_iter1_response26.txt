import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a two-step look-ahead heuristic.
    For each candidate node, we compute the total estimated tour cost if we go there
    and then complete the tour greedily (nearest neighbor) from that point back to the destination.
    The candidate leading to the smallest total estimated cost is chosen.
    """
    def _nearest_neighbor_tour_cost(start: int, destination: int, unvisited: set, dist: np.ndarray) -> float:
        """Estimate the cost of a tour starting from 'start', visiting all nodes in 'unvisited'
        in greedy nearest-neighbor order, and ending at 'destination'."""
        if not unvisited:
            return dist[start][destination]
        current = start
        total = 0.0
        remaining = set(unvisited)
        while remaining:
            next_node = min(remaining, key=lambda n: dist[current][n])
            total += dist[current][next_node]
            current = next_node
            remaining.remove(next_node)
        total += dist[current][destination]
        return total

    best_node = None
    best_estimated = float('inf')

    for candidate in unvisited_nodes:
        # Cost to move to the candidate
        first_step_cost = distance_matrix[current_node][candidate]
        remaining_nodes = set(unvisited_nodes)
        remaining_nodes.remove(candidate)

        # Estimate remaining tour cost from candidate back to destination
        remaining_cost = _nearest_neighbor_tour_cost(candidate, destination_node, remaining_nodes, distance_matrix)

        total_estimated = first_step_cost + remaining_cost
        if total_estimated < best_estimated:
            best_estimated = total_estimated
            best_node = candidate

    return best_node
