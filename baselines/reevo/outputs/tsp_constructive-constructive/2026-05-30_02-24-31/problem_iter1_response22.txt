import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead heuristic combining insertion cost,
    best next step, and regret.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Precompute direct distance from current to destination
    direct_cur_dest = distance_matrix[current_node][destination_node]
    
    best_score = float('inf')
    best_node = None
    
    for u in unvisited_nodes:
        # Insertion cost if we place u immediately after current
        insertion_cost = distance_matrix[current_node][u] + distance_matrix[u][destination_node] - direct_cur_dest
        
        remaining = unvisited_nodes - {u}
        if remaining:
            # Look-ahead: best next step from u
            next_distances = [distance_matrix[u][v] for v in remaining]
            next_distances.sort()
            best_next = next_distances[0]
            # Regret: difference between best and second best
            regret = next_distances[1] - next_distances[0] if len(next_distances) > 1 else 0.0
        else:
            # Only destination left
            best_next = distance_matrix[u][destination_node]
            regret = 0.0
        
        # Composite score (lower is better)
        score = insertion_cost + 0.5 * best_next - 0.2 * regret
        
        if score < best_score:
            best_score = score
            best_node = u
            
    return best_node
