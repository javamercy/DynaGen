import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using a 2-step look-ahead heuristic."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_score = float('inf')
    best_node = None
    tie_breaker = None
    
    for i in unvisited_nodes:
        # cost from current to i
        cost_ci = distance_matrix[current_node][i]
        # exclude i from future candidates
        future = unvisited_nodes - {i}
        # find best continuation j (including destination effect)
        best_next_cost = float('inf')
        for j in future:
            cost_ij = distance_matrix[i][j]
            cost_jd = distance_matrix[j][destination_node]
            total = cost_ij + cost_jd
            if total < best_next_cost:
                best_next_cost = total
        # if no future (should not happen because len>1)
        if future:
            score = cost_ci + best_next_cost
        else:
            score = cost_ci + distance_matrix[i][destination_node]
        
        # tie breaking by smaller current distance
        if score < best_score or (score == best_score and distance_matrix[current_node][i] < tie_breaker):
            best_score = score
            best_node = i
            tie_breaker = distance_matrix[current_node][i]
    
    return best_node
