import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a two-step look-ahead heuristic.

    For each candidate node u in unvisited_nodes:
        - immediate cost: distance from current_node to u
        - look-ahead cost: minimal distance from u to any remaining node v plus distance from v to destination_node
          (if no remaining nodes, use direct distance from u to destination_node)
        - total = immediate + look-ahead

    The candidate with smallest total cost is chosen. Ties are broken by node index.
    """
    best = None
    best_total = float('inf')
    best_node = None

    for u in unvisited_nodes:
        immediate = distance_matrix[current_node][u]
        remaining = unvisited_nodes - {u}
        if remaining:
            # look-ahead: best v in remaining to reach destination
            look_ahead = min(
                distance_matrix[u][v] + distance_matrix[v][destination_node]
                for v in remaining
            )
        else:
            look_ahead = distance_matrix[u][destination_node]

        total = immediate + look_ahead

        # Deterministic tie-breaking: smaller node index
        if total < best_total or (total == best_total and u < best_node):
            best_total = total
            best_node = u

    return best_node
