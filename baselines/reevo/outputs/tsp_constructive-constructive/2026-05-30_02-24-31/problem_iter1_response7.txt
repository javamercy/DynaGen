```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a 2-step look-ahead with regret.
    For each candidate, evaluates the cost of going there plus the best and second best
    continuation to the destination, and chooses the one minimizing (immediate cost - 0.5 * regret).
    """
    best_score = float('inf')
    best_node = None

    for candidate in unvisited_nodes:
        # Immediate cost: current -> candidate
        immediate_cost = distance_matrix[current_node, candidate]

        # Remaining unvisited nodes after visiting candidate
        remaining = unvisited_nodes - {candidate}

        # Collect costs of continuing from candidate to any remaining node and then to destination
        continuation_costs = []
        if remaining:
            for next_node in remaining:
                cont_cost = distance_matrix[candidate, next_node] + distance_matrix[next_node, destination_node]
                continuation_costs.append(cont_cost)
        # Also consider direct return to destination from candidate
        continuation_costs.append(distance_matrix[candidate, destination_node])

        continuation_costs.sort()
        best_cont = continuation_costs[0]
        second_best_cont = continuation_costs[1] if len(continuation_costs) > 1 else best_cont
        regret = second_best_cont - best_cont

        # Score: lower immediate cost is good, higher regret is valuable (so subtract regret)
        score = immediate_cost - 0.5 * regret

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
```
