import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a weighted combination of immediate cost, regret, and distance to destination.
    Weights are exactly balanced (1/3 each) to avoid over-penalizing any single component.
    This simplification follows the reflection: balanced weights outperform many small components.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = list(unvisited_nodes)
    w = 1.0 / 3.0  # equal balance

    scores = {}
    for i in unvisited:
        immediate = distance_matrix[current_node, i]

        # Regret: best alternative two-step path that visits i later
        best_alternative = float('inf')
        for j in unvisited:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_alternative:
                best_alternative = cost
        regret = best_alternative - immediate if best_alternative != float('inf') else 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Lower score is better; subtract regret and dest_dist to prefer high regret (costly to postpone)
        # and nodes far from destination (to handle them earlier).
        score = w * immediate - w * regret - w * dest_dist
        scores[i] = score

    # Deterministic tie-breaking: smallest index among minimal scores
    min_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(candidates)
