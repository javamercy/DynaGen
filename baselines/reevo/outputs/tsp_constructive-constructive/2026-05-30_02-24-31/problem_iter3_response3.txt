import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a composite score that combines:
    - immediate cost (distance from current to candidate)
    - regret look-ahead (cost difference of choosing this node as first step in a two-step plan)
    - destination distance (distance from candidate to the final destination)
    """
    # Weights: tuned for balanced performance
    w1, w2, w3 = 0.4, 0.35, 0.25

    # Precompute two-step costs and the best among them
    two_step_costs = {}
    best_two_step = float('inf')
    for n in unvisited_nodes:
        remaining = unvisited_nodes - {n}
        if remaining:
            min_next = min(distance_matrix[n][m] for m in remaining)
        else:
            min_next = 0  # no further step
        two_step = distance_matrix[current_node][n] + min_next
        two_step_costs[n] = two_step
        if two_step < best_two_step:
            best_two_step = two_step

    # Score each candidate
    scores = {}
    for n in unvisited_nodes:
        immediate = distance_matrix[current_node][n]
        regret = two_step_costs[n] - best_two_step
        dest_dist = distance_matrix[n][destination_node]
        score = w1 * immediate + w2 * regret + w3 * dest_dist
        # deterministic tie‑break by node index (lower index wins)
        scores[n] = (score, n)

    # Return the node with minimum score (ties broken by index)
    return min(scores, key=lambda x: scores[x])
