```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using an exact two-step look-ahead heuristic.

    For each candidate node u in unvisited_nodes:
        - immediate cost: distance from current_node to u
        - look-ahead cost: minimum over v in remaining of
          (distance[u][v] + distance[v][destination_node])
          where remaining = unvisited_nodes - {u}. If no remaining node exists,
          look-ahead is distance[u][destination_node].
        - total = immediate + look-ahead

    The candidate with the smallest total cost is chosen. Ties are broken by
    smaller node index. This heuristic is simple, avoids complex weighting or regret.

    Parameters
    ----------
    current_node : int
        The node we are currently at.
    destination_node : int
        The final node we must return to (e.g., the starting depot).
    unvisited_nodes : set of int
        The set of nodes not yet visited (must be non-empty).
    distance_matrix : np.ndarray
        Square matrix of pairwise distances.

    Returns
    -------
    int
        The selected next node to visit.
    """
    best_node = None
    best_total = float('inf')

    for u in unvisited_nodes:
        immediate = distance_matrix[current_node, u]
        remaining = unvisited_nodes - {u}

        if remaining:
            # Exact two-step look-ahead: pick the best v from remaining to reach destination
            look_ahead = min(
                distance_matrix[u, v] + distance_matrix[v, destination_node]
                for v in remaining
            )
        else:
            # Only one node left, go directly from u to destination
            look_ahead = distance_matrix[u, destination_node]

        total = immediate + look_ahead

        # Deterministic tie-breaking by smaller node index
        if total < best_total or (total == best_total and (best_node is None or u < best_node)):
            best_total = total
            best_node = u

    return best_node
```
