import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a regret-based look-ahead heuristic.
    
    For each candidate v, compute the regret of not visiting it first:
    regret(v) = min_{u != v} (dist(current,u) + dist(u,v)) - dist(current,v).
    The node with the highest regret is chosen, as it is most costly to postpone.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_regret = -np.inf
    best_node = None
    for v in unvisited_nodes:
        cost_now = distance_matrix[current_node][v]
        # Find the best alternative first node u (different from v)
        best_alternative = np.inf
        for u in unvisited_nodes:
            if u == v:
                continue
            alt = distance_matrix[current_node][u] + distance_matrix[u][v]
            if alt < best_alternative:
                best_alternative = alt
        regret = best_alternative - cost_now
        if regret > best_regret:
            best_regret = regret
            best_node = v
    return best_node
