import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray
) -> int:
    """
    Select the next node using a composite score that balances immediate cost,
    regret of postponement, look-ahead future connectivity, and distance to the
    final destination. Lower scores are preferred.

    Score = w1 * immediate - w2 * regret + w3 * look_ahead - w4 * dest_dist

    - immediate: distance from current_node to candidate
    - regret: best alternative two-step cost (via another unvisited node) minus immediate
      (high regret means it is costly to postpone the candidate)
    - look_ahead: minimum distance from the candidate to any other unvisited node
      or the destination (encourages good onward connections)
    - dest_dist: distance from candidate to the destination node (favors far-away nodes)

    Weights: w1=0.3, w2=0.3, w3=0.2, w4=0.2
    Tie-breaking: smallest index wins.
    """
    if len(unvisited_nodes) == 0:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = list(unvisited_nodes)
    w1, w2, w3, w4 = 0.35, 0.35, 0.15, 0.15

    scores = {}
    for u in unvisited:
        # immediate cost
        imm = distance_matrix[current_node, u]

        # regret: best alternative two-step path that visits u later
        best_alt = float('inf')
        for v in unvisited:
            if v == u:
                continue
            alt = distance_matrix[current_node, v] + distance_matrix[v, u]
            if alt < best_alt:
                best_alt = alt
        regret = best_alt - imm if best_alt != float('inf') else 0.0

        # look-ahead: best future edge from u
        look_candidates = [distance_matrix[u, v] for v in unvisited if v != u]
        if look_candidates:
            look_ahead = min(look_candidates)
        else:
            # only u remains; use distance to destination
            look_ahead = distance_matrix[u, destination_node]

        # distance to final destination
        dest_dist = distance_matrix[u, destination_node]

        # composite score (lower is better)
        score = w1 * imm - w2 * regret + w3 * look_ahead - w4 * dest_dist
        scores[u] = score

    # deterministic tie-breaking: smallest index among minimal score
    min_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(candidates)
