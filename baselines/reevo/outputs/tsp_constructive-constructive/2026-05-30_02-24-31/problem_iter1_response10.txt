import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a 2-step look‑ahead savings heuristic."""
    # If only one node left, we must choose it
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_combined_savings = -np.inf

    for node in unvisited_nodes:
        # Savings of inserting node between current and destination
        savings_direct = (distance_matrix[current_node, destination_node] -
                          distance_matrix[current_node, node] -
                          distance_matrix[node, destination_node])

        # Look ahead: find the best next node to insert after this node
        best_next_savings = -np.inf
        for next_node in unvisited_nodes:
            if next_node == node:
                continue
            savings_next = (distance_matrix[node, destination_node] -
                            distance_matrix[node, next_node] -
                            distance_matrix[next_node, destination_node])
            if savings_next > best_next_savings:
                best_next_savings = savings_next

        combined_savings = savings_direct + best_next_savings
        if combined_savings > best_combined_savings:
            best_combined_savings = combined_savings
            best_node = node

    return best_node
