import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int | None:
    """
    Improved constructive heuristic for TSP: combines immediate cost, regret,
    destination distance, and lookahead savings with tuned weights emphasising
    regret and future savings.

    Parameters
    ----------
    current_node : int
        The current node index.
    destination_node : int
        The final node to reach after visiting all nodes.
    unvisited_nodes : set of int
        Set of node indices not yet visited.
    distance_matrix : np.ndarray
        Square matrix of distances.

    Returns
    -------
    int or None
        The next node to visit, or None if unvisited_nodes is empty.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for reproducibility
    unvisited = sorted(unvisited_nodes)

    # Tuned weights: regret (w2) and lookahead savings (w4) are emphasized
    w1, w2, w3, w4 = 0.20, 0.40, 0.20, 0.20

    scores = {}
    for i in unvisited:
        # Immediate cost from current to candidate
        immediate = distance_matrix[current_node, i]

        # Regret: difference between best alternative (going to a different node first then to i)
        # and the direct immediate cost (which is the best if we go directly to i)
        best_alternative = float('inf')
        for j in unvisited:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_alternative:
                best_alternative = cost
        regret = best_alternative - immediate if best_alternative != float('inf') else 0.0

        # Distance to destination
        dest_dist = distance_matrix[i, destination_node]

        # Lookahead savings: how much cheaper it is to continue from i compared to staying at current
        # Positive savings mean the next step from i is cheaper than from current.
        best_next_from_i = min(distance_matrix[i, j] for j in unvisited if j != i)
        best_next_from_current = min(distance_matrix[current_node, j] for j in unvisited)
        lookahead_savings = best_next_from_current - best_next_from_i

        # Combined score (lower is better): high regret and high savings reduce the score
        score = w1 * immediate - w2 * regret - w3 * dest_dist - w4 * lookahead_savings
        scores[i] = score

    # Deterministic tie-breaking: smallest index among those with minimal score
    min_score = min(scores.values())
    candidates = [node for node, sc in scores.items() if sc == min_score]
    return min(candidates)
