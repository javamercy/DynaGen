import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Simplified selection using immediate cost, two-step regret, and destination distance.
    Removes future regret term for cleaner prioritization of regret.
    Weights: immediate 0.3, two-step regret 0.5, destination distance 0.2.
    Deterministic tie-breaking via smallest node index.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Deterministic ordering for reproducibility
    candidates = sorted(unvisited_nodes)

    # Fixed tuned weights – prioritizing regret
    w_imm = 0.3
    w_two = 0.5
    w_dest = 0.2

    best_score = float('inf')
    best_node = None

    for i in candidates:
        # Immediate distance from current node to candidate
        imm = distance_matrix[current_node, i]

        # Two-step regret: best indirect path via any other j minus direct edge
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret_two = (best_two_step - imm) if best_two_step != float('inf') else 0.0

        # Distance from candidate to final destination
        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better)
        score = w_imm * imm - w_two * regret_two + w_dest * dest_dist  # note: positive sign for dest_dist (higher distance is bad)

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
