```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a constructive heuristic that balances:
    - immediate distance from current node,
    - regret (penalty for missing the best outgoing edge from the candidate),
    - distance to the final destination.

    Lower scores are better. Deterministic tie-breaking by node index.
    """
    n = len(unvisited_nodes)
    if n == 0:
        raise ValueError("No unvisited nodes left")
    if n == 1:
        return next(iter(unvisited_nodes))

    # Work with sorted list for deterministic ordering
    candidates = sorted(unvisited_nodes)

    # Weights (emphasis on regret and destination proximity)
    w_imm = 0.20
    w_reg = 0.50
    w_dst = 0.30

    best_score = float('inf')
    best_node = candidates[0]

    for i in candidates:
        # Immediate cost from current node to candidate i
        immediate = distance_matrix[current_node, i]

        # Compute regret: difference between second-best and best outgoing edge from i
        # to other unvisited nodes. O(n) per candidate avoids sorting all distances.
        best_out = float('inf')
        second_best = float('inf')
        for j in candidates:
            if j == i:
                continue
            d = distance_matrix[i, j]
            if d < best_out:
                second_best = best_out
                best_out = d
            elif d < second_best:
                second_best = d
        if second_best == float('inf'):  # only one other node
            regret = 0.0
        else:
            regret = second_best - best_out

        # Distance from candidate to the final destination
        dest_dist = distance_matrix[i, destination_node]

        # Score: lower is better
        score = w_imm * immediate - w_reg * regret - w_dst * dest_dist

        # Deterministic tie-breaking: prefer smaller node index
        if score < best_score or (score == best_score and i < best_node):
            best_score = score
            best_node = i

    return best_node
```
