```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a combination of immediate cost, two-step regret (relative to direct edge),
    and destination distance. Prioritizes regret (two-step) and uses deterministic tie-breaking.
    Simpler than v1: no future regret component.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = sorted(unvisited_nodes)  # deterministic order

    # Weights for immediate cost, two-step regret, destination distance
    w_imm = 0.25
    w_two = 0.50  # prioritize regret
    w_dest = 0.25

    best_score = float('inf')
    best_node = None

    for i in candidates:
        imm = distance_matrix[current_node, i]

        # Two-step regret: best indirect path via any other j minus direct edge
        best_two_step = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two_step:
                best_two_step = cost
        regret_two = best_two_step - imm if best_two_step != float('inf') else 0.0

        dest_dist = distance_matrix[i, destination_node]

        # Combined score (lower is better)
        score = w_imm * imm - w_two * regret_two - w_dest * dest_dist

        # Deterministic tie-breaking: smaller node index wins
        if score < best_score or (np.isclose(score, best_score) and i < best_node):
            best_score = score
            best_node = i

    return best_node
```
