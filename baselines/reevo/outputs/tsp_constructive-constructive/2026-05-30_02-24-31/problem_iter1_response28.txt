import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using a regret-based look-ahead heuristic.
    For each candidate, the regret is defined as the difference between the second-best and best distance
    from that candidate to any other unvisited node. A high regret indicates that postponing the node would
    likely force a costly future connection. The score combines the direct distance from current_node and a
    penalty for the distance to destination_node, weighted against the regret.
    """
    alpha = 0.4        # weight for current_node distance
    beta = 0.5         # weight for regret (positive -> encourage high regret)
    gamma = 0.1        # small weight for distance to destination (future cost)

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        # distance from current to candidate
        dist_curr = distance_matrix[current_node][node]

        # distance from candidate to destination (future closing distance)
        dist_dest = distance_matrix[node][destination_node]

        # compute regret: need at least two other unvisited nodes
        other_unvisited = [n for n in unvisited_nodes if n != node]
        if len(other_unvisited) >= 2:
            dists = sorted([distance_matrix[node][n] for n in other_unvisited])
            regret = dists[1] - dists[0]
        else:
            regret = 0.0

        # score: lower is better (small distance, high regret, moderate destination distance)
        score = alpha * dist_curr - beta * regret + gamma * dist_dest
        if score < best_score:
            best_score = score
            best_node = node

    return best_node
