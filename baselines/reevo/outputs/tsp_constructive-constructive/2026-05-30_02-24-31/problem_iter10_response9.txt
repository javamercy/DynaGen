import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a balanced combination of:
        - immediate cost from current node to candidate,
        - regret (how much more expensive it is to postpone the candidate),
        - distance from candidate to the destination.

    All three components are weighted equally (1/3 each). Ties are broken by node index.

    Args:
        current_node: Current position.
        destination_node: Final destination (or return node).
        unvisited_nodes: Set of unvisited node indices.
        distance_matrix: Square matrix of pairwise distances.

    Returns:
        The next node to visit.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    candidates = list(unvisited_nodes)
    weight = 1.0 / 3.0
    best_score = float('inf')
    best_node = None

    for i in candidates:
        imm = distance_matrix[current_node, i]
        dest = distance_matrix[i, destination_node]

        # Compute regret: minimal cost of going current -> j -> i for some j != i
        best_alternative = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_alternative:
                best_alternative = cost
        regret = best_alternative - imm if best_alternative != float('inf') else 0.0

        # Lower score is better; subtract regret (prefer large regret) and dest (prefer far nodes)
        score = weight * imm - weight * regret - weight * dest

        # Deterministic tie‑breaking: smallest index wins
        if score < best_score or (score == best_score and i < best_node):
            best_score = score
            best_node = i

    return best_node
