import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select the next node to visit using a composite score that combines:
    - immediate cost (distance from current to candidate)
    - look-ahead (best connection from candidate to any other unvisited node or destination)
    - regret (difference between second-best and best connection, reflecting urgency)
    - proximity to destination (subtracted to favor nodes closer to destination)

    Weights are chosen to balance the contributions and normalize the score.
    Lower score is better.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_score = float('inf')

    for u in unvisited_nodes:
        d_cur_u = distance_matrix[current_node][u]

        # Build list of distances from u to all other candidates (including destination)
        neighbors = [distance_matrix[u][v] for v in unvisited_nodes if v != u]
        neighbors.append(distance_matrix[u][destination_node])

        # Sort to get best and second-best connection costs
        neighbors.sort()
        best_conn = neighbors[0]
        second_best_conn = neighbors[1]  # at least two entries (unvisited and destination)

        look_ahead = best_conn
        regret = second_best_conn - best_conn
        d_u_dest = distance_matrix[u][destination_node]

        # Composite score: lower is better
        score = (
            d_cur_u
            + 0.6 * look_ahead
            - 0.2 * d_u_dest
            - 0.2 * regret
        )

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
