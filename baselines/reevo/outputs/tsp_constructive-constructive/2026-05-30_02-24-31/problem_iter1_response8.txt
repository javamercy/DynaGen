import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node using a two‑step lookahead with regret.
    
    For each candidate i in unvisited_nodes, we compute the cost of
    visiting i now then its nearest remaining neighbour j:
        cost_now = d(current,i) + d(i,j)
    and the cost of visiting j first then i:
        cost_later = d(current,j) + d(j,i)
    The regret is cost_later - cost_now. The candidate with the
    largest regret is chosen.
    """
    # Trivial case: only one node left
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_regret = -np.inf
    best_node = None
    
    for i in unvisited_nodes:
        # Find nearest neighbour j to i among the other unvisited nodes
        others = unvisited_nodes - {i}
        # Compute distances from i to all others
        d_i_to_others = [distance_matrix[i, j] for j in others]
        j = list(others)[np.argmin(d_i_to_others)]  # nearest neighbour of i
        d_ij = distance_matrix[i, j]
        
        # Cost if we visit i now, then j
        cost_now = distance_matrix[current_node, i] + d_ij
        
        # Cost if we visit j now, then i (skip i first)
        cost_later = distance_matrix[current_node, j] + d_ij  # symmetric d(j,i) = d(i,j)
        
        regret = cost_later - cost_now
        
        if regret > best_regret:
            best_regret = regret
            best_node = i
    
    # Fallback: should never happen because unvisited_nodes is non‑empty
    if best_node is None:
        best_node = next(iter(unvisited_nodes))
    
    return best_node
