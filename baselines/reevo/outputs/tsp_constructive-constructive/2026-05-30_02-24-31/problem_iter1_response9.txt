```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Look-ahead based constructive heuristic for TSP.
    Scoring combines immediate cost, a one-step look-ahead cost, a regret term,
    and distance to destination.
    """
    alpha = 0.3   # weight for the minimum distance from candidate to remaining unvisited
    beta  = 0.5   # weight for the regret (second_min - min) from candidate to remaining unvisited
    gamma = 0.2   # weight for the distance from candidate to destination

    best_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        # Immediate cost from current to candidate
        imm_cost = distance_matrix[current_node][node]

        # Prepare list of other unvisited nodes (excluding candidate and current)
        others = [n for n in unvisited_nodes if n != node]
        n_others = len(others)

        # Look-ahead and regret based on distances from candidate to others
        if n_others > 0:
            dists_from_node = [distance_matrix[node][n] for n in others]
            dists_from_node.sort()
            min_dist = dists_from_node[0]
            second_min_dist = dists_from_node[1] if n_others > 1 else min_dist
            regret = second_min_dist - min_dist
        else:
            min_dist = 0.0
            regret = 0.0

        # Distance from candidate to destination
        dest_cost = distance_matrix[node][destination_node]

        # Combined score (smaller is better)
        score = imm_cost + alpha * min_dist - beta * regret + gamma * dest_cost

        if score < best_score:
            best_score = score
            best_node = node

    return best_node
```
