import numpy as np

def select_next_node_v2(current_node: int, destination_node: int,
                        unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Improved heuristic combining immediate cost, regret, distance to destination,
    and lookahead savings. Weights are tuned to balance these factors.
    """
    if not unvisited_nodes:
        return None
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    unvisited = list(unvisited_nodes)
    # Weights: immediate cost, regret, destination distance, lookahead savings
    w_imm, w_reg, w_dest, w_sav = 0.20, 0.35, 0.20, 0.25

    best_score = float('inf')
    best_node = None

    for u in unvisited:
        # Immediate cost from current node to candidate u
        imm = distance_matrix[current_node, u]

        # Regret: difference between best two‑step alternative and immediate cost
        best_alternative = float('inf')
        for j in unvisited:
            if j == u:
                continue
            alt = distance_matrix[current_node, j] + distance_matrix[j, u]
            if alt < best_alternative:
                best_alternative = alt
        regret = best_alternative - imm if best_alternative != float('inf') else 0.0

        # Distance from candidate u to final destination
        dest_dist = distance_matrix[u, destination_node]

        # Lookahead savings: best edge from current to any unvisited minus
        # best edge from candidate u to any other unvisited
        best_from_current = min(distance_matrix[current_node, j] for j in unvisited)
        best_from_u = min(distance_matrix[u, j] for j in unvisited if j != u)
        lookahead_savings = best_from_current - best_from_u

        # Composite score: lower is better.
        # Subtract regret and lookahead_savings to reward high regret and positive savings.
        score = (w_imm * imm) - (w_reg * regret) + (w_dest * dest_dist) - (w_sav * lookahead_savings)

        # Deterministic tie‑breaking: prefer smaller node index
        if score < best_score - 1e-12:
            best_score = score
            best_node = u
        elif abs(score - best_score) < 1e-12 and u < best_node:
            best_node = u

    return best_node
