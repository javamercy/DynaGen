```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Mutated version that emphasizes two-step regret and adds worst-case future look-ahead.
    Weights: immediate 0.1, two-step regret 0.5, future regret (best vs second) 0.2, destination 0.2.
    Also normalizes components by max distance for scale invariance and adds a small penalty
    for high worst-case future edges.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    candidates = sorted(unvisited_nodes)
    max_dist = np.max(distance_matrix)
    
    w_imm, w_two, w_future, w_dest = 0.1, 0.5, 0.2, 0.2
    w_worst = 0.1
    
    best_score = float('inf')
    best_node = None
    
    for i in candidates:
        # Immediate cost (normalized)
        imm = distance_matrix[current_node, i] / max_dist
        
        # Two-step lookahead regret: best indirect cost via j vs direct (normalized)
        best_two = float('inf')
        for j in candidates:
            if j == i:
                continue
            cost = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost < best_two:
                best_two = cost
        regret_two = (best_two - distance_matrix[current_node, i]) / max_dist if best_two != float('inf') else 0.0
        
        # Future regret: difference between second-best and best edge from i to other unvisited nodes (normalized)
        future_costs = [distance_matrix[i, v] for v in unvisited_nodes if v != i]
        future_costs.sort()
        if len(future_costs) >= 2:
            regret_future = (future_costs[1] - future_costs[0]) / max_dist
        else:
            regret_future = 0.0
        
        # Distance to final destination (normalized)
        dest_dist = distance_matrix[i, destination_node] / max_dist
        
        # Additional look-ahead: worst-case future edge (largest distance from i to any other unvisited node)
        worst_future = max(future_costs) / max_dist if future_costs else 0.0
        
        # Combined score (lower is better); regret and destination contribute negatively,
        # worst_future penalizes high maximum future edge.
        score = (w_imm * imm 
                 - w_two * regret_two 
                 - w_future * regret_future 
                 - w_dest * dest_dist 
                 + w_worst * worst_future)
        
        # Deterministic tie-breaking (smaller index wins)
        if score < best_score - 1e-12:
            best_score = score
            best_node = i
        elif abs(score - best_score) <= 1e-12 and i < best_node:
            best_score = score
            best_node = i
    
    return best_node
```
