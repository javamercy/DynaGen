import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using a 2-step look-ahead that estimates the cost to 
    reach the destination via the best immediate successor."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_node = None
    best_score = float('inf')
    
    # Precompute distances to destination for efficiency (optional, but fine)
    dest_dist = distance_matrix[:, destination_node]
    
    for u in unvisited_nodes:
        # Immediate cost
        immediate = distance_matrix[current_node][u]
        
        # Look-ahead: best path u -> v -> ... -> destination
        remaining = unvisited_nodes - {u}
        if remaining:
            # For each possible next node v, consider the cost (u->v) + weighted (v->dest)
            best_future = min(
                distance_matrix[u][v] + 0.5 * dest_dist[v] 
                for v in remaining
            )
        else:
            # Only one node left: must go directly to destination after u
            best_future = dest_dist[u]
        
        # Score combines immediate cost, look-ahead, and a small direct penalty
        score = immediate + best_future + 0.2 * dest_dist[u]
        
        if score < best_score:
            best_score = score
            best_node = u
            
    return best_node
