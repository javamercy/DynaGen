```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a look-ahead regret-based heuristic.
    
    For each candidate, the score combines:
    - immediate distance from current node,
    - average distance to other unvisited nodes,
    - distance to destination,
    - a regret term that estimates the extra cost if this node is deferred.
    
    Lower score is better.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    # Parameters (tunable)
    alpha = 0.4
    beta  = 0.2
    gamma = 0.2
    delta = 0.2
    
    best_score = float('inf')
    best_node = None
    
    for c in unvisited_nodes:
        immediate = distance_matrix[current_node][c]
        dest_dist = distance_matrix[c][destination_node]
        
        # Distances from c to all other unvisited nodes
        others = [node for node in unvisited_nodes if node != c]
        if others:
            dists_to_others = np.array([distance_matrix[c][o] for o in others])
            avg_dist = np.mean(dists_to_others)
            
            # Regret: extra cost if we skip c now and visit another node j first,
            # then later go to c from j.
            # future_best = min_{j in others} [ dist(current, j) + dist(j, c) ]
            future_best = min(
                distance_matrix[current_node][j] + distance_matrix[j][c]
                for j in others
            )
            regret = future_best - immediate  # positive -> deferring costs more
        else:
            avg_dist = 0.0
            regret = 0.0
        
        score = (alpha * immediate +
                 beta  * avg_dist +
                 gamma * dest_dist -
                 delta * regret)   # high regret makes score lower (better)
        
        if score < best_score:
            best_score = score
            best_node = c
    
    return best_node
```
