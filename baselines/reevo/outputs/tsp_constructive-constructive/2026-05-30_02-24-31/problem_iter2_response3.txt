import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    # If only one node remains, it must be chosen
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_score = float('inf')
    best_node = None
    # Weights for the three components (immediate cost, look-ahead, destination distance)
    w1 = 1.0   # weight for immediate cost from current node to candidate
    w2 = 0.7   # weight for look-ahead cost (minimum distance to next unvisited node)
    w3 = 0.3   # weight for proximity to destination (subtracted, since closer is better)

    for u in unvisited_nodes:
        d_curr_u = distance_matrix[current_node][u]

        # Look-ahead: best distance from u to any other unvisited node
        remaining = unvisited_nodes - {u}
        if remaining:
            look_ahead = min(distance_matrix[u][v] for v in remaining)
        else:
            # Only candidate node left (should not happen due to early check), fallback to destination distance
            look_ahead = distance_matrix[u][destination_node]

        d_u_dest = distance_matrix[u][destination_node]

        # Composite score: lower is better
        score = w1 * d_curr_u + w2 * look_ahead - w3 * d_u_dest

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
