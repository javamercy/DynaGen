import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select next node using look-ahead (minimum distance to the subsequent node)."""
    # If only one node remains, it must be chosen
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_score = float('inf')
    best_node = None
    alpha = 0.6  # weight for look-ahead distance
    beta = 0.4   # weight for proximity to destination

    for u in unvisited_nodes:
        d_current_u = distance_matrix[current_node][u]

        # Look-ahead: best distance from u to any other unvisited node
        remaining = unvisited_nodes - {u}
        if remaining:
            min_next = min(distance_matrix[u][v] for v in remaining)
            look_ahead = min_next
        else:
            # Should not happen due to early exit, but handle gracefully
            look_ahead = distance_matrix[u][destination_node]

        # Distance from u to destination (to encourage progress toward goal)
        d_u_dest = distance_matrix[u][destination_node]

        # Composite score: lower is better
        score = d_current_u + alpha * look_ahead - beta * d_u_dest

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
