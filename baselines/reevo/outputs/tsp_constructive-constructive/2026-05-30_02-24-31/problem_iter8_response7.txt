```python
import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select next node using immediate cost and regret (difference between best alternative and direct cost),
    plus a lookahead (best distance from candidate to another unvisited node).
    High regret suggests it is costly to delay visiting this node, so we favor it.
    Lower score is better. Deterministic tie-breaking by smallest node index.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    unvisited = sorted(unvisited_nodes)
    best_score = float('inf')
    best_node = None
    
    # Weights for immediate cost, regret (subtracted), lookahead (added)
    w_imm, w_reg, w_look = 0.3, 0.4, 0.3
    
    for i in unvisited:
        immediate = distance_matrix[current_node, i]
        
        # Regret: best cost to reach i via another node (j != i)
        best_alternative = float('inf')
        best_next_from_i = float('inf')
        for j in unvisited:
            if j == i:
                continue
            cost_via_j = distance_matrix[current_node, j] + distance_matrix[j, i]
            if cost_via_j < best_alternative:
                best_alternative = cost_via_j
            # Lookahead: best edge from i to j
            d = distance_matrix[i, j]
            if d < best_next_from_i:
                best_next_from_i = d
        
        regret = best_alternative - immediate if best_alternative != float('inf') else 0.0
        lookahead = best_next_from_i if best_next_from_i != float('inf') else 0.0
        
        # Score: lower is better. High regret reduces score (good), low lookahead reduces score (good)
        score = w_imm * immediate - w_reg * regret + w_look * lookahead
        
        if score < best_score or (np.isclose(score, best_score) and (best_node is None or i < best_node)):
            best_score = score
            best_node = i
    
    return best_node
```
