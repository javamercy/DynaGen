import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using immediate cost, regret-based look-ahead,
    and destination proximity. Weights are tuned to prioritise regret (look-ahead
    advantage) while keeping a moderate contribution from immediate cost and
    destination distance.
    """
    # Trivial case: only one unvisited node remains
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Tuned weights (regret is given highest weight)
    w_imm = 0.2
    w_reg = 0.5
    w_dest = 0.1

    best_score = float('inf')
    best_node = None

    for u in unvisited_nodes:
        # Immediate cost from current node
        imm = distance_matrix[current_node][u]

        # Regret look-ahead: difference between the two best future edges from u
        future_costs = [distance_matrix[u][v] for v in unvisited_nodes if v != u]
        if len(future_costs) >= 2:
            sorted_future = sorted(future_costs)
            regret = sorted_future[1] - sorted_future[0]  # second best - best
        else:
            regret = 0.0

        # Distance to the final destination
        dest_dist = distance_matrix[u][destination_node]

        # Composite score (lower is better)
        score = w_imm * imm - w_reg * regret + w_dest * dest_dist

        # Deterministic tie‑breaking: prefer smaller node index
        if score < best_score or (np.isclose(score, best_score) and u < best_node):
            best_score = score
            best_node = u

    return best_node
