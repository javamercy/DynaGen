import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """Select the next node with a 2-step look-ahead that accounts for the best subsequent move."""
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))
    
    best_score = float('inf')
    best_node = None
    alpha = 0.5   # weight for the look-ahead distance
    beta = 0.2    # weight for the distance to the destination
    
    for node in unvisited_nodes:
        # immediate distance from current node
        d_curr = distance_matrix[current_node][node]
        
        # look-ahead: distance to the nearest unvisited node (excluding current and node)
        remaining = unvisited_nodes - {node}
        if remaining:
            d_lookahead = min(distance_matrix[node][j] for j in remaining)
        else:
            d_lookahead = 0.0
        
        # distance to the final destination
        d_dest = distance_matrix[node][destination_node]
        
        # combined score (lower is better)
        score = d_curr + alpha * d_lookahead + beta * d_dest
        
        if score < best_score:
            best_score = score
            best_node = node
    
    return best_node
