import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select the next node using:
    - immediate distance from current_node
    - look-ahead: best future edge from the candidate node (to any other unvisited node or destination)
    - proximity to the final destination (subtracted, so nodes closer to destination are favoured)

    Score = d(current, u) + alpha * look_ahead - gamma * d(u, dest)   (lower is better)

    Parameters:
        current_node (int): index of the current node
        destination_node (int): index of the final destination (start node)
        unvisited_nodes (set): set of node indices not yet visited
        distance_matrix (np.ndarray): square symmetric distance matrix

    Returns:
        int: index of the selected next node
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Tuned weights
    alpha = 0.5
    gamma = 0.3

    best_node = None
    best_score = float('inf')

    for u in unvisited_nodes:
        d_cur_u = distance_matrix[current_node][u]

        # Look-ahead: best future connection from u
        # Consider all other unvisited nodes and the destination
        candidates = [distance_matrix[u][v] for v in unvisited_nodes if v != u]
        candidates.append(distance_matrix[u][destination_node])
        look_ahead = min(candidates)

        d_u_dest = distance_matrix[u][destination_node]

        # Composite score (lower = better)
        score = d_cur_u + alpha * look_ahead - gamma * d_u_dest

        if score < best_score:
            best_score = score
            best_node = u
        # Deterministic tie-breaking (prefer smaller index)
        elif np.isclose(score, best_score) and (best_node is None or u < best_node):
            best_node = u

    return best_node
