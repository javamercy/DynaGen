import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node to visit using a look-ahead mechanism.
    For each candidate, simulate the rest of the tour via nearest neighbor
    and choose the candidate that minimizes the total estimated tour length.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    best_node = None
    best_total = float('inf')

    for candidate in unvisited_nodes:
        # Simulate starting with current -> candidate
        remaining = set(unvisited_nodes)
        remaining.remove(candidate)
        current = candidate
        total = distance_matrix[current_node][current]

        # Nearest neighbor completion
        while remaining:
            # Find nearest unvisited node
            nearest = None
            min_dist = float('inf')
            for node in remaining:
                d = distance_matrix[current][node]
                if d < min_dist:
                    min_dist = d
                    nearest = node
            total += min_dist
            current = nearest
            remaining.remove(nearest)

        # Add return to destination
        total += distance_matrix[current][destination_node]

        if total < best_total:
            best_total = total
            best_node = candidate

    return best_node
