import numpy as np

def select_next_node_v2(current_node: int, destination_node: int, unvisited_nodes: set, distance_matrix: np.ndarray) -> int:
    """
    Select the next node using a look-ahead heuristic that balances immediate cost,
    the cheapest onward connection, and proximity to the destination.
    """
    if not unvisited_nodes:
        return None

    # Weights for the three components (tuned for typical TSP instances)
    alpha = 0.5   # weight for immediate edge cost
    beta  = 0.3   # weight for cheapest onward edge (look-ahead)
    gamma = 0.2   # weight for distance to destination

    best_node = None
    best_score = float('inf')

    for candidate in unvisited_nodes:
        # Immediate cost from current to candidate
        immediate = distance_matrix[current_node][candidate]

        # Look-ahead: cheapest edge from candidate to any other unvisited node
        # Exclude candidate itself and the destination? Keep destination in set for potential.
        other_unvisited = unvisited_nodes - {candidate}
        if other_unvisited:
            min_future = min(distance_matrix[candidate][j] for j in other_unvisited)
        else:
            min_future = 0.0  # no future step

        # Distance from candidate to destination (encourage closeness to goal)
        dest_distance = distance_matrix[candidate][destination_node]

        # Combined score (lower is better)
        score = alpha * immediate + beta * min_future + gamma * dest_distance

        if score < best_score:
            best_score = score
            best_node = candidate

    return best_node
