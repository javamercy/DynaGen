import numpy as np

def select_next_node_v2(
    current_node: int,
    destination_node: int,
    unvisited_nodes: set,
    distance_matrix: np.ndarray,
) -> int:
    """
    Select next node using a composite score that combines:
    - immediate distance from current node
    - look‑ahead (best connection to any other unvisited node or destination)
    - regret (difference between best and second‑best connection)
    - proximity to the final destination

    The score is: d(current, u) + alpha * look_ahead - beta * regret - gamma * d(u, dest)
    Lower score is better.
    """
    if len(unvisited_nodes) == 1:
        return next(iter(unvisited_nodes))

    # Weights – tuned for general TSP instances
    alpha = 0.5
    beta  = 0.4
    gamma = 0.3

    best_node = None
    best_score = float('inf')

    for u in unvisited_nodes:
        d_cur_u = distance_matrix[current_node][u]

        # Build list of distances from u to all other candidate next nodes
        # (the other unvisited nodes and the destination)
        candidates = [distance_matrix[u][d] for d in unvisited_nodes if d != u]
        candidates.append(distance_matrix[u][destination_node])

        # Sort to get best and second‑best connection costs
        candidates.sort()
        best_conn   = candidates[0]
        second_best = candidates[1]  # there are at least two entries

        look_ahead = best_conn
        regret     = second_best - best_conn
        d_u_dest   = distance_matrix[u][destination_node]

        # Composite score (lower = better)
        score = d_cur_u + alpha * look_ahead - beta * regret - gamma * d_u_dest

        if score < best_score:
            best_score = score
            best_node = u

    return best_node
