        times = np.zeros(n_trucks)   # elapsed travel time per truck

            assigned[c_idx] = True

            pos[t_idx] = available_customers[c_idx]

        while not np.all(assigned):

                    times[t_idx] += best_dist
                    pos[t_idx] = available_customers[best_c]
