            return 0
        return sum(dist[route[i]][route[i+1]] for i in range(len(route)-1))

    def objective(routes):
        return max(route_distance(r) for r in routes)

                            prev = node
                        new_dist += dist[prev][route[k]]
                        prev = route[k]
