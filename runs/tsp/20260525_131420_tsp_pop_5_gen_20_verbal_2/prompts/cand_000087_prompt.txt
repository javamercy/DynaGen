    def build_pos(tour):
        pos = np.empty(n, dtype=int)
        for idx, city in enumerate(tour):
            pos[city] = idx
        return pos

    def nearest_neighbor(start=None):
        if start is None:
            start = np.random.randint(n)
        tour = [start]
        visited = {start}
        cur = start
        while len(tour) < n:
            # find nearest unvisited neighbor
            best = -1
            best_dist = float('inf')
            for j in range(n):
                if j not in visited:
                    d = distance_matrix[cur, j]
                    if d < best_dist:
                        best_dist = d
                        best = j
            tour.append(best)
            visited.add(best)
            cur = best
        return np.array(tour, dtype=np.int32)

    def double_bridge(tour):
        # standard double-bridge: split into 4 segments and reorder
        n = len(tour)
        # pick three distinct cut points
        a = np.random.randint(1, n//3)
        b = a + np.random.randint(1, n//3)
        c = b + np.random.randint(1, n//3)
