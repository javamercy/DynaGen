        return np.array([0, 1])

    np.random.seed(seed)

            curr = next_node

    def fast_2opt(tour, current_dist, local_budget):
        t = tour.copy()
        d = current_dist

        for i in range(n): pos_map[t[i]] = i

        improved = True

            improved = False
            for i in range(n):

                v = t[(i + 1) % n]
