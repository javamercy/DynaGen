                        prev = tour[-1]
                        nxt = tour[0]

                        break

                    b = (i+1) % n
                    c = j

def total_distance(tour, dist):
    n = len(tour)
    return sum(dist[tour[i], tour[(i+1)%n]] for i in range(n))
