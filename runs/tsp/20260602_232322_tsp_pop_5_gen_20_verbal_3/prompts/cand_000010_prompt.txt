        for j in range(i+1, n):
            d = distance_matrix[i, j]
            if d > max_dist:
                max_dist = d
                start, end = i, j
    tour = [start, end]
    remaining = set(range(n)) - {start, end}
    while remaining:
        # find farthest node from current tour
        farthest_dist = -1
        farthest_node = None

                farthest_node = node
