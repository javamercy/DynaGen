def solve_tsp(distance_matrix: np.ndarray) -> np.ndarray:
    n = len(distance_matrix)
    if n == 1:
        return np.array([0])
    if n == 2:
        return np.array([0, 1])

    max_dist = -1
    start_pair = (0, 1)
    for i in range(n):
        for j in range(i+1, n):
            if distance_matrix[i][j] > max_dist:
                max_dist = distance_matrix[i][j]
                start_pair = (i, j)
    tour = [start_pair[0], start_pair[1]]

    while unvisited:
        best_node = None
        best_increase = -1
        best_pos = None

                if inc < min_increase:
                    min_increase = inc
                    min_pos = j

                best_increase = min_increase
                best_node = node
                best_pos = min_pos

def _tour_length(dm, tour):
    n = len(tour)
    total = 0.0
    for k in range(n):
        total += dm[tour[k]][tour[(k+1) % n]]
    return total
