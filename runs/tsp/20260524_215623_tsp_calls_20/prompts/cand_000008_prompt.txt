    tour = list(max_idx)
    unvisited = set(range(n)) - set(tour)
    used_budget = 0
    while unvisited and used_budget < budget:
        best_node = None
        best_inc = -1.0
        best_pos = 0
        for node in unvisited:
            min_inc = np.inf
            ins_pos = 0

                best_inc = min_inc

                best_pos = ins_pos
