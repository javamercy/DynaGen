import numpy as np
import random

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        np.random.seed(seed)
        random.seed(seed)

    def __call__(self, func):
        budget = self.budget
        dim = self.dim
        lb = func.bounds.lb
        ub = func.bounds.ub

        # Handle tiny budget: just evaluate one point
        if budget < 1:
            best_x = np.full(dim, np.nan)
            best_f = np.inf
            return best_f, best_x

        # Population size
        pop_size = min(5 * dim, max(10, budget // 4))
        if pop_size < 1:
            pop_size = 1

        # Initialize population
        pop = np.random.uniform(lb, ub, (pop_size, dim))
        pop_f = np.full(pop_size, np.inf)
        best_x = None
        best_f = np.inf
        fcalls = 0

        # Evaluate initial population
        for i in range(pop_size):
            if fcalls >= budget:
                break
            x = np.clip(pop[i], lb, ub)
            val = func(x)
            fcalls += 1
            pop_f[i] = val
            if val < best_f:
                best_f = val
                best_x = x.copy()
                report_best(best_f, best_x)

        # Fallback if no evaluation happened (should not happen)
        if best_x is None:
            best_x = np.clip(np.random.uniform(lb, ub), lb, ub)
            best_f = func(best_x)
            fcalls += 1
            report_best(best_f, best_x)

        # DE parameters
        F = 0.8
        CR = 0.9

        # Main loop
        while fcalls < budget:
            for i in range(pop_size):
                if fcalls >= budget:
                    break

                # Select distinct random indices for mutation
                candidates = list(range(pop_size))
                candidates.remove(i)
                if len(candidates) >= 2:
                    r1, r2 = random.sample(candidates, 2)
                else:
                    # If pop_size==1, use i itself (but then mutation degenerate)
                    r1 = r2 = i

                # Mutation: current-to-best/1
                mutant = pop[i] + F * (best_x - pop[i]) + F * (pop[r1] - pop[r2])
                mutant = np.clip(mutant, lb, ub)

                # Binomial crossover
                trial = pop[i].copy()
                j_rand = random.randint(0, dim-1)
                for j in range(dim):
                    if random.random() < CR or j == j_rand:
                        trial[j] = mutant[j]

                # Evaluate trial
                val = func(trial)
                fcalls += 1

                # Selection
                if val < pop_f[i]:
                    pop[i] = trial
                    pop_f[i] = val
                    if val < best_f:
                        best_f = val
                        best_x = trial.copy()
                        report_best(best_f, best_x)

        return best_f, best_x
