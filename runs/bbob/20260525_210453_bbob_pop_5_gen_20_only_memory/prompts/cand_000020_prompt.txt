class Optimizer:

        self.budget = budget
        self.dim = dim
        self.seed = seed
        np.random.seed(seed)

    def __call__(self, func):

        dim = self.dim
        lb = func.bounds.lb
        ub = func.bounds.ub

        pop_size = min(4 * dim, budget)
        if pop_size < 4:
            best_x = None
            best_f = np.inf
            for _ in range(budget):
                x = np.random.uniform(lb, ub, dim)
                val = func(x)
                if val < best_f:
                    best_f = val
                    best_x = x.copy()
                    report_best(best_f, best_x)
            return best_f, best_x

        pop = np.random.uniform(lb, ub, (pop_size, dim))
        pop_f = np.full(pop_size, np.inf)
        best_x = None
        best_f = np.inf
        fcalls = 0

        for i in range(pop_size):
            x = np.clip(pop[i], lb, ub)
            val = func(x)
            fcalls += 1
            pop_f[i] = val
            if val < best_f:
                best_f = val
                best_x = x.copy()
                report_best(best_f, best_x)

        while fcalls < budget:
