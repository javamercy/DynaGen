import random

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):

        dim = self.dim

        best_f = np.inf
        fcalls = 0
        for i in range(pop_size):
            if fcalls >= budget:
                break
            x = np.clip(pop[i], lb, ub)
            val = func(x)
            fcalls += 1
            pop_f[i] = val
            if val < best_f:
                best_f = val
                best_x = x.copy()
                report_best(best_f, best_x)
        F = 0.8
        CR = 0.9
        while fcalls < budget:

                if fcalls >= budget:
                    break

                candidates.remove(i)
                if random.random() < 0.5:
                    # DE/best/1
                    r1, r2 = random.sample(candidates, 2)
                    mutant = best_x + F * (pop[r1] - pop[r2])
