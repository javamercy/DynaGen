class Optimizer:

        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

        ub = func.bounds.ub
        dim = self.dim
        budget = self.budget
        rng = self.rng

        best_f = func(best_x)

        report_best(best_f, best_x)

        step = 0.2 * (ub - lb)
        stagnation = 0
        max_stag = max(1, budget // 10)

        while evals < budget:
            success = False
            perm = rng.permutation(dim)
            for i in perm:
                if evals >= budget:
                    break

                trial = best_x.copy()
                trial[i] = np.clip(best_x[i] + step[i], lb[i], ub[i])
                f = func(trial)
                evals += 1
                if f < best_f:
                    best_f = f
                    best_x = trial
                    report_best(best_f, best_x)
                    step[i] = min(step[i] * 2, ub[i] - lb[i])
                    success = True
                    stagnation = 0
                    break
                # Negative direction

                trial[i] = np.clip(best_x[i] - step[i], lb[i], ub[i])
                f = func(trial)
                evals += 1
                if f < best_f:
                    best_f = f
                    best_x = trial
                    report_best(best_f, best_x)
                    step[i] = min(step[i] * 2, ub[i] - lb[i])
                    success = True

            if not success and evals < budget:

                norm = np.linalg.norm(direction)
                if norm > 0:
                    direction = direction / norm
                trial = np.clip(best_x + step * direction, lb, ub)
                f = func(trial)
                evals += 1

                    best_f = f
                    best_x = trial
                    report_best(best_f, best_x)
                    step = np.minimum(step * 2, ub - lb)
