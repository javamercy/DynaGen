        ub = func.bounds.ub
        dim = self.dim
        budget = self.budget
        rng = np.random.RandomState(self.seed)

        # population size
        pop_size = min(10 * dim, budget // 3)
        pop_size = max(pop_size, 5)

        best_val = np.inf
        best_x = None
        evals = 0

        intervals = np.linspace(0, 1, pop_size + 1)
        points = np.zeros((pop_size, dim))
        for d in range(dim):
            # choose one random point in each interval, then permute
            samples = rng.uniform(intervals[:-1], intervals[1:], size=pop_size)
            points[:, d] = samples[rng.permutation(pop_size)]
        # scale to bounds
        pop = lb + points * (ub - lb)
        pop_vals = np.full(pop_size, np.inf)
        for i in range(pop_size):
            if evals >= budget:
                break
            x = pop[i]
            val = func(x)
            evals += 1
            pop_vals[i] = val
            if val < best_val:
                best_val = val
                best_x = x.copy()
                report_best(best_val, best_x)

        if evals >= budget:
            return best_val, best_x

        # DE parameters
        F = 0.5
        CR = 0.9
        max_gen = max(1, int(0.8 * budget / pop_size))
        for gen in range(max_gen):

                if evals >= budget:
                    break

                val = func(trial)
                evals += 1
                if val < pop_vals[i]:
                    pop[i] = trial
                    pop_vals[i] = val
                    if val < best_val:
                        best_val = val
                        best_x = trial.copy()
