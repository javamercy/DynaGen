        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

        ub = func.bounds.ub
        dim = self.dim

        rng = self.rng

            de_evals += 1
            pop_fit[i] = val

                best_val = val
                best_x = pop[i].copy()
                report_best(best_val, best_x)
