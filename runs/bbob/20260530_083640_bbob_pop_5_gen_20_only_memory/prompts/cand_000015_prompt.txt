        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

        lb = func.bounds.lb
        ub = func.bounds.ub
        dim = self.dim
        budget = self.budget
        rng = self.rng

        # Initial random point
        best_x = rng.uniform(lb, ub)
        best_val = func(best_x)
        calls = 1
        report_best(best_val, best_x)

        # Current point
        current_x = best_x.copy()
        current_val = best_val

        # Temperature and step size schedule
        max_iter = budget - 1
        if max_iter <= 0:
            return best_val, best_x
        T0 = 1.0
        T_end = 1e-4
        step0 = 0.1 * (ub - lb)
        step_end = 1e-6 * (ub - lb)

        for i in range(max_iter):
            t = i / max_iter
            T = T0 * (T_end / T0) ** t
            step = step0 * (step_end / step0) ** t

            # Generate perturbation
            candidate = current_x + step * rng.randn(dim)
            candidate = np.clip(candidate, lb, ub)
            val = func(candidate)
            calls += 1

                current_x = candidate
                current_val = val
                if val < best_val:
                    best_val = val
                    best_x = candidate.copy()
                    report_best(best_val, best_x)
            else:
                if rng.rand() < np.exp(-delta / T):
                    current_x = candidate
                    current_val = val
