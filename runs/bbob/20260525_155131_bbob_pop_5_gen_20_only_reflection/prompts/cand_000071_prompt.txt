class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        np.random.seed(seed)
        self.NP = max(4, min(int(budget/2), 10*dim))
        self.CR = 0.9

        self.ub = None
        self.best_x = None
        self.best_val = float('inf')
        self.calls = 0

                if val < fitness[i]:

                    fitness[i] = val

                        self.best_x = trial.copy()

                        improved_this_gen = True
            generation += 1
