            self.calls += 1
            fitness[i] = val

        CR = np.random.uniform(0, 1, self.NP)
        generation = 0
        while self.calls < self.budget:
            tau_F = self.tau_F_init - (self.tau_F_init - 0.01) * (self.calls / self.budget)

            F_l = self.F_l_init + (self.F_l_final - self.F_l_init) * (self.calls / self.budget)
