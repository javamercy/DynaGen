class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.pop_size = max(4, min(20, budget // 20))
        if self.pop_size > budget:
            self.pop_size = budget
        self.F = 0.8
        self.CR = 0.9
        self.rng = np.random.RandomState(seed)
        self.best_val = float('inf')
        self.best_x = None

    def __call__(self, func):

        ub = func.bounds.ub
        evals = 0

                break
            pop_fit[i] = func(pop[i])
            evals += 1
            if pop_fit[i] < self.best_val:
                self.best_val = pop_fit[i]
                self.best_x = pop[i].copy()
                report_best(self.best_val, self.best_x)

        de_budget = int(0.8 * self.budget)
        while evals < min(self.budget, de_budget):
            fraction = evals / max(1, de_budget)
            F = 0.9 - fraction * 0.7
            CR = 0.9 - fraction * 0.4
            new_pop = pop.copy()
            new_fit = pop_fit.copy()
