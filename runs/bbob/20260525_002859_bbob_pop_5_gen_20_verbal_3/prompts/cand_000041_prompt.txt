class Optimizer:

        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.best_value = None
        self.best_x = None

        lb = func.bounds.lb
        ub = func.bounds.ub

            evals += 1
            if evals == 1 or pop_fitness[i] < self.best_value:
                self.best_value = pop_fitness[i]
                self.best_x = pop[i].copy()

        if evals >= budget:
            return self.best_value, self.best_x

        CR = 0.9 * np.ones(popsize)
        tau1 = 0.1
        tau2 = 0.1
        F_l = 0.1
        F_u = 0.9
        while evals < budget:

                # adapt F and CR
                if rng.rand() < tau1:
                    F[i] = F_l + rng.rand() * (F_u - F_l)
                if rng.rand() < tau2:
                    CR[i] = rng.rand()
