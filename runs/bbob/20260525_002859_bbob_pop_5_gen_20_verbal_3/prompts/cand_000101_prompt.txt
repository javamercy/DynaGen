        self.dim = dim
        self.seed = seed
        self.best_value = float('inf')

            return self.best_value, self.best_x

        CR_memory = np.ones(H) * 0.5
        memory_counter = 0

            CR_success = []
            for i in range(popsize):
                # generate F and CR for this individual

                F = F_memory[r] + 0.1 * rng.randn()
                CR = CR_memory[r] + 0.1 * rng.randn()
