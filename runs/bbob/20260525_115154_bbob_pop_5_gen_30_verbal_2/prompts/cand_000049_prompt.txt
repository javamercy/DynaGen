        self.dim = dim

        self.rng = RandomState(seed)

        budget = self.budget
        rng = self.rng

        ub = func.bounds.ub

                val = func(x)
                evals += 1
                if val < best_val:
                    best_val = val
                    best_x = x.copy()

            return best_val, best_x

        pop = rng.uniform(lb, ub, size=(pop_size, dim))

        best_x = None

                break
            fitness[i] = func(pop[i])
            evals += 1

                report_best(best_val, best_x)

        # DE parameters

        CR = 0.9
        stagnation_limit = 20
