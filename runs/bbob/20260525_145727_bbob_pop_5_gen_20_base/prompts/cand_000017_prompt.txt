class Optimizer:

        self.dim = dim
        self.seed = seed

        self.CR = 0.9

    def __call__(self, func):

        lb = func.bounds.lb
        ub = func.bounds.ub
        pop_size = self.pop_size
        F = self.F
        CR = self.CR

        pop = lb + rng.rand(pop_size, dim) * (ub - lb)

        best_val = np.inf
        evals = 0

        for i in range(pop_size):

                break
            x = pop[i]
            val = func(x)
            evals += 1
            fitness[i] = val
            if val < best_val:
                best_val = val
                best_x = x.copy()

                candidates.remove(i)

                a, b, c = candidates[:3]
