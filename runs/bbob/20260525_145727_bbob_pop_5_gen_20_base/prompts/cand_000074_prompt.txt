class Optimizer:

        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

    def __call__(self, func):

        lb = func.bounds.lb
        ub = func.bounds.ub
        rng = self.rng
        budget = self.budget

        pop_size = max(4, min(10*dim, budget // 4))

        budget_de = int(0.8 * budget)
        if budget_de < pop_size:

        fitness = np.full(pop_size, np.inf)
        best_x = None
        best_val = np.inf
        evals = 0

        # initial evaluation
        for i in range(pop_size):
            if evals >= budget:
                break
            val = func(pop[i])
            evals += 1
            fitness[i] = val
            if val < best_val:
                best_val = val
                best_x = pop[i].copy()
                report_best(best_val, best_x)

        # DE parameters: exploitative
        F = 0.5
        CR = 0.9

        # main DE loop
        while evals < budget_de and evals < budget:

                if evals >= budget_de or evals >= budget:
                    break
                # choose two distinct random indices different from i
                candidates = list(range(pop_size))
                candidates.remove(i)
                rng.shuffle(candidates)
                a, b = candidates[:2]
                # DE/best/1

                mutant = np.clip(mutant, lb, ub)
