class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

        ub = np.asarray(func.bounds.ub)

        x0 = rng.uniform(lb, ub, size=dim)
        best_x = x0.copy()

        cs = (mueff + 2.0) / (dim + mueff + 5.0)

        p_s = np.zeros(dim)

        restart_threshold = int(0.25 * budget)
        while evals < budget:
            if no_improve_evals >= restart_threshold:
                # Decide restart type: 50% near best, 50% random
                if rng.rand() < 0.5:
                    # Reinitialize near best with small sigma
                    m = best_x.copy() + 0.1 * sigma * rng.randn(dim)

                    m = rng.uniform(lb, ub, size=dim)
                    sigma = (ub - lb).mean() / 4.0
                m = np.clip(m, lb, ub)
                C = np.eye(dim)
                p_c = np.zeros(dim)
                p_s = np.zeros(dim)
                no_improve_evals = 0
            lam_gen = min(lam, budget - evals)
            if lam_gen < 2:

            except np.linalg.LinAlgError:

                    best_value = vals[i]

                    no_improve_evals = 0
                    report_best(best_value, best_x)
                else:
                    no_improve_evals += 1

            # Update mean
            x_old = m.copy()
            m = w @ pop[idx[:mu]]
            delta = (m - x_old) / sigma
            # Update p_c

            C += 1e-10 * np.eye(dim)
        return best_value, best_x
