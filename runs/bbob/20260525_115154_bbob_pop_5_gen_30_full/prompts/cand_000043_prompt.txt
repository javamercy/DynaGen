from numpy.random import RandomState

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed

    def __call__(self, func):

        budget = self.budget

                CR = CR_memory[k] + 0.1 * rng.randn()
                CR = np.clip(CR, 0.0, 1.0)
