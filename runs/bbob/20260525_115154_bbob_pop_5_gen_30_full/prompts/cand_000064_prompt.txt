from numpy.random import RandomState

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = RandomState(seed)

    def __call__(self, func):
        dim = self.dim
        budget = self.budget
        rng = self.rng
        lb = func.bounds.lb
        ub = func.bounds.ub

        # population size
        pop_size = max(4 * dim, 3)
        if pop_size > budget // 2:
            pop_size = max(3, budget // 2)

            pop_size = 3

        archive = []
        archive_max = pop_size

            if evals >= budget:
                break
            fitness[i] = func(pop[i])
            evals += 1

                best_val = fitness[i]
                best_x = pop[i].copy()
                report_best(best_val, best_x)

        # JADE parameters
        mu_F = 0.5
        mu_CR = 0.5
        p_best = 0.2
        c = 0.1

        gen_max_restart = max(1, budget // (2 * pop_size))
        gen_no_improve = 0
        prev_best_val = best_val
        successful_F = []
        successful_CR = []

        generation = 0
        while evals < budget:
            generation += 1
            # generate F and CR for each individual
            F = np.clip(mu_F + mu_F * rng.standard_cauchy(pop_size), 0, 1)
            F = np.where(F <= 0, 0.1, F)
            CR = np.clip(mu_CR + 0.1 * rng.standard_cauchy(pop_size), 0, 1)
            CR = np.where(CR <= 0, 0.1, CR)

            for i in range(pop_size):
                if evals >= budget:
                    break

                candidates.remove(i)

                    partner = archive[archive_idx]

                mutant = pop[i] + F[i] * (pbest - pop[i]) + F[i] * (pop[a] - partner)
                mutant = np.clip(mutant, lb, ub)

                trial = pop[i].copy()

                        trial[j] = mutant[j]

                evals += 1
                if trial_fit < fitness[i]:

                    archive.append(pop[i].copy())
                    if len(archive) > archive_max:
                        archive.pop(0)
                    fitness[i] = trial_fit
                    pop[i] = trial

                    successful_CR.append(CR[i])
                    if trial_fit < best_val:

                        best_x = trial.copy()
                        report_best(best_val, best_x)
