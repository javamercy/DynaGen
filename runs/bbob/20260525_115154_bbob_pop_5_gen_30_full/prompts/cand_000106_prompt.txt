class Optimizer:

        self.dim = dim
        self.seed = seed

                break

                    fitness[i] = val

                        best_val = val
                        best_x = pop[i].copy()
                        report_best(best_val, best_x)

                mu_F = 0.5
                mu_CR = 0.5
                archive = []
                prev_best_val = best_val
