class Optimizer:

        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

        budget = self.budget
        rng = self.rng
        lb = func.bounds.lb
        ub = func.bounds.ub

            samples = np.zeros((n, d))

                samples[:, j] = (perm + rng.uniform(size=n)) / n
            return lb + samples * (ub - lb)
