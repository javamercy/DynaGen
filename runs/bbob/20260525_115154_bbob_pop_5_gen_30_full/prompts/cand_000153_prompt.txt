class Optimizer:

        self.dim = dim
        self.seed = seed

                best_x = pop[i].copy()

        mu_CR = 0.5

            sigma = start_sigma
            p_sigma = np.zeros(dim)

            d = 1.0 + dim / 2.0     # damping factor
            evals_used = 0
            best_val_internal = func(x)
            evals_used += 1

                y = x + sigma * z
                y = np.clip(y, lb, ub)
                val_y = func(y)
                evals_used += 1
                if val_y < best_val_internal:
                    best_val_internal = val_y
                    x = y.copy()

                    p_sigma = (1 - c) * p_sigma + np.sqrt(c * (2 - c)) * z
                    # update sigma

                gen_no_improve = 0

                gen_no_improve += 1

            restart = False

                    evals += local_evals

                        best_val = local_val
                        best_x = local_x.copy()

                pop_std = np.std(pop, axis=0)
                pop_std = np.maximum(pop_std, 1e-12)
                new_pop = rng.uniform(lb, ub, size=(pop_size, dim))
                new_pop[0] = best_x
                for i in range(1, pop_size):
                    if rng.rand() < 0.5:
                        noise = rng.normal(0, pop_std, dim)
                        new_pop[i] = np.clip(new_pop[i] + noise, lb, ub)
                    else:
                        radius = 0.1 * (ub - lb)
                        new_pop[i] = np.clip(best_x + rng.uniform(-radius, radius), lb, ub)
                pop = new_pop
                fitness = np.full(pop_size, np.inf)
                fitness[0] = best_val
                for i in range(1, pop_size):
                    if evals >= budget:
                        break
                    val = func(pop[i])
                    evals += 1
                    fitness[i] = val
                    if val < best_val:
                        best_val = val
                        best_x = pop[i].copy()
                mu_F = 0.5
                mu_CR = 0.5
                archive = []
                prev_best_val = best_val
                gen_no_improve = 0

        return best_val, best_x
