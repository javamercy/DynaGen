class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

    def __call__(self, func):

        ub = func.bounds.ub
        dim = self.dim

        rng = self.rng

                break
            x = points[i]
            f = func(x)
