import numpy as np

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

    def __call__(self, func):
        lb = np.array(func.bounds.lb)
        ub = np.array(func.bounds.ub)
        dim = self.dim
        budget = self.budget
        rng = self.rng

        pop_size = min(max(4 * dim, 20), budget // 2)
        pop_size = max(pop_size, 4)

        # Latin Hypercube Sampling
        def lhs(n, d, lb, ub):
            points = np.zeros((n, d))
            for j in range(d):
                perm = rng.permutation(n)
                for i in range(n):
                    points[i, j] = lb[j] + (perm[i] + rng.uniform()) * (ub[j] - lb[j]) / n
            return points

        best_f = np.inf

            if evals >= budget:
                break
            f = func(pop[i])
            evals += 1
            fitness[i] = f
            if f < best_f:
                best_f = f
                best_x = pop[i].copy()
                report_best(best_f, best_x)

        # main DE loop
        stagnation_counter = 0
        max_stagnation_gen = pop_size * 5

        use_cov_prob = 0.5

        while evals < budget:

            if stagnation_counter >= max_stagnation_gen and pop_size > 1:
                # restart: keep best, reinitialize rest with LHS
                new_pop = lhs(pop_size - 1, dim, lb, ub)
                pop = np.vstack([best_x, new_pop])
                # evaluate new individuals except best
                for i in range(1, pop_size):
                    if evals >= budget:
                        break
                    f = func(pop[i])
                    evals += 1
                    fitness[i] = f
                    if f < best_f:
                        best_f = f
                        best_x = pop[i].copy()
                        report_best(best_f, best_x)
                stagnation_counter = 0
                continue

            if pop_size >= 4:

                half = pop_size // 2
                best_half = pop[sorted_idx[:half]]

                cov = np.cov(best_half, rowvar=False)
                # make positive definite

                except np.linalg.LinAlgError:

            else:

                    break
