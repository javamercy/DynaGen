class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)

    def __call__(self, func):

        ub = func.bounds.ub

                break
            x = points[i]
            f = func(x)
            evals += 1
            pop_fitness[i] = f
            if f < best_f:

                best_x = x.copy()

        CR = 0.9

        step_size = 0.1 * (ub - lb)  # initial step size scaled by domain
        # Main DE loop
        while evals < self.budget:
            target_idx = self.rng.randint(pop_size)
            candidates = list(range(pop_size))
            candidates.remove(target_idx)
            if len(candidates) < 3:

            a, b, c = idx

            # Crossover
            trial = points[target_idx].copy()

                    trial[j] = mutant[j]
            # Clip to bounds
            trial = np.clip(trial, lb, ub)
            # Evaluate

            evals += 1
            if f_trial < pop_fitness[target_idx]:

                pop_fitness[target_idx] = f_trial

                candidate = np.clip(best_x + perturbation, lb, ub)
