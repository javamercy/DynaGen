class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        np.random.seed(seed)

        self.pop_size = max(4, min(4 * dim, budget // 2))

        budget = self.budget

                val = func(x)

                    best_x = x.copy()
                    report_best(best_val, best_x)

            val = func(x)
