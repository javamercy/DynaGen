import numpy as np

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        np.random.seed(seed)
        # population size based on dimension, but smaller to allow more generations
        self.pop_size = max(4, min(4 * dim, budget // 2))
        self.pop_size = max(self.pop_size, 1)
        self.max_generations = (budget - self.pop_size) // self.pop_size if self.pop_size > 0 else 0
        # restart threshold
        self.restart_threshold = max(5, 2 * dim)

    def __call__(self, func):

        ub = func.bounds.ub
        dim = self.dim
        pop_size = self.pop_size
        if pop_size <= 0:

            best_val = func(best_x)
            report_best(best_val, best_x)

                x = np.random.uniform(lb, ub, dim)
                val = func(x)
                evals += 1
                if val < best_val:
                    best_val = val
                    best_x = x.copy()
                    report_best(best_val, best_x)
            return best_val, best_x

        fitness = np.full(pop_size, np.inf)
        best_val = np.inf

        for i in range(pop_size):
            if evals >= self.budget:
                break
            x = pop[i]
            val = func(x)
            evals += 1
            fitness[i] = val
            if val < best_val:
                best_val = val
                best_x = x.copy()
                report_best(best_val, best_x)

        # parameters
        F = 0.5
        CR = 0.9
        no_improve = 0

        # main loop
        generation = 0
        while evals < self.budget and generation < self.max_generations:
            improved_this_gen = False
            for i in range(pop_size):
                if evals >= self.budget:
                    break
                # mutation: random differential
                candidates = list(range(pop_size))
                candidates.remove(i)
                if len(candidates) < 3:
                    continue
                a, b, c = np.random.choice(candidates, size=3, replace=False)
                mutant = pop[a] + F * (pop[b] - pop[c])
                mutant = np.clip(mutant, lb, ub)
                # crossover
                cross_points = np.random.rand(dim) < CR
                if not np.any(cross_points):
                    cross_points[np.random.randint(0, dim)] = True
                trial = np.where(cross_points, mutant, pop[i])

                    fitness[i] = val
                    improved_this_gen = True

                F *= 1.1
                F = min(F, 0.9)
                no_improve = 0

                F = max(F, 0.1)
                no_improve += 1

                if best_x is not None:

                        best_x = x.copy()
                        report_best(best_val, best_x)
                pop = new_pop

                F = 0.5
                no_improve = 0
            generation += 1
