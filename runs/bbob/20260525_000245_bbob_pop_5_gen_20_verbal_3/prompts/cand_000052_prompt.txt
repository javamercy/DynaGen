class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.rng = np.random.default_rng(seed)

        lb = func.bounds.lb
        ub = func.bounds.ub
        budget = self.budget
        rng = self.rng

        NP = max(4, min(20, budget // (dim + 1)))
        if NP < 4:

        best_val = np.inf

        for i in range(NP):
            if calls >= budget:
                break
            x = np.clip(pop[i], lb, ub)
            val = func(x)
            calls += 1
            pop_fitness[i] = val
            if val < best_val:
                best_val = val
                best_x = x.copy()
                report_best(best_val, best_x)

        # DE parameters
        F_start = 0.9
        F_end = 0.5
        CR_min = 0.5
        CR_max = 0.9
        CR_period = 10  # for sinusoidal variation
        generation = 0

        # Main loop
        while calls < budget:
            # Adaptive F
            if budget > 0:
                F = F_start - (F_start - F_end) * (generation / (budget // NP if NP > 0 else 1))
                F = np.clip(F, F_end, F_start)
            else:
                F = F_start
            # Sinusoidal CR
            CR = CR_min + (CR_max - CR_min) * (0.5 + 0.5 * np.sin(2 * np.pi * generation / CR_period))

                # Keep the best individual

                indices = [i for i in range(NP) if i != best_idx]
