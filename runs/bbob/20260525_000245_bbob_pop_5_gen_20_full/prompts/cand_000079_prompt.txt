        self.dim = dim
        self.seed = seed

        ub = func.bounds.ub
        n = self.dim
        budget = self.budget

        report_best(best_val, best_x)

                x = np.random.uniform(lb, ub, n)
                val = func(x)
                calls += 1
                if val < best_val:
                    best_val = val
                    best_x = x
                    report_best(best_val, best_x)
            return best_val, best_x

        lambda_ = max(2, lambda_)

        ps = np.zeros(n)

                pc = np.zeros(n)
                ps = np.zeros(n)
                no_improve_iter = 0
                local_refine_counter = 0

        return best_val, best_x
