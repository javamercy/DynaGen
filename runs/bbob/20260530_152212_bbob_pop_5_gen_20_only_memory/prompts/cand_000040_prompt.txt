class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        # dynamic population size
        self.pop_size = max(3, min(budget // 10, 5 * dim))

    def __call__(self, func):

        lb = func.bounds.lb
        ub = func.bounds.ub
        dim = self.dim
        budget = self.budget
        pop_size = self.pop_size

        F = 0.5
        CR = 0.5

                break
            x = pop[i]
            val = func(x)
            evals += 1
            fitness[i] = val

                best_val = val
                best_x = x.copy()
                report_best(best_val, best_x)

        # main DE loop
        while evals < budget:
            # DE/rand/1 mutation (not best/1, to encourage exploration)

                    break

                idxs = [j for j in range(pop_size) if j != i]
                a, b, c = np.random.choice(idxs, 3, replace=False)
                mutant = pop[a] + F * (pop[b] - pop[c])
                mutant = np.clip(mutant, lb, ub)

                trial = np.where(np.random.rand(dim) < CR, mutant, pop[i])
                trial[j_rand] = mutant[j_rand]
                val = func(trial)
                evals += 1

                    pop[i] = trial
                    fitness[i] = val
                    if val < best_val:
                        best_val = val
                        best_x = trial.copy()
                        report_best(best_val, best_x)

            # local refinement around best (adaptive step size)
            if evals < budget:
                sigma = 0.1 * (ub - lb).mean() * (1 - evals / budget)
                for _ in range(min(5, budget - evals)):
                    candidate = best_x + sigma * np.random.randn(dim)
                    candidate = np.clip(candidate, lb, ub)
                    val = func(candidate)
                    evals += 1

                                best_val = v

                        pop[idx] = candidate
                        fitness[idx] = val
                        break  # after improvement, continue DE loop

                        sigma *= 0.9

                            best_x = new_x.copy()
                            report_best(best_val, best_x)

        while self.evals < self.budget:
            improved = False
            order = np.random.permutation(self.dim)
