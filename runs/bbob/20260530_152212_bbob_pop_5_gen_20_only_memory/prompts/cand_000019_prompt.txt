class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed
        self.pop_size = max(3, min(budget // 2, 10 * dim))

    def __call__(self, func):

        lb = func.bounds.lb
        ub = func.bounds.ub
        budget = self.budget
        dim = self.dim
        pop_size = self.pop_size

        CR = 0.9
        sigma0 = 0.2 * (ub - lb).mean()

        best_x = None
        best_val = np.inf

        # Evaluate initial population
        for i in range(pop_size):
            if evals >= budget:
                break
            x = pop[i]
            val = func(x)
            evals += 1
            fitness[i] = val
            if val < best_val:
                best_val = val
                best_x = x.copy()

        # Main loop
        generation = 0
        while evals < budget:
            # Adaptive mutation factor
            F = F0 * (1 - evals / budget) ** 0.5
            # Local refinement step (every generation)
            if generation > 0 and evals < budget:
                sigma = sigma0 * (1 - evals / budget) ** 0.5
                trial = best_x + np.random.normal(0, sigma, size=dim)
                trial = np.clip(trial, lb, ub)
                val = func(trial)
                evals += 1
                if val < best_val:

                    best_x = trial.copy()

                    pop[idx] = trial
                    fitness[idx] = val

            for i in range(pop_size):
                if evals >= budget:
