import numpy as np

class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed

    def __call__(self, func):
        rng = np.random.RandomState(self.seed)
        lb = func.bounds.lb
        ub = func.bounds.ub
        dim = self.dim
        budget = self.budget

        if NP > budget:
            NP = budget
        NP = max(1, NP)

        fitness = np.array([func(pop[i]) for i in range(NP)])
        evals = NP

        best_x = pop[best_idx].copy()
        report_best(best_val, best_x)

            return best_val, best_x

        H = 7

        M_CR = np.full(H, 0.9)
        p = 0

        gen = 0
        while evals < budget and gen < max_gen + 1:
            # Restart interval = 8 generations
            if gen > 0 and gen % 8 == 0:
                std_dim = np.std(pop, axis=0)
                diversity = np.mean(std_dim) / np.mean(ub - lb)
                if diversity < 1e-6:
                    remaining = budget - evals
                    needed = NP - 1
                    if remaining >= needed:
                        pop[0] = best_x.copy()
                        new_pop = lb + (ub - lb) * rng.rand(NP - 1, dim)
                        for i in range(1, NP):
                            pop[i] = new_pop[i-1]
                        for i in range(1, NP):

                            evals += 1
                            if f < best_val:
                                best_val = f
                                best_x = pop[i].copy()
                                report_best(best_val, best_x)
                        M_F[:] = 0.5
                        M_CR[:] = 0.9
                        p = 0

            S_F = []
            S_CR = []
            delta_f = []
            for i in range(NP):
                if evals >= budget:
                    break
                r = rng.randint(H)
                F = M_F[r] + 0.1 * rng.standard_cauchy()
                if F <= 0:
                    F = 0.1
                elif F > 1:
                    F = 1.0
                CR = M_CR[r] + 0.1 * rng.randn()
                CR = np.clip(CR, 0, 1)

                candidates.remove(i)

                    continue

                a, b, c = pop[idx[0]], pop[idx[1]], pop[idx[2]]
                mutant = a + F * (b - c)

                trial = pop[i].copy()

                for j in range(dim):
                    if rng.rand() < CR or j == j_rand:

                trial = np.clip(trial, lb, ub)

                evals += 1
                if f_trial < fitness[i]:

                    S_F.append(F)
                    S_CR.append(CR)
                    pop[i] = trial
                    fitness[i] = f_trial

                p = (p + 1) % H
            gen += 1

        return best_val, best_x
