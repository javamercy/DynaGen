        ub = func.bounds.ub

        coord_order = np.arange(dim)
        while evals < self.budget:
            for coord in coord_order:
                if evals >= self.budget:
                    break

                candidate = np.clip(candidate, lb, ub)
                f_candidate = func(candidate)
                evals += 1
