class Optimizer:
    def __init__(self, budget: int, dim: int, seed: int):
        self.budget = budget
        self.dim = dim
        self.seed = seed

        ub = func.bounds.ub
        dim = self.dim

                diversity = np.mean(std_dim) / np.mean(ub - lb)
                if diversity < 1e-6:
                    remaining = budget - evals
                    needed = NP - 1  # keep best
                    if remaining >= needed:

                        for i in range(1, NP):

                                report_best(best_val, best_x)
                        M_F[:] = 0.5
                        M_CR[:] = 0.9

            S_F = []
            S_CR = []
            delta_f = []

                r = rng.randint(H)

                if F <= 0:
                    F = 0.1
                elif F > 1:

                CR = M_CR[r] + 0.1 * rng.randn()

                candidates = list(range(NP))

                mutant = a + F * (b - c)

                # Binomial crossover
                trial = pop[i].copy()

                for j in range(dim):
                    if rng.rand() < CR or j == j_rand:
                        trial[j] = mutant[j]
                trial = np.clip(trial, lb, ub)

                evals += 1

                    S_F.append(F)
                    S_CR.append(CR)

                    fitness[i] = f_trial

                        best_x = trial.copy()

                p = (p + 1) % H
            gen += 1

        return best_val, best_x
