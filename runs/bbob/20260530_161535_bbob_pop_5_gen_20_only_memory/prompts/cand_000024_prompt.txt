class Optimizer:
    def __init__(self, budget, dim, seed):

        self.dim = dim
        self.seed = seed
        self.rng = np.random.RandomState(seed)
        self.lam = 4 + int(3 * np.log(dim))
        self.mu = self.lam // 2
        self.weights = np.log(self.mu + 0.5) - np.log(np.arange(1, self.mu + 1))
        self.weights /= self.weights.sum()
