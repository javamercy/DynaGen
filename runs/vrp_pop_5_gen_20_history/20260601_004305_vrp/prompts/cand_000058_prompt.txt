def solve_vrp(distance_matrix, truck_count):

    n = distance_matrix.shape[0]
    customers = list(range(1, n))
    m = n - 1
    if truck_count >= m:
        routes = [[0, i, 0] for i in customers]
        while len(routes) < truck_count:
            routes.append([0, 0])
        return routes

    tour = []
    visited = [False] * n

    current = 0

                    best_dist = d
                    best = v

        visited[best] = True
        current = best

            else:
                seg_dist[l][r] = acc + distance_matrix[tour[r - 1]][0]
