def solve_vrp(distance_matrix, truck_count):

    customers = list(range(1, n))

    if truck_count >= m:

            routes.append([0, 0])
        return routes

    tour = []
    visited = [False] * n
    visited[0] = True

        best = -1

            if not visited[v]:
                d = distance_matrix[current][v]
                if d < best_dist or (d == best_dist and v < best):
                    best_dist = d
                    best = v

        visited[best] = True
        current = best
