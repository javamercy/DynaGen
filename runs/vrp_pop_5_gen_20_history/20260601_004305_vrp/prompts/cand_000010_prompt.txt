        best = -1

            if not visited[v]:
                d = distance_matrix[current][v]
                if d < best_dist or (d == best_dist and v < best):
                    best_dist = d
                    best = v
        tour.append(best)
        visited[best] = True
        current = best

            if r == l+1:

            else:
                route_dist = acc + distance_matrix[custs[r-1]][0]
            seg_dist[l][r] = route_dist
