                for pos in range(1, len(route)):

                    new_max = new_dist

                        best_inc = new_max

                        best_pos = pos

            if iterations_since_improvement >= improvement_threshold and current_tenure < tabu_tenure_max:
                current_tenure += 1
                iterations_since_improvement = 0
    
    return best_routes
