            best_route_idx = -1
            best_pos = -1

                    best_local_delta = float('inf')
                    best_local_pos = -1

                        nex = route[pos]
                        delta = distance_matrix[prev, cust] + distance_matrix[cust, nex] - distance_matrix[prev, nex]
                        if delta < best_local_delta:
                            best_local_delta = delta
                            best_local_pos = pos
                    # compute new route distance and new max
                    current_route_dist = route_dist(route)
                    new_route_dist = current_route_dist + best_local_delta
                    other_max = 0.0
                    for j, r in enumerate(routes):
                        if j == idx:
                            continue
