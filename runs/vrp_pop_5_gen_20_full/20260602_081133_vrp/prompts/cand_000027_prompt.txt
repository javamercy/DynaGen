def solve_vrp(distance_matrix, truck_count):
    n = distance_matrix.shape[0]
    customer_count = n - 1
    
    # If truck_count >= customer_count, each customer gets its own truck, rest empty
    if truck_count >= customer_count:
        routes = [[0, 0] for _ in range(truck_count)]
        for i in range(1, n):
            routes[i-1] = [0, i, 0]
        try:
            report_best_vrp(routes)
        except:
            pass
        return routes

    # Initialize routes as empty: each starts and ends at depot
    routes = [[0, 0] for _ in range(truck_count)]
    assigned = [False] * n  # depot always assigned
    assigned[0] = True
    unassigned = list(range(1, n))

    # Helper to compute route distance
    def compute_distance(route):
        if len(route) <= 2:
            return 0.0
        total = 0.0
        for i in range(len(route)-1):
            total += distance_matrix[route[i]][route[i+1]]
        return total

    # Function to find best insertion position in a route for a customer
    def best_insertion(route, cust):
        best_pos = 1
        best_increase = float('inf')
        for pos in range(1, len(route)):
            prev = route[pos-1]
            next_ = route[pos]
            increase = distance_matrix[prev][cust] + distance_matrix[cust][next_] - distance_matrix[prev][next_]
            if increase < best_increase - 1e-9:
                best_increase = increase
                best_pos = pos
        return best_pos, best_increase

    # Step 1: Select seeds using farthest-first from depot and each other
    seeds = []
    # First seed: farthest from depot
    if customer_count > 0:
        max_dist = -1
        seed = -1
        for cust in range(1, n):
            d = distance_matrix[0][cust]
            if d > max_dist + 1e-9:
                max_dist = d
                seed = cust
        seeds.append(seed)
        # Subsequent seeds: farthest from any chosen seed
        for _ in range(1, truck_count):
            max_min_dist = -1
            new_seed = -1
            for cust in range(1, n):
                if cust in seeds:
                    continue
                min_dist_to_seeds = min(distance_matrix[cust][s] for s in seeds)
                if min_dist_to_seeds > max_min_dist + 1e-9:
                    max_min_dist = min_dist_to_seeds
                    new_seed = cust
            if new_seed != -1:
                seeds.append(new_seed)
            else:
                break
        # If fewer seeds than trucks, fill with customers not yet seeds (or any)
        for cust in range(1, n):
            if len(seeds) >= truck_count:
                break
            if cust not in seeds:
                seeds.append(cust)
    else:
        seeds = []

    # Assign each seed to its own route, remove from unassigned
    for t in range(min(truck_count, len(seeds))):
        cust = seeds[t]
        routes[t] = [0, cust, 0]
        unassigned.remove(cust)

    # Remaining trucks (if any) stay empty

    # Actually we want balanced routes, so we assign to the route that results in smallest new max distance?
    # For simplicity, assign to the route with smallest insertion cost (like greedy nearest insertion)
    while unassigned:
        best_cust = None
        best_route = None
        best_pos = None
        best_increase = float('inf')
        for cust in unassigned:
            for t in range(truck_count):
                if len(routes[t]) == 2:  # empty route, insertion at position 1
                    increase = 2 * distance_matrix[0][cust]  # 0->cust->0
                    if increase < best_increase - 1e-9:
                        best_increase = increase
                        best_cust = cust
                        best_route = t
                        best_pos = 1
                else:
                    pos, inc = best_insertion(routes[t], cust)
                    if inc < best_increase - 1e-9:
                        best_increase = inc
                        best_cust = cust
                        best_route = t
                        best_pos = pos
        if best_cust is not None:
            routes[best_route] = routes[best_route][:best_pos] + [best_cust] + routes[best_route][best_pos:]
            unassigned.remove(best_cust)
        else:
            break

    # Compute initial distances
    distances = [compute_distance(r) for r in routes]
    max_dist = max(distances)
    try:
        report_best_vrp(routes)
    except:
        pass

    # Improvement phase
    max_iter = n * truck_count * 2
    for _ in range(max_iter):
        improved = False

        # Intra-route 2-opt
        for t in range(truck_count):
            route = routes[t]
            if len(route) <= 3:
                continue
            best_route = route[:]
            best_dist = compute_distance(route)
            for i in range(1, len(route)-2):

                    new_route = route[:i] + route[i:j+1][::-1] + route[j+1:]
                    new_dist = compute_distance(new_route)
                    if new_dist < best_dist - 1e-9:
                        best_dist = new_dist
                        best_route = new_route
            if best_dist < compute_distance(routes[t]) - 1e-9:
                routes[t] = best_route
                improved = True
        if improved:
            distances = [compute_distance(r) for r in routes]
            max_dist = max(distances)
            try:
                report_best_vrp(routes)
            except:
                pass
            continue

        best_move = None
        best_reduction = 0.0

            if len(src_route) <= 2:
                continue

                        continue

                    if len(dst_route) == 2:
                        pos = 1

                    else:

                    new_dst = dst_route[:pos] + [cust] + dst_route[pos:]

                    new_dist_src = compute_distance(new_src)
                    new_dist_dst = compute_distance(new_dst)
                    other_dists = [distances[i] for i in range(truck_count) if i not in (src, dst)]
                    new_max = max(new_dist_src, new_dist_dst, *other_dists)
                    reduction = max_dist - new_max
                    if reduction > best_reduction + 1e-9:
                        best_reduction = reduction
                        best_move = (src, dst, new_src, new_dst)
        if best_reduction > 1e-9:
            src, dst, new_src, new_dst = best_move
            routes[src] = new_src
            routes[dst] = new_dst
            distances = [compute_distance(r) for r in routes]
            max_dist = max(distances)
            try:
                report_best_vrp(routes)
            except:
                pass
            improved = True
            continue

        # Inter-route 2-opt* (swap suffixes)
        best_move = None
        best_reduction = 0.0
        for t1 in range(truck_count):
            for t2 in range(t1+1, truck_count):
                route1 = routes[t1]
                route2 = routes[t2]
                if len(route1) <= 2 or len(route2) <= 2:
                    continue
                for i in range(1, len(route1)-2):
                    for j in range(1, len(route2)-2):
                        # New routes: swap suffixes after positions i and j
                        new_route1 = route1[:i+1] + route2[j+1:]
                        new_route2 = route2[:j+1] + route1[i+1:]
                        # Ensure both end with depot (they should as suffixes include depot)
                        new_dist1 = compute_distance(new_route1)
                        new_dist2 = compute_distance(new_route2)
                        other_dists = [distances[k] for k in range(truck_count) if k not in (t1, t2)]
                        new_max = max(new_dist1, new_dist2, *other_dists)
                        reduction = max_dist - new_max
                        if reduction > best_reduction + 1e-9:
                            best_reduction = reduction
                            best_move = (t1, t2, new_route1, new_route2)
        if best_reduction > 1e-9:
            t1, t2, new_route1, new_route2 = best_move
            routes[t1] = new_route1
            routes[t2] = new_route2
            distances = [compute_distance(r) for r in routes]
            max_dist = max(distances)
            try:
                report_best_vrp(routes)
            except:
                pass
            improved = True
            continue

        if not improved:
            break

    return routes
