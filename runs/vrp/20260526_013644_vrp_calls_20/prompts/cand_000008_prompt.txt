import numpy as np

def solve_vrp(distance_matrix, truck_count):
    n = distance_matrix.shape[0]
    depot = 0
    customers = list(range(1, n))
    unassigned = set(customers)
    routes = [[depot, depot] for _ in range(truck_count)]
    
    # Helper to compute route distance
    def route_dist(route):
        d = 0
        for i in range(len(route)-1):
            d += distance_matrix[route[i], route[i+1]]
        return d
    
    # Helper to compute insertion cost and best position for a customer in a route
    def best_insertion(cust, route):
        best_cost = float('inf')
        best_pos = -1
        # route has at least two entries (depot at start and end)
        for pos in range(1, len(route)):
            i = route[pos-1]
            j = route[pos]
            cost = distance_matrix[i, cust] + distance_matrix[cust, j] - distance_matrix[i, j]
            if cost < best_cost:
                best_cost = cost
                best_pos = pos
        return best_cost, best_pos
    
    # Regret insertion construction
    while unassigned:
        # Compute regret for each unassigned customer
        best_regret = -1
        best_cust = -1
        best_route_idx = -1
        best_pos = -1
        best_cost_for_cust = float('inf')
        
        for cust in unassigned:
            costs = []
            for r_idx, route in enumerate(routes):
                cost, pos = best_insertion(cust, route)
                costs.append((cost, r_idx, pos))
            # Sort by cost
            costs.sort(key=lambda x: x[0])
            if len(costs) == 1:
                # Only one feasible route
                regret = costs[0][0] * 2  # large regret
            else:
                regret = costs[1][0] - costs[0][0]
            # Tie-breaking: if regret same, prefer higher best cost
            if regret > best_regret or (regret == best_regret and costs[0][0] > best_cost_for_cust):
                best_regret = regret
                best_cust = cust
                best_cost_for_cust = costs[0][0]
                best_route_idx = costs[0][1]
                best_pos = costs[0][2]

                    best_pos = costs[0][2]

        route = routes[best_route_idx]
        route.insert(best_pos, best_cust)
        unassigned.remove(best_cust)

    best_routes = [list(r) for r in routes]
    best_max = max(route_dist(r) for r in best_routes)

    # Limit iterations
    n_customers = n - 1
    max_iters = 2 * n_customers
    for _ in range(max_iters):

        for r_idx, route in enumerate(routes):

                continue
            # Consider each customer in this route (excluding depots)
            customers_in_route = route[1:-1]
            for cust in customers_in_route:
                # Remove cust from current route
                new_route = [x for x in route if x != cust]

                for other_idx, other_route in enumerate(routes):
                    if other_idx == r_idx:
                        continue
                    cost, pos = best_insertion(cust, other_route)
                    # Create candidate routes
                    candidate_routes = [list(r) for r in routes]
                    candidate_routes[r_idx] = new_route
                    other_new = list(other_route)
                    other_new.insert(pos, cust)
                    candidate_routes[other_idx] = other_new
                    # Compute max distance
                    dists = [route_dist(r) for r in candidate_routes]
                    new_max = max(dists)
                    if new_max < best_max:
                        best_max = new_max
                        routes = candidate_routes
                        improved = True
                        break  # restart iteration
                if improved:
                    break
            if improved:
                break
        if not improved:

            for r_idx, route in enumerate(routes):
                if len(route) <= 4:
                    continue

                n_nodes = len(route)
                best_imp = False
                for i in range(1, n_nodes-2):
                    for j in range(i+1, n_nodes-1):

                r.append(0)
