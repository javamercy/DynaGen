def route_distance(route, dm):
    return sum(dm[route[i], route[i+1]] for i in range(len(route)-1))

    while len(routes) > truck_count:
        best_saving = -1e9

            for j in range(i+1, len(routes)):

                last_i = r_i[-2]

                if s1 > best_saving:
                    best_saving = s1
