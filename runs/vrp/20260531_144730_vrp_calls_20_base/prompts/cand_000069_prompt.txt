                    regret = second_val - best_val if second_val != INF else INF
                    candidates.append((-regret, c, best_route, best_pos, best_val))
                if not candidates:
                    break
                max_regret = max(x[0] for x in candidates)
                top = [x for x in candidates if abs(x[0] - max_regret) < 1e-12]
                if random.random() < 0.3 and len(top) > 1:
                    chosen = random.choice(top)
                else:
                    top.sort(key=lambda x: x[1])
                    chosen = top[0]
                _, c, best_route, best_pos, _ = chosen
                route = routes[best_route]
                route.insert(best_pos, c)
                route_dists[best_route] = route_dist(route)
                unrouted.remove(c)
            cur_max = max(route_dists)
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_routes = [route[:] for route in routes]
                no_improve_streak = 0
            else:
                routes = saved_routes
                route_dists = saved_dists
                no_improve_streak += 1
        else:
            no_improve_streak += 1
        # More frequent restart
        if no_improve_streak >= max_iter // 8:  # increased frequency
            def random_giant_tour():
                unvisited = set(customers)
                tour = [0]
                current = 0
                while unvisited:
                    min_dist = min(distance_matrix[current, x] for x in unvisited)
                    candidates = [x for x in unvisited if distance_matrix[current, x] == min_dist]
                    next_node = random.choice(candidates)
                    tour.append(next_node)
                    unvisited.remove(next_node)
                    current = next_node
                tour.append(0)
                return tour[1:-1]

            def dp_split(giant_tour):
                n_c = len(giant_tour)
                if n_c == 0:
                    return [[0, 0] for _ in range(truck_count)]
                K = min(truck_count, n_c)
                seg = [[0.0] * n_c for _ in range(n_c)]
                for i in range(n_c):
                    d = distance_matrix[0, giant_tour[i]]
                    seg[i][i] = d + distance_matrix[giant_tour[i], 0]
                    for j in range(i+1, n_c):
                        d += distance_matrix[giant_tour[j-1], giant_tour[j]]
                        seg[i][j] = d + distance_matrix[giant_tour[j], 0]
                dp = [[INF] * (n_c + 1) for _ in range(K + 1)]
                parent = [[-1] * (n_c + 1) for _ in range(K + 1)]
                dp[0][0] = 0.0
                for k in range(1, K+1):
                    for i in range(k, n_c+1):
                        for j in range(k-1, i):
                            cand = max(dp[k-1][j], seg[j][i-1])
                            if cand < dp[k][i]:
                                dp[k][i] = cand
                                parent[k][i] = j
                new_routes = []
                k = K
                i = n_c
                while k > 0:
                    j = parent[k][i]
                    if j == -1:
                        break
                    segment = giant_tour[j:i]
                    new_routes.append([0] + segment + [0])
                    i = j
                    k -= 1
                new_routes.reverse()
                while len(new_routes) < truck_count:
                    new_routes.append([0, 0])
                return new_routes

            giant = random_giant_tour()
            routes = dp_split(giant)
            route_dists = [route_dist(r) for r in routes]
            for r_idx in range(truck_count):
                routes[r_idx] = two_opt(routes[r_idx])
                route_dists[r_idx] = route_dist(routes[r_idx])
            report_best_vrp(routes)
            cur_max = max(route_dists)
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_routes = [route[:] for route in routes]
            no_improve_streak = 0

    return best_routes
