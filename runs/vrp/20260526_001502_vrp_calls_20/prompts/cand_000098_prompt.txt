def solve_vrp(distance_matrix: np.ndarray, truck_count: int) -> list[list[int]]:
    n = distance_matrix.shape[0]
    if n == 1:
        return [[0, 0] for _ in range(truck_count)]

    # Helper functions
    def route_distance(route):
        if len(route) <= 1:
            return 0.0
        d = 0.0
        for i in range(len(route)-1):
            d += distance_matrix[route[i], route[i+1]]
        return d

    # Clarke-Wright savings initialization
    # Start with one route per customer
    routes = [[0, i, 0] for i in range(1, n)]
    # If we already have enough or fewer routes, handle later
    if len(routes) > truck_count:
        # Compute savings
        savings = []
        for i in range(1, n):
            for j in range(i+1, n):
                s = distance_matrix[0, i] + distance_matrix[0, j] - distance_matrix[i, j]
                savings.append((s, i, j))
        savings.sort(reverse=True)  # descending savings

        # Merge until we have exactly truck_count routes
        for s, i, j in savings:
            if len(routes) == truck_count:
                break
            # Find routes containing i and j
            idx_i = idx_j = -1
            for idx, route in enumerate(routes):
                if i in route:
                    idx_i = idx
                if j in route:
                    idx_j = idx
            if idx_i == idx_j or idx_i == -1 or idx_j == -1:
                continue
            route_i = routes[idx_i]
            route_j = routes[idx_j]
            # Check if i is at end of route_i and j at start of route_j
            if route_i[-2] == i and route_j[1] == j:
                new_route = route_i[:-1] + route_j[1:]
                # Remove route_j (higher index)
                if idx_i < idx_j:
                    routes[idx_i] = new_route
                    routes.pop(idx_j)
                else:
                    routes[idx_j] = new_route
                    routes.pop(idx_i)
            # Check reverse: j at end of route_j and i at start of route_i
            elif route_j[-2] == j and route_i[1] == i:
                new_route = route_j[:-1] + route_i[1:]
                if idx_i < idx_j:
                    routes[idx_j] = new_route
                    routes.pop(idx_i)
                else:
                    routes[idx_i] = new_route
                    routes.pop(idx_j)

    while len(routes) < truck_count:

                        for i in range(1, len(route2))
                    )
