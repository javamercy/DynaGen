        if len(route) <= 1:
            return 0.0

            d += distance_matrix[route[i], route[i+1]]
        return d

        best_cost = float('inf')

            cost = distance_matrix[route[i-1], node] + distance_matrix[node, route[i]] - distance_matrix[route[i-1], route[i]]
            if cost < best_cost:
                best_cost = cost
                best_pos = i
        return best_cost, best_pos
