```python
                best_delta = 0
                best_move = None
                for r1 in range(truck_count):
                    for r2 in range(r1+1, truck_count):
                        route1 = routes[r1]
                        route2 = routes[r2]
                        if len(route1) <= 3 or len(route2) <= 3:
                            continue
                        for i in range(1, len(route1)-1):
                            for j in range(1, len(route2)-1):
                                # cross: route1[0..i] + route2[j..-1]  and  route2[0..j] + route1[i..-1]
                                new_route1 = route1[:i] + route2[j:]
                                new_route2 = route2[:j] + route1[i:]
                                # ensure start and end with depot? They already do.
                                new_len1 = route_length(new_route1)
                                new_len2 = route_length(new_route2)
                                other_lens = [lengths[k] for k in range(truck_count) if k not in (r1, r2)]
                                new_max_candidate = max(new_len1, new_len2, *other_lens)
                                if new_max_candidate < current_max - 1e-12:
                                    delta = current_max - new_max_candidate
                                    if delta > best_delta:
                                        best_delta = delta
                                        best_move = (r1, i, r2, j, new_route1, new_route2)
                if best_move is not None:
                    r1, i, r2, j, new_route1, new_route2 = best_move
                    routes[r1] = new_route1
                    routes[r2] = new_route2
                    current_max = current_max - best_delta
                    improved = True
                    stagnation = 0
                    if current_max < best_max:
                        best_max = current_max
                        best_routes = [r[:] for r in routes]
                        report_best_vrp(routes)

            if not improved:
                # 4. Intra-route 2-opt (same as parent)
                for r_idx in range(truck_count):
                    route = routes[r_idx]
                    if len(route) <= 3:
                        continue
                    improved_intra = False
                    for i in range(1, len(route)-2):
                        for k in range(i+1, len(route)-1):
                            new_route = route[:i] + route[i:k+1][::-1] + route[k+1:]
                            if route_length(new_route) < route_length(route) - 1e-12:
                                route[:] = new_route
                                improved_intra = True
                                break
                        if improved_intra:
                            break
                    if improved_intra:
                        improved = True
                        stagnation = 0
                        new_max = max_route_len(routes)
                        if new_max < current_max - 1e-12:
                            current_max = new_max
                            if current_max < best_max:
                                best_max = current_max
                                best_routes = [r[:] for r in routes]
                                report_best_vrp(routes)
                        break

            if not improved:
                stagnation += 1
                if stagnation >= perturbation_threshold:
                    # Light perturbation: move 5% customers randomly
                    customers = list(range(1, n))
                    random.shuffle(customers)
                    num_perturb = max(1, n // 20)
                    for cust in customers[:num_perturb]:
                        for r_idx, route in enumerate(routes):
                            if cust in route:
                                routes[r_idx] = [x for x in route if x != cust]
                                break
                        r_idx = random.randrange(truck_count)
                        pos = random.randrange(1, len(routes[r_idx]))
                        routes[r_idx].insert(pos, cust)
                    current_max = max_route_len(routes)
                    stagnation = 0
                    improved = True

        if current_max < best_max:
            best_max = current_max
            best_routes = [r[:] for r in routes]
            report_best_vrp(routes)

    if best_routes is None:
        best_routes = routes
    return best_routes
```
